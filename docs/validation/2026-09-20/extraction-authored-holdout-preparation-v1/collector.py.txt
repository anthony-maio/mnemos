from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
temp = Path('C:/Users/Ant/AppData/Local/Temp')
base = repo / 'docs/validation/2026-09-20'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()

def archive(source, name, terminal, readme):
    prep = json.loads((source / 'preparation.json').read_bytes())
    for path, digest in prep['files'].items():
        assert sha(source / path) == digest
    names = list(prep['files']) + ['preparation.json']
    if terminal:
        state = json.loads((source / 'status.json').read_bytes())
        assert state['status'] == 'completed' and state['processed'] == 36 and state['errors'] == 0
        assert sha(source / 'observations.jsonl') == state['observations_sha256']
        assert sha(source / 'preparation.json') == state['preparation_sha256']
        names += ['status.json', 'observations.jsonl', 'stdout.log', 'stderr.log']
    out = base / name
    out.mkdir(exist_ok=False)
    (out / '.gitattributes').write_bytes(b'** -text -whitespace\n')
    for name in names:
        shutil.copyfile(source / name, out / name)
    if terminal:
        rows = [json.loads(line) for line in (source / 'observations.jsonl').read_text().splitlines()]
        assert len({(r['case_id'], r['arm'], r['repetition']) for r in rows}) == 36
        assert all(r['status'] == 'runtime_valid' for r in rows)
        review = [{'case_id': r['case_id'], 'repetition': r['repetition'], 'proposals': r['proposals']} for r in rows]
        (out / 'proposal-review.json').write_text(json.dumps(review, indent=2) + '\n')
    shutil.copyfile(__file__, out / 'collector.py.txt')
    files = {p.name: {'bytes': p.stat().st_size, 'sha256': sha(p)} for p in sorted(out.iterdir()) if p.is_file()}
    (out / 'evidence-index.json').write_text(json.dumps({'files': files}, indent=2) + '\n')
    (out / 'README.md').write_text(readme, encoding='utf-8', newline='\n')
    print(json.dumps({'archive': str(out), 'indexed_files': len(files)}))
