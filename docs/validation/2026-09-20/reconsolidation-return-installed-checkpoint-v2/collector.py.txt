from pathlib import Path
import hashlib, json, shutil

repo = Path(__file__).resolve().parents[1]
temp = Path('C:/Users/Ant/AppData/Local/Temp')
source = temp / 'mnemos-installed-checks-reconsolidation-return-v2'
record = json.loads((source / 'installation.json').read_bytes())
contract = json.loads((source / 'contract.json').read_bytes())
assert record['status'] == contract['status'] == 'passed'
assert record['bindings_after']['commit'] == 'd93789250ed1cd9aac31a58da1cb932e93b41582'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
assert record['runner_sha256'] == sha(repo / '.venv/run_combined_installed_check_v50.py')
assert record['bindings_before']['checks_sha256'] == sha(temp / 'mnemos-package-checks-reconsolidation-return-v1/verification.json')
for command in record['commands']:
    for stream in ('stdout', 'stderr'):
        assert sha(source / (command['name'] + '.' + stream + '.log')) == command[stream + '_sha256']
out = repo / 'docs/validation/2026-09-20/reconsolidation-return-installed-checkpoint-v2'
assert not (out / 'evidence-index.json').exists()
out.mkdir(exist_ok=True)
(out / '.gitattributes').write_bytes(b'** -text -whitespace\n')
for path in source.iterdir():
    if path.is_file() and path.suffix in {'.json', '.xml', '.log'}:
        shutil.copyfile(path, out / path.name)
for name in ['run_combined_installed_check_v50.py', 'installed_contract_probe_v4.py', 'verify_claude_stop_replay_v2.py']:
    shutil.copyfile(repo / '.venv' / name, out / (name + '.txt'))
shutil.copyfile(__file__, out / 'collector.py.txt')
summary = record['contract']['junit_summary']
assert any('test_direct_review_returns_authoritative_metadata_after_captured_merge' in str(x) for x in json.loads((source / 'expected-contract-cases.json').read_bytes()))
(out / 'README.md').write_text(f'''# Reconsolidation return installed-package checkpoint

Frozen product d937892 passes {summary['tests'] - summary['skipped']} installed cases
with {summary['skipped']} skips. The isolated runner verifies each selected test
outcome against the frozen source JUnit, along with installed module origins,
wheel bytes, input hashes and the replay contract. Captured correction merge and direct-return reconsolidation regressions
are included. The earlier v1 selection omitted the reconsolidation test file;
that narrower result is preserved separately. Counts and individual outcomes are preserved here.

This run uses fresh output, environment and replay paths.

This establishes the selected installed contract, not complete native-host
coverage, full-module quality, or pilot acceptance. The separate Hermes v35
archive proves its bounded installed correction/forgetting lifecycle. Nothing
was installed into live host configuration or published.
''')
files = {p.name: {'bytes': p.stat().st_size, 'sha256': sha(p)} for p in sorted(out.iterdir()) if p.is_file()}
(out / 'evidence-index.json').write_text(json.dumps({'files': files}, indent=2) + '\n')
print(json.dumps({'archive': str(out), 'files': len(files), 'junit': summary}))
