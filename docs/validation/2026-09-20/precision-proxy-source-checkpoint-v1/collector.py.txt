from pathlib import Path
import hashlib
import json
import shutil
import xml.etree.ElementTree as ET

repo = Path.cwd()
source = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-package-checks-precision-proxy-v1')
record = json.loads((source / 'verification.json').read_bytes())
assert record['status'] == 'passed'
assert record['source_commit'] == 'a128ef7b58a9fc9be395c3b157565408b865eda7'
assert len(record['checks']) == 6 and all(c['exit_code'] == 0 for c in record['checks'])
cases = list(ET.parse(source / 'pytest.xml').getroot().iter('testcase'))
for name, count in [('test_sleep_support_version', 4), ('test_support_revision_fence', 5),
                    ('test_spreading_visit_budget', 4), ('test_qualified_inventory_v3', 10)]:
    selected = [c for c in cases if name in c.attrib.get('classname', '').split('.')]
    assert len(selected) == count and all(len(c) == 0 for c in selected), (name, len(selected))
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
assert sha(repo / '.venv/final_checkpoint_checks_v3.py') == record['runner_sha256']
for item in record['artifacts']:
    path = Path(item['file'])
    assert sha(path) == item['sha256'] and path.stat().st_size == item['bytes']
out = repo / 'docs/validation/2026-09-20/precision-proxy-source-checkpoint-v1'
out.mkdir(exist_ok=False)
(out / '.gitattributes').write_bytes(b'** -text -whitespace\n')
for path in source.iterdir():
    if path.is_file() and path.suffix in {'.json', '.xml', '.log'}:
        shutil.copyfile(path, out / path.name)
shutil.copyfile(repo / '.venv/final_checkpoint_checks_v3.py', out / 'runner.py.txt')
shutil.copyfile(__file__, out / 'collector.py.txt')
counts = record['checks'][0]['counts']
old_record = json.loads((repo / 'docs/validation/2026-09-20/spreading-budget-installed-checkpoint-v1/installation.json').read_bytes())
wheel = next(item for item in record['artifacts'] if item['file'].endswith('.whl'))
assert wheel['sha256'] == old_record['wheel']['sha256']
assert Path(wheel['file']).read_bytes() == Path(old_record['wheel']['path']).read_bytes()
(out / 'wheel-parity.json').write_text(json.dumps({
    'sha256': wheel['sha256'], 'bytes': wheel['bytes'],
    'previous_installed_source': old_record['bindings_after']['commit'],
    'current_source': record['source_commit'], 'byte_identical': True,
    'limits': 'Existing installed and Hermes v36 checks cover these identical wheel bytes. Benchmark reporting is source-only and validated by the current source suite.'
}, indent=2) + '\n')
(out / 'README.md').write_text(f'''# Precision proxy source/package checkpoint

Frozen source a128ef7 passes {counts['tests'] - counts['skipped']:,} tests with
{counts['skipped']} skips and all six source/package checks. The full suite
includes the benchmark report-label regression, standalone visit budgets,
support-revision fencing and experimental qualified-inventory v3 contracts.

The rebuilt wheel is byte-identical to the 995d9c2 wheel, SHA-256
{wheel['sha256']}.
The previous 893-case installed suite with four skips and the six-phase
Hermes v36 lifecycle therefore bind the same wheel bytes. Their original
source and fixture identities remain unchanged. Benchmark report labels live
outside the wheel and are covered by this new source check.

The extraction precision fields explicitly label exact-text matching as a
proxy and keep semantic factual precision NOT MEASURED. Neither a passing
suite nor byte parity supplies model quality, full-module benefit, new native
capture or daily-use acceptance. No live configuration, migration or publication
occurred. The ongoing extraction experiments use their separate frozen inputs.
''')
(out / 'evidence-index.json').write_text(json.dumps({'files': {
    p.name: {'bytes': p.stat().st_size, 'sha256': sha(p)} for p in out.iterdir() if p.is_file()
}}, indent=2) + '\n')
print(json.dumps({'archive': str(out), 'counts': counts}))
