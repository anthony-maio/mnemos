from pathlib import Path
from zipfile import ZipFile
import asyncio
import hashlib
import inspect
import json
import os
import shutil
import tempfile

for key in tuple(os.environ):
    if key.startswith('MNEMOS_'):
        os.environ.pop(key)
from mnemos.client import MnemosClient
from mnemos.config import MnemosConfig, ModuleConfig
from mnemos.engine import MnemosEngine
from mnemos.memory_service import LocalMemoryService
from mnemos.scopes import ScopeRegistry
from mnemos.service import MnemosService
from mnemos.utils.embeddings import SimpleEmbeddingProvider
from mnemos.utils.llm import UnavailableLLMProvider
from mnemos.utils.storage import SQLiteStore

candidate = Path('/tmp/mnemos-hermes-compaction-origin-candidate-v1')
boundary = json.loads((candidate / 'integration_api_probe_v28.json').read_bytes())
source_outbox = Path(boundary['root']) / 'valid/outbox.json'
original = source_outbox.read_bytes()
entries = json.loads(original)
assert len(entries) == 1
root = Path(tempfile.mkdtemp(prefix='mnemos-hermes-outbox-delivery-v29-'))
descriptor = root / 'runtime.json'
outbox = root / 'outbox.json'
outbox.write_bytes(original)
wheel = Path('/mnt/c/Users/Ant/AppData/Local/Temp/mnemos-package-checks-generation-settings-v2/dist/mnemos_memory-0.7.0-py3-none-any.whl')
assert hashlib.sha256(wheel.read_bytes()).hexdigest() == boundary['wheel_sha256']
installed = Path(inspect.getfile(MnemosClient)).parents[1]
verified = 0
with ZipFile(wheel) as archive:
    for name in archive.namelist():
        if name.startswith('mnemos/') and name.endswith('.py'):
            assert (installed / name).read_bytes() == archive.read(name)
            verified += 1
registry = ScopeRegistry()
registry.register_profile('profile')
registry.register_project(Path(boundary['root']) / 'valid', user_profile_id='profile', enabled=True)
config = MnemosConfig(modules=ModuleConfig(surprisal=False, mutable_rag=False,
                                          affective=False, sleep=False, spreading=False))
instances = []
def factory():
    memory = LocalMemoryService(MnemosEngine(config=config,
        llm=UnavailableLLMProvider(), embedder=SimpleEmbeddingProvider(),
        store=SQLiteStore(str(root / 'memory.sqlite'))), registry)
    instances.append(memory)
    return memory

def runtime():
    return MnemosService(dispatcher_factory=factory, descriptor_path=descriptor,
                         startup_timeout=10, dispatch_timeout=10)

def snapshot():
    store = instances[-1].engine.store
    return {'events': [x.model_dump(mode='json') for x in store.list_episodic_events()],
            'jobs': [x.model_dump(mode='json') for x in store.list_jobs()],
            'fact_count': len(store.get_all())}

async def main():
    result = {'root': str(root), 'source_outbox_sha256': hashlib.sha256(original).hexdigest(),
              'wheel_sha256': boundary['wheel_sha256'], 'verified_modules': verified,
              'modules': config.modules.model_dump(), 'phases': [],
              'limits': 'Actual loopback service and SQLite raw delivery; no extraction/model quality or full interactive CLI proof.'}
    client = MnemosClient(descriptor_path=descriptor, outbox_path=outbox, auto_start=False)
    service = runtime()
    try:
        service.start()
        first = await client.replay_outbox()
        assert (first['acknowledged'], first['failed'], first['pending']) == (1, 0, 0), first
        state = snapshot()
        assert len(state['events']) == 1 and state['fact_count'] == 0, state
        event = state['events'][0]
        assert event['content'] == entries[0]['payload']['content']
        assert event['session_id'] == entries[0]['payload']['context']['session_id']
        result['phases'].append({'phase': 'delivery', 'replay': first, 'state': state})
        await client.close()
        service.stop()
        assert not descriptor.exists()
        # Reintroduce the exact saved entry to model an acknowledgement lost
        # after durable capture. Never alter its event identity or payload.
        outbox.write_bytes(original)
        client = MnemosClient(descriptor_path=descriptor, outbox_path=outbox, auto_start=False)
        offline = await client.replay_outbox()
        assert (offline['acknowledged'], offline['failed'], offline['pending']) == (0, 1, 1), offline
        result['phases'].append({'phase': 'service_offline', 'replay': offline})
        await client.close()
        service = runtime()
        service.start()
        reopened = snapshot()
        assert [x['id'] for x in reopened['events']] == [event['id']]
        replay = await client.replay_outbox()
        assert (replay['acknowledged'], replay['failed'], replay['pending']) == (1, 0, 0), replay
        after = snapshot()
        assert [x['id'] for x in after['events']] == [event['id']]
        assert [x['id'] for x in after['jobs']] == [x['id'] for x in state['jobs']]
        assert after['fact_count'] == 0
        result['phases'].append({'phase': 'restart_replay', 'replay': replay, 'state': after})
        result['status'] = 'passed'
    finally:
        await client.close()
        service.stop()
        result['descriptor_removed'] = not descriptor.exists()
        (root / 'result.json').write_text(json.dumps(result, indent=2) + '\n')
    assert result['descriptor_removed']
    assert source_outbox.read_bytes() == original
    shutil.copyfile(__file__, root / 'runner.py.txt')
    output = candidate / 'outbox_delivery_v29.json'
    assert not output.exists()
    output.write_text(json.dumps(result, indent=2) + '\n')
    print(json.dumps({'status': result['status'], 'root': str(root), 'phases': len(result['phases'])}))

asyncio.run(main())
