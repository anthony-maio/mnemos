from pathlib import Path
from contextlib import closing
import hashlib
import json
import shutil
import sqlite3
import sqlite_vec

repo = Path(__file__).resolve().parents[1]
source = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-historical-claim-service-v3')
out = repo / 'docs/validation/2026-09-20/historical-claim-service-v3-verified'
state = json.loads((source / 'status.json').read_bytes())
prep = json.loads((source / 'preparation.json').read_bytes())
inputs = json.loads((source / 'inputs.json').read_bytes())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
assert state['status'] == 'completed' and state['all_lifecycle_checks_passed'] is True
assert state['preparation_sha256'] == sha(source / 'preparation.json')
for path, digest in prep['files'].items():
    assert sha(source / path) == digest
out.mkdir(exist_ok=False)
(out / '.gitattributes').write_bytes(b'** -text -whitespace\n')
databases = []
for path in sorted(source.rglob('*')):
    if not path.is_file() or path.suffix not in {'.json', '.jsonl', '.py', '.txt', '.log', '.sqlite'}:
        continue
    dest = out / path.relative_to(source)
    dest.parent.mkdir(parents=True, exist_ok=True)
    if path.suffix == '.sqlite':
        with closing(sqlite3.connect(path.as_uri() + '?mode=ro', uri=True)) as original:
            with closing(sqlite3.connect(dest)) as backup:
                original.backup(backup)
                backup.enable_load_extension(True)
                sqlite_vec.load(backup)
                backup.enable_load_extension(False)
                assert backup.execute('PRAGMA integrity_check').fetchone() == ('ok',)
                foreign_errors = backup.execute('PRAGMA foreign_key_check').fetchall()
                assert foreign_errors == [], foreign_errors
                counts = {}
                for (table,) in backup.execute("SELECT name FROM sqlite_master WHERE type='table'").fetchall():
                    quoted = '"' + table.replace('"', '""') + '"'
                    counts[table] = backup.execute('SELECT count(*) FROM ' + quoted).fetchone()[0]
                backup.execute('PRAGMA journal_mode=DELETE')
                backup.commit()
        databases.append({'path': dest.relative_to(out).as_posix(), 'integrity': 'ok', 'foreign_key_errors': [], 'table_counts': counts})
    else:
        shutil.copyfile(path, dest)
cases = []
for case in inputs['cases']:
    result = json.loads((source / case['id'] / 'result.json').read_bytes())
    assert result['checks'] and all(result['checks'].values())
    cases.append({'id': case['id'], 'status': result['status'], 'criterion': case['criterion'],
                  'checks': result['checks'], 'committed_texts': [r['fact']['content'] for r in result.get('extracted', [])]})
assert [case['status'] for case in cases] == ['completed_lifecycle'] * 3 + ['completed_negative_control'] * 4
(out / 'summary.json').write_text(json.dumps({'cases': cases, 'database_inspection': databases}, indent=2) + '\n')
shutil.copyfile(__file__, out / 'collector.py.txt')
files = {p.relative_to(out).as_posix(): {'bytes': p.stat().st_size, 'sha256': sha(p)} for p in sorted(out.rglob('*')) if p.is_file()}
assert not any(name.endswith(('-wal', '-shm')) for name in files)
(out / 'evidence-index.json').write_text(json.dumps({'files': files}, indent=2) + '\n')
(out / 'README.md').write_text('''# History-clause extraction through the real local service
