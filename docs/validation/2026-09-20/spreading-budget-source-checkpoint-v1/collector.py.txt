from pathlib import Path
import hashlib
import json
import shutil
import xml.etree.ElementTree as ET

repo = Path.cwd()
source = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-package-checks-spreading-budget-v1')
record = json.loads((source / 'verification.json').read_bytes())
assert record['status'] == 'passed'
assert record['source_commit'] == '995d9c27ca71d30e2a55f548f2541d87614f0c08'
assert len(record['checks']) == 6 and all(c['exit_code'] == 0 for c in record['checks'])
cases = list(ET.parse(source / 'pytest.xml').getroot().iter('testcase'))
for name, count in [('test_sleep_support_version', 4), ('test_support_revision_fence', 5),
                    ('test_spreading_visit_budget', 4), ('test_qualified_inventory_v3', 10)]:
    selected = [c for c in cases if name in c.attrib.get('classname', '').split('.')]
    assert len(selected) == count and all(len(c) == 0 for c in selected), (name, len(selected))
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
assert sha(repo / '.venv/final_checkpoint_checks_v3.py') == record['runner_sha256']
for item in record['artifacts']:
    path = Path(item['file'])
    assert sha(path) == item['sha256'] and path.stat().st_size == item['bytes']
out = repo / 'docs/validation/2026-09-20/spreading-budget-source-checkpoint-v1'
out.mkdir(exist_ok=False)
(out / '.gitattributes').write_bytes(b'** -text -whitespace\n')
for path in source.iterdir():
    if path.is_file() and path.suffix in {'.json', '.xml', '.log'}:
        shutil.copyfile(path, out / path.name)
shutil.copyfile(repo / '.venv/final_checkpoint_checks_v3.py', out / 'runner.py.txt')
shutil.copyfile(__file__, out / 'collector.py.txt')
counts = record['checks'][0]['counts']
(out / 'README.md').write_text(f'''# Spreading visit-budget source/package checkpoint

Frozen source 995d9c2 passes {counts['tests'] - counts['skipped']:,} tests with
{counts['skipped']} skips and all six source/package checks. The four standalone
visit-budget cases, nine support-revision cases and ten experimental inventory
v3 cases are included and pass. Formatting, Python 3.10 type checking,
JavaScript syntax, build and package metadata checks pass.

Standalone spreading calls now honor the configured max_visited when no call
override is supplied. The managed engine already passed that limit explicitly;
this does not supersede the real-model graph recall negatives. See the sibling
[repair record](../spreading-visit-budget-v1.md) for the reproduced baseline
failures, controls and focused checks.

The wheel changes production code and needs its own installed verification.
The separate extraction experiment remains unadopted and uses its own frozen
candidate/provider source. Semantic quality, full-module benefit, wider host
recovery and daily-use acceptance remain open. No live configuration, migration
or publication changed.
''')
(out / 'evidence-index.json').write_text(json.dumps({'files': {
    p.name: {'bytes': p.stat().st_size, 'sha256': sha(p)} for p in out.iterdir() if p.is_file()
}}, indent=2) + '\n')
print(json.dumps({'archive': str(out), 'counts': counts}))
