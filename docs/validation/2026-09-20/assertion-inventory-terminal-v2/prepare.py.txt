from pathlib import Path

import hashlib, json, shutil, subprocess

shutil.copyfile(repo/".venv/assertion_inventory_runner_v2.py",root/"runner.py")
shutil.copyfile(__file__,root/"prepare.py.txt")
(root/"protocol.json").write_text(json.dumps({
    "status":"prepared_not_executed", "calls":46,"repetitions":1,
    "scope":"Source-only qualified assertion inventory v2 over reused development cases. No existing facts are shown, no target reconciliation or storage is attempted.",
    "comparison":"Compare retained meanings, qualification attachment, attribution, unwanted retention and unsupported interpretations with frozen v1 inventory and earlier 14B prompt outcomes. Pipeline and schema change together; no isolated component causal claim.",
