"""Preserve verified source and installed evidence for the embedding-outage lexical retrieval fallback."""
from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
temp = Path('C:/Users/Ant/AppData/Local/Temp')
checks = temp / 'mnemos-package-checks-retrieval-outage-v1'
installed = temp / 'mnemos-installed-checks-retrieval-outage-v1'
output = repo / 'docs/validation/2026-09-20/retrieval-provider-outage-checkpoint-v1'
verification = json.loads((checks / 'verification.json').read_text(encoding='utf-8'))
installation = json.loads((installed / 'installation.json').read_text(encoding='utf-8'))
contract = json.loads((installed / 'contract.json').read_text(encoding='utf-8'))
assert verification['status'] == installation['status'] == contract['status'] == 'passed'
commit = verification['source_commit']
assert commit == '4bbab4f01fd6b5a8e1e4d76504eaa8d59d349da8'
assert commit == installation['bindings_after']['commit']
assert all(c['exit_code'] == 0 for c in verification['checks'])

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

assert sha(checks / 'verification.json') == installation['bindings_before']['checks_sha256']
assert sha(checks / 'pytest.xml') == installation['successful_pytest_xml_sha256']
for artifact in verification['artifacts']:
    p = Path(artifact['file'])
    assert sha(p) == artifact['sha256'] and p.stat().st_size == artifact['bytes']
for command in installation['commands']:
    for stream in ('stdout', 'stderr'):
        assert sha(installed / (command['name'] + '.' + stream + '.log')) == command[stream + '_sha256']
assert sha(repo / '.venv/final_checkpoint_checks_v3.py') == verification['runner_sha256']
assert sha(repo / '.venv/run_combined_installed_check_v43.py') == installation['runner_sha256']
output.mkdir(exist_ok=False)
(output / '.gitattributes').write_bytes(b'** -text -whitespace\n')

def copy(path, relative):
    target = output / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(path, target)

for kind, folder in (('source-checks', checks), ('installed-checks', installed)):
    for p in folder.iterdir():
        if p.is_file() and p.suffix in {'.json', '.xml', '.log'}:
            copy(p, Path(kind) / p.name)
for name in ('final_checkpoint_checks_v3.py', 'run_combined_installed_check_v43.py', 'installed_contract_probe_v4.py', 'verify_claude_stop_replay_v2.py'):
    copy(repo / '.venv' / name, Path('helpers') / (name + '.txt'))
copy(Path(__file__), Path('helpers') / (Path(__file__).name + '.txt'))
files = {p.relative_to(output).as_posix(): {'bytes': p.stat().st_size, 'sha256': sha(p)}
         for p in sorted(output.rglob('*')) if p.is_file()}
(output / 'evidence-index.json').write_text(json.dumps({'files': files}, indent=2) + '\n', encoding='utf-8')
s = verification['checks'][0]['counts']
i = installation['contract']['junit_summary']
wheel = next(a for a in verification['artifacts'] if a['file'].endswith('.whl'))
(output / 'README.md').write_text(f"""# Embedding-outage retrieval package checkpoint

Frozen product `{commit}` passes {s['tests'] - s['skipped']:,} source tests
with {s['skipped']} skips, all six source/package checks, {i['tests'] - i['skipped']:,}
isolated installed cases with {i['skipped']} matching skips, and seven replay
assertions. Installed checks include transient-provider fallback and host
status propagation, and compare individual outcomes, imported module origins,
and package bytes with the frozen source and wheel.

Wheel: {wheel['bytes']:,} bytes; SHA-256 `{wheel['sha256']}`.

The [development record](../retrieval-provider-outage-development-v1/README.md)
preserves two failing closed-endpoint cases, four failing host status cases,
85 focused retrieval passes, and 104 host passes. These suites overlap.
Existing SQLite memories now remain available through scoped lexical matches
after a transient embedding failure. Empty results retain degraded status,
semantic scores remain unmeasured, and returned context identifies the weaker
retrieval mode. Scope, evidence views, correction, forgetting, and budgets
remain enforced. Provider timeouts and retries still precede fallback;
paraphrases without matching words may return no facts.

These checks establish package behavior. Native provider-outage recall is a
separate experiment and has not run on this product. Previous native Codex
and Claude stopped-service tests remain bound to their original packages and
establish raw-capture recovery only. Full-module quality, Desktop, broader
outage recovery, and the operational pilot remain open. No live host
configuration or memory database was changed by these package checks.

The evidence index binds copied raw records and validation helpers. This
README is outside that index.
""", encoding='utf-8', newline='\n')
print(json.dumps({'archive': str(output), 'files': len(files), 'source_commit': commit,
                  'source_passes': s['tests']-s['skipped'], 'source_skips': s['skipped'],
                  'installed_passes': i['tests']-i['skipped'], 'installed_skips': i['skipped'],
                  'wheel': wheel}))
