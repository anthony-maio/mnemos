from pathlib import Path
from datetime import datetime,timezone
from urllib.request import urlopen
import hashlib,json,shutil
repo=Path(__file__).resolve().parents[1]
prior=Path('C:/Users/Ant/AppData/Local/Temp/mnemos-extraction-authored-holdout-v1')
out=Path('C:/Users/Ant/AppData/Local/Temp/mnemos-extraction-temperature-one-v1')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
state=json.loads((prior/'status.json').read_bytes())
assert state['status']=='completed' and state['processed']==40
assert sha(prior/'observations.jsonl')==state['observations_sha256']
prep=json.loads((prior/'preparation.json').read_bytes())
for name,digest in prep['files'].items(): assert sha(prior/name)==digest
inputs=json.loads((prior/'inputs.json').read_bytes())
before=dict(inputs)
assert inputs['temperature']==0.0 and inputs['think'] is True and inputs['max_tokens']==8192
inputs['temperature']=1.0
assert [k for k in before if before[k]!=inputs[k]]==['temperature']
with urlopen(inputs['endpoint']+'/api/tags',timeout=10) as f:
 model=next(m for m in json.loads(f.read())['models'] if m['name']==inputs['model'])
assert model['digest']==prep['model']['digest']
with urlopen(inputs['endpoint']+'/api/version',timeout=10) as f: version=json.loads(f.read())
out.mkdir(exist_ok=False)
(out/'inputs.json').write_text(json.dumps(inputs,indent=2)+'\n')
for name in ['runner.py','extraction_holdout_authored_v1.json','extraction_holdout_rubric_review_v1.json']: shutil.copyfile(prior/name,out/name)
shutil.copyfile(__file__,out/'prepare.py.txt')
(out/'diagnostic-contract.json').write_text(json.dumps({
 'comparison':'Previously executed development cases, not unseen holdout.',
 'changed_setting':'temperature only: 0.0 -> 1.0',
 'reason':'Qwen model card recommends nonzero temperature for thinking general tasks. This isolated change tests sampling sensitivity; it does not apply or validate the full recommended sampling recipe.',
 'reference':'https://huggingface.co/Qwen/Qwen3.5-9B#best-practices',
 'reference_checked_at':datetime.now(timezone.utc).isoformat(),
 'ollama_version':version,
 'unchanged':'Original production/history prompt arms (without missing-target clause), source, schemas, all 20 cases, identities, ordering, model, think=true, max_tokens=8192, timeout=180, rubric and one repetition.',
 'calls':40,'selective_retry':False,
 'semantic_acceptance':'Review authored required and forbidden meanings. Use extraction_holdout_authored_v1.json, not the unused legacy criterion field. Preserve P07/N10 rubric ambiguities. No aggregate semantic accuracy claim or production adoption from one stochastic repetition.',
 'original_observations_sha256':state['observations_sha256'],
 'start_condition':'Start only after the existing missing-target diagnostic is terminal and its process has exited.'
},indent=2)+'\n')
manifest={'prepared_at':datetime.now(timezone.utc).isoformat(),'source_commit':prep['source_commit'],'source_files':prep['source_files'],'model':model,'parent_preparation_sha256':sha(prior/'preparation.json'),'files':{p.name:sha(p) for p in sorted(out.iterdir())}}
(out/'preparation.json').write_text(json.dumps(manifest,indent=2)+'\n')
print(json.dumps({'root':str(out),'calls':40,'started':False}))
