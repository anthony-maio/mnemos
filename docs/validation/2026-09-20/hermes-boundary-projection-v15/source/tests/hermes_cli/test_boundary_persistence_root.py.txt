from datetime import datetime
import sqlite3

import pytest

from agent.context_compressor import _DB_PERSISTED_MARKER
from hermes_cli.cli_session_mixin import CLISessionMixin
from hermes_state import SessionDB
from run_agent import AIAgent


@pytest.fixture
def boundary(tmp_path, monkeypatch):
    import cli as cli_module
    monkeypatch.setattr(cli_module, 'CLI_CONFIG', {'model': {'default': 'fixture', 'provider': ''}, 'agent': {}})
    db = SessionDB(tmp_path / 'state.db')
    db.create_session('old', 'cli', cwd=str(tmp_path), model='fixture')
    agent = AIAgent.__new__(AIAgent)
    agent._memory_manager = None
    agent._session_db = db
    agent._session_db_created = True
    agent.session_id = 'old'
    agent._last_flushed_db_idx = 0
    agent._flushed_db_message_ids = set()
    agent._memory_store = None
    agent.context_compressor = None
    agent.model = 'fixture'
    cli = CLISessionMixin.__new__(CLISessionMixin)
    cli.session_id, cli.session_start = 'old', datetime.now()
    cli._session_db, cli.conversation_history, cli.agent = db, [], agent
    for key, value in {
        'model': 'fixture', 'provider': '', 'base_url': '', 'api_key': '', 'requested_provider': '',
        '_explicit_api_key': '', '_explicit_base_url': '', '_explicit_model_override': False,
        '_pending_one_turn_model_restore': None, 'reasoning_config': {}, 'service_tier': None,
        'max_turns': 1, '_resumed': False, '_pending_title': None, '_status_bar_title_checked_at': 0.0,
    }.items():
        setattr(cli, key, value)
    notifications = []
    cli._notify_session_boundary = lambda event: notifications.append(event)
    cli._write_terminal_breadcrumb = lambda: None
    yield cli, agent, db, notifications
    db.close()


def reset(cli):
    CLISessionMixin.new_session(cli, silent=True)


def assert_retained(cli, db, notifications, history):
    assert cli.session_id == cli.agent.session_id == 'old'
    assert cli.conversation_history == history
    assert db.get_session('old')['ended_at'] is None
    assert notifications == []


def test_new_saves_uncommitted_message_before_rotating(boundary):
    cli, agent, db, notifications = boundary
    cli.conversation_history = [{'role': 'user', 'content': 'uncommitted'}]
    reset(cli)
    assert cli.session_id != 'old'
    assert [r['content'] for r in db.get_messages('old')] == ['uncommitted']
    assert notifications == ['on_session_finalize', 'on_session_reset']


def test_new_does_not_duplicate_actual_previously_flushed_rows(boundary):
    cli, agent, db, _ = boundary
    cli.conversation_history = [{'role': 'user', 'content': 'durable'}]
    assert agent._flush_messages_to_session_db(cli.conversation_history, conversation_history=[]) is True
    original_id = db.get_messages('old')[0]['id']
    reset(cli)
    assert cli.session_id != 'old'
    assert [r['id'] for r in db.get_messages('old')] == [original_id]


def test_new_preserves_legacy_prefix_and_identical_new_suffix(boundary):
    cli, agent, db, _ = boundary
    db.append_message('old', 'user', 'repeat')
    old_id = db.get_messages('old')[0]['id']
    cli.conversation_history = [{'role': 'user', 'content': 'repeat'}, {'role': 'user', 'content': 'repeat'}]
    assert all('_row_id' not in r and _DB_PERSISTED_MARKER not in r for r in cli.conversation_history)
    reset(cli)
    assert cli.session_id != 'old'
    rows = db.get_messages('old')
    assert len(rows) == 2 and rows[0]['id'] == old_id
    assert rows[1]['id'] != old_id
    assert [r['content'] for r in rows] == ['repeat', 'repeat']


def test_new_rejects_foreign_row_identity_without_reset(boundary, tmp_path):
    cli, agent, db, notifications = boundary
    db.append_message('old', 'user', 'same text')
    db.create_session('foreign', 'cli', cwd=str(tmp_path), model='fixture')
    foreign_id = db.append_message('foreign', 'user', 'same text')
    history = [{'role': 'user', 'content': 'same text', '_row_id': foreign_id}]
    cli.conversation_history = history.copy()
    reset(cli)
    assert_retained(cli, db, notifications, history)
    assert len(db.get_messages('old')) == len(db.get_messages('foreign')) == 1


def test_new_rejects_unverified_persisted_marker(boundary):
    cli, agent, db, notifications = boundary
    history = [{'role': 'user', 'content': 'not durable', _DB_PERSISTED_MARKER: True}]
    cli.conversation_history = history.copy()
    reset(cli)
    assert_retained(cli, db, notifications, history)
    assert db.get_messages('old') == []


def test_new_keeps_session_when_projection_read_fails(boundary, monkeypatch):
    cli, agent, db, notifications = boundary
    history = [{'role': 'user', 'content': 'unsaved'}]
    cli.conversation_history = history.copy()
    def fail(*args, **kwargs):
        raise sqlite3.OperationalError('injected read failure')
    monkeypatch.setattr(db, 'get_messages_as_conversation', fail)
    reset(cli)
    assert_retained(cli, db, notifications, history)
    assert db.get_messages('old') == []


def test_new_keeps_session_when_flush_reports_failure(boundary, monkeypatch):
    cli, agent, db, notifications = boundary
    history = [{'role': 'user', 'content': 'unsaved'}]
    cli.conversation_history = history.copy()
    monkeypatch.setattr(agent, '_flush_messages_to_session_db', lambda *args, **kwargs: False)
    reset(cli)
    assert_retained(cli, db, notifications, history)
    assert db.get_messages('old') == []


def test_new_preserves_interrupted_admission_message(boundary):
    cli, agent, db, _ = boundary
    cli.conversation_history = [{'role': 'user', 'content': 'interrupted', '_persist_after_admission_interrupt': True}]
    reset(cli)
    assert cli.session_id != 'old'
    assert [r['content'] for r in db.get_messages('old')] == ['interrupted']


def test_new_does_not_skip_changed_tool_call_prefix(boundary):
    cli, agent, db, notifications = boundary
    original = {'role': 'assistant', 'content': None, 'tool_calls': [{'id': 'one', 'type': 'function', 'function': {'name': 'read_file', 'arguments': '{}'}}]}
    assert agent._flush_messages_to_session_db([original], conversation_history=[]) is True
    history = [{'role': 'assistant', 'content': None, 'tool_calls': [{'id': 'two', 'type': 'function', 'function': {'name': 'read_file', 'arguments': '{}'}}]}]
    cli.conversation_history = history.copy()
    reset(cli)
    assert_retained(cli, db, notifications, history)
    assert len(db.get_messages('old')) == 1


@pytest.mark.parametrize('content', [
    ' hello ',
    [{'type': 'text', 'text': 'describe'}, {'type': 'image_url', 'image_url': {'url': 'data:image/png;base64,fixture'}}],
])
def test_new_accepts_durable_content_projection(boundary, content):
    cli, agent, db, _ = boundary
    cli.conversation_history = [{'role': 'user', 'content': content}]
    assert agent._flush_messages_to_session_db(cli.conversation_history, conversation_history=[]) is True
    before = db.get_messages('old')
    reset(cli)
    assert cli.session_id != 'old'
    assert db.get_messages('old') == before


def test_new_does_not_skip_changed_tool_name(boundary):
    cli, agent, db, notifications = boundary
    original = {'role': 'tool', 'content': 'ok', 'tool_name': 'read_file', 'tool_call_id': 'call1'}
    assert agent._flush_messages_to_session_db([original], conversation_history=[]) is True
    history = [{**original, 'tool_name': 'write_file'}]
    cli.conversation_history = history.copy()
    reset(cli)
    assert_retained(cli, db, notifications, history)
    assert db.get_messages('old')[0]['tool_name'] == 'read_file'


def test_new_accepts_restored_sanitized_projection(boundary):
    cli, agent, db, _ = boundary
    db.append_message('old', 'user', ' hello ')
    cli.conversation_history = db.get_messages_as_conversation('old', include_row_ids=True)
    assert cli.conversation_history[0]['content'] == 'hello'
    before = db.get_messages('old')
    reset(cli)
    assert cli.session_id != 'old'
    assert db.get_messages('old') == before
