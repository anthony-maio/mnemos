from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import shutil
from urllib.request import urlopen, Request

repo = Path(__file__).resolve().parents[1]
parent = repo / 'docs/validation/2026-09-20/history-eligibility-development-v1'
root = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-history-reasoning-development-v1')
assert not root.exists()
inputs = json.loads((parent / 'inputs.json').read_bytes())
inputs.update(schema_version='history-reasoning-development-v1', arms=['history_only'], model='qwen3.5:9b', think=True, max_tokens=2048)
inputs['interpretation'] = ('Model/inference-configuration development follow-up selected after Qwen2.5 eligibility regression. '
    'Same twelve authored cases, source/target IDs, history-only prompt, and criteria, with Qwen3.5:9b thinking enabled and 2048-token budget. '
    'Both model/configuration and output budget differ from the earlier 14b run; do not attribute differences to one factor. '
    'Three repetitions, no fact commits, no unseen holdout or general semantic score.')
with urlopen(inputs['endpoint'] + '/api/tags', timeout=10) as response:
    model = next(m for m in json.loads(response.read())['models'] if m['name'] == inputs['model'])
request = Request(inputs['endpoint'] + '/api/show', data=json.dumps({'model': inputs['model']}).encode(), headers={'Content-Type': 'application/json'})
with urlopen(request, timeout=10) as response:
    show = json.loads(response.read())
assert 'thinking' in show['capabilities']
root.mkdir()
(root / 'inputs.json').write_text(json.dumps(inputs, indent=2) + '\n', encoding='utf-8')
(root / 'model-show.json').write_text(json.dumps(show, indent=2) + '\n', encoding='utf-8')
shutil.copyfile(repo / '.venv/run_history_eligibility_development_v1.py', root / 'runner.py')
shutil.copyfile(Path(__file__), root / 'prepare.py.txt')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
parent_prep = json.loads((parent / 'preparation.json').read_bytes())
manifest = {'prepared_at': datetime.now(timezone.utc).isoformat(), 'model': model,
            'source_commit': inputs['source_commit'], 'source_files': parent_prep['source_files'],
            'parent_inputs_sha256': sha(parent / 'inputs.json'),
            'files': {p.name: sha(p) for p in sorted(root.iterdir())}}
(root / 'preparation.json').write_text(json.dumps(manifest, indent=2) + '\n', encoding='utf-8')
print(json.dumps({'root': str(root), 'calls': len(inputs['cases']) * 3, 'model': model['name'], 'digest': model['digest'], 'think': inputs['think'], 'max_tokens': inputs['max_tokens']}))
