"""Verify the portable writing handoff and manifest without extracting the archive."""
from pathlib import Path
import hashlib
import json
import posixpath
import re
import zipfile

path = Path('D:/Development/mnemos/deliverables/mnemos-substack-research-packet-2026-09-20-v18.zip')
with zipfile.ZipFile(path) as archive:
    assert archive.testzip() is None
    names = archive.namelist()
    assert len(names) == len(set(names)), 'Duplicate archive member'
    manifest = json.loads(archive.read('manifest.json'))
    assert set(names) == set(manifest['files']) | {'manifest.json'}
    for name, expected in manifest['files'].items():
        data = archive.read(name)
        assert len(data) == expected['bytes'], name
        assert hashlib.sha256(data).hexdigest() == expected['sha256'], name
    handoff = 'repository/docs/handoffs/substack-local-memory.md'
    text = archive.read(handoff).decode('utf-8')
    assert text.isascii()
    checked = 0
    for target in re.findall(r'\]\(([^)]+)\)', text):
        if '://' in target or target.startswith('#'):
            continue
        relative = target.split('#')[0]
        resolved = posixpath.normpath(posixpath.join(posixpath.dirname(handoff), relative))
        assert resolved in names, (target, resolved)
        checked += 1
    print(json.dumps({'path': str(path), 'source_commit': manifest['source_commit'],
                      'manifest_files': len(manifest['files']), 'handoff_local_links': checked,
                      'bytes': path.stat().st_size,
                      'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}))
