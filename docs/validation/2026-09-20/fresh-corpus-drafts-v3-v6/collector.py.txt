from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
out = repo / 'docs/validation/2026-09-20/fresh-corpus-drafts-v3-v6'
out.mkdir(exist_ok=False)
revisions = []
for version in range(3, 7):
    corpus = repo / f'.venv/fresh-corpus-authored-v{version}.jsonl'
    log = repo / f'.venv/fresh-corpus-authoring-v{version}.md'
    rows = [json.loads(line) for line in corpus.read_text().splitlines()]
    sessions = [r for r in rows if 'split' in r]
    held = [r for r in sessions if r['split'] == 'heldout']
    pools = [sum(f['gold_status'] == 'supported' for f in r['facts']) for r in held]
    revisions.append({'version': version, 'sha256': hashlib.sha256(corpus.read_bytes()).hexdigest(),
                      'all_fact_count': sum(len(r['facts']) for r in sessions),
                      'heldout_supported_fact_count': sum(pools),
                      'heldout_pool_sizes': {str(n): pools.count(n) for n in sorted(set(pools))}})
    for path in (corpus, log):
        shutil.copyfile(path, out / path.name)
(out / 'verified-counts.json').write_text(json.dumps(revisions, indent=2) + '\n')
shutil.copyfile(__file__, out / 'collector.py.txt')
(out / 'README.md').write_text('''# Fresh corpus authoring revisions v3 through v6

These are preserved, unexecuted synthetic drafts. No evaluated-model inference
has been run on them. V6 is under semantic review, not frozen for acceptance.
V1 and V2 remain in their earlier archives. Version strings within event IDs
describe authoring history, not proof of novel semantic coverage.

V3 repairs substantive corrections and their source links but still has only
two current facts in 100 ranking pools. Its author log says 320 total facts;
the actual total is 360. V4 has seven current facts in those pools, despite its
six-fact wording, and asks a generic primary-decision question without defining
which decision is primary in the source. V5 uses six-fact pools, but some
questions ask for missing names, command values or links; other questions embed
the complete expected answer. These are preparation defects, not model failures.

V6 uses content-specific requirement questions and removes full-answer
questions. It has 760 total facts: 700 current supported heldout facts,
20 superseded, 20 forgotten and 20 development facts. One hundred heldout
sessions have six current facts; another hundred have one. All 80 state pairs
retain a shared question while changing confident/uncertain context.

The verified counts are computed from each preserved corpus, independently of
its author log. The logs remain unchanged even where their claims were wrong.
Structural validity, matching gold IDs and empty no-answer labels do not prove
semantic validity. Normalized family review and source checks remain separate.
Repeated release-policy competitors and shared vocabulary limit generalization;
no independent 200-domain coverage or real-conversation acceptance is claimed.
Optional graph seed/target fields are empty and establish no graph benefit.
''')
(out / '.gitattributes').write_text('* -text\n')
(out / 'evidence-index.json').write_text(json.dumps({'files': {
    p.name: {'sha256': hashlib.sha256(p.read_bytes()).hexdigest(), 'bytes': p.stat().st_size}
    for p in out.iterdir() if p.is_file()
}}, indent=2) + '\n')
print(json.dumps(revisions, indent=2))
