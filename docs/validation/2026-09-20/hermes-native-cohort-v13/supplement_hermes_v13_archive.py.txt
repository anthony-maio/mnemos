from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
root = repo / 'docs/validation/2026-09-20/hermes-native-cohort-v13'
for name, target in (
    ('hermes-v13-terminal-analysis.json', 'terminal-analysis.json'),
    ('inspect_hermes_v13_terminal.py', 'inspect_hermes_v13_terminal.py.txt'),
):
    shutil.copyfile(repo / '.venv' / name, root / target)
shutil.copyfile(__file__, root / (Path(__file__).name + '.txt'))
summary = []
for number in (1, 2, 3):
    path = root / f'attempt-{number}/forget/trace.jsonl'
    trace = [json.loads(line) for line in path.read_text().splitlines()]
    latest = None
    calls = []
    for row in trace:
        if row['operation'] != 'tool_result':
            continue
        if row['name'] == 'mnemos_forget_preview':
            latest = row['result']
        elif row['name'] == 'mnemos_forget_related':
            assert latest is not None
            args, result = row['args'], row['result']
            selected = set(args.get('selected_source_event_ids', []))
            offered = {source['source_event_id'] for source in latest['source_events']}
            calls.append({'result': result, 'preview_limit': latest['limit'],
                          'apply_limit': args.get('preview_limit', 100),
                          'digest_matches': args['preview_digest'] == latest['preview_digest'],
                          'sources_outside_selectable_list': sorted(selected - offered),
                          'expected_version_supplied': 'expected_version' in args})
    summary.append({'attempt': number, 'trace_sha256': hashlib.sha256(path.read_bytes()).hexdigest(),
                    'related_forget_calls': calls,
                    'target_only_receipts': [r['result'] for r in trace if r['operation'] == 'tool_result'
                                             and r['name'] == 'mnemos_forget']})
assert all(call['preview_limit'] == 100 and call['apply_limit'] == 20
           for attempt in summary[:2] for call in attempt['related_forget_calls'])
assert summary[2]['related_forget_calls'][0]['result']['error'] == 'related_forget_selection_invalid'
assert summary[2]['related_forget_calls'][0]['sources_outside_selectable_list']
(root / 'selection-analysis.json').write_text(json.dumps(summary, indent=2) + '\n')
index_path = root / 'evidence-index.json'
index = json.loads(index_path.read_text())
for name, expected in index['files'].items():
    data = (root / name).read_bytes()
    assert len(data) == expected['bytes'] and hashlib.sha256(data).hexdigest() == expected['sha256']
for path in root.iterdir():
    if path.is_file() and path.name not in {'evidence-index.json', 'README.md'}:
        index['files'][path.name] = {'bytes': path.stat().st_size,
                                    'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}
index_path.write_text(json.dumps(index, indent=2) + '\n')
