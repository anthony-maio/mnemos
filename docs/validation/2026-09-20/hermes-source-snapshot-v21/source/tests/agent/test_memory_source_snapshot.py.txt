from copy import deepcopy
from types import SimpleNamespace
import pytest
from hermes_state import SessionDB
from agent.memory_source_snapshot import prepare_memory_source_snapshot


@pytest.fixture
def state(tmp_path):
    db = SessionDB(tmp_path / 'state.db')
    db.create_session('parent', 'cli', cwd=str(tmp_path))
    yield db, SimpleNamespace(_session_db=db, session_id='parent')
    db.close()


def test_marker_only_originals_are_saved_and_idempotent(state):
    db, agent = state
    messages = [{'role': 'user', 'content': 'original', '_db_persisted': True}]
    prepare_memory_source_snapshot(agent, messages)
    assert messages[0]['_row_id'] == db.get_messages('parent')[0]['id']
    assert db.get_memory_evidence('parent')[0]['content'] == 'original'
    prepare_memory_source_snapshot(agent, messages)
    assert len(db.get_messages('parent')) == 1


def test_legacy_prefix_and_uncommitted_suffix(state):
    db, agent = state
    row_id = db.append_message('parent', 'user', 'same')
    messages = [{'role': 'user', 'content': 'same'}, {'role': 'user', 'content': 'same', '_db_persisted': True}]
    prepare_memory_source_snapshot(agent, messages)
    assert [m['content'] for m in db.get_messages('parent')] == ['same', 'same']
    assert messages[0]['_row_id'] == row_id != messages[1]['_row_id']


def test_foreign_identity_rejected_without_writes(state, tmp_path):
    db, agent = state
    db.create_session('foreign', 'cli', cwd=str(tmp_path / 'foreign'))
    row_id = db.append_message('foreign', 'user', 'same')
    messages = [{'role': 'user', 'content': 'same', '_row_id': row_id}]
    original = deepcopy(messages)
    with pytest.raises(ValueError, match='unverified identity'):
        prepare_memory_source_snapshot(agent, messages)
    assert messages == original and db.get_messages('parent') == []


def test_failure_rolls_back_rows_and_caller_stamps(state, monkeypatch):
    db, agent = state
    messages = [{'role': 'user', 'content': 'first'}, {'role': 'assistant', 'content': 'second'}]
    original = deepcopy(messages)
    real = db._insert_evidence_components
    def fail_second(conn, row_id, sid, msg, *args):
        if msg.get('content') == 'second':
            raise RuntimeError('injected evidence failure')
        return real(conn, row_id, sid, msg, *args)
    monkeypatch.setattr(db, '_insert_evidence_components', fail_second)
    with pytest.raises(RuntimeError, match='injected evidence failure'):
        prepare_memory_source_snapshot(agent, messages)
    assert messages == original and db.get_messages('parent') == []


def test_multimodal_and_whitespace_durable_prefix(state):
    db, agent = state
    messages = [{'role': 'user', 'content': ' hello '},
                {'role': 'user', 'content': [{'type': 'text', 'text': 'describe'}, {'type': 'image_url', 'image_url': {'url': 'fixture'}}]}]
    prepare_memory_source_snapshot(agent, messages)
    prepare_memory_source_snapshot(agent, messages)
    restored = db.get_messages_as_conversation('parent', include_row_ids=True)
    prepare_memory_source_snapshot(agent, restored)
    assert len(db.get_messages('parent')) == 2


def test_changed_tool_name_rejected(state):
    db, agent = state
    messages = [{'role': 'tool', 'content': 'ok', 'tool_name': 'read', 'tool_call_id': 'a'}]
    prepare_memory_source_snapshot(agent, messages)
    messages[0]['tool_name'] = 'write'
    with pytest.raises(ValueError, match='durable prefix'):
        prepare_memory_source_snapshot(agent, messages)
    assert db.get_messages('parent')[0]['tool_name'] == 'read'


def test_empty_session_can_bind_explicit_host_directory(state, tmp_path):
    db, agent = state
    db.create_session('empty', 'cli')
    agent.session_id, agent.working_directory = 'empty', str(tmp_path)
    messages = [{'role': 'user', 'content': 'new source'}]
    prepare_memory_source_snapshot(agent, messages)
    assert db.get_session('empty')['cwd'] == str(tmp_path)
    assert db.get_memory_evidence('empty')[0]['_memory_origin']['cwd'] == str(tmp_path)


def test_existing_unscoped_history_is_not_retroactively_scoped(state, tmp_path):
    db, agent = state
    db.create_session('legacy', 'cli')
    db.append_message('legacy', 'user', 'old source')
    agent.session_id, agent.working_directory = 'legacy', str(tmp_path)
    messages = [{'role': 'user', 'content': 'old source'}]
    prepare_memory_source_snapshot(agent, messages)
    assert not db.get_session('legacy')['cwd']
    assert db.get_memory_evidence('legacy') == []


def test_unrelated_transcript_cannot_claim_saved_prefix(state):
    db, agent = state
    db.append_message('parent', 'assistant', 'saved source')
    messages = [{'role': 'user', 'content': 'different source'}]
    before = deepcopy(messages)
    with pytest.raises(ValueError, match='disagrees with durable prefix'):
        prepare_memory_source_snapshot(agent, messages)
    assert messages == before
    assert [row['content'] for row in db.get_messages('parent')] == ['saved source']
