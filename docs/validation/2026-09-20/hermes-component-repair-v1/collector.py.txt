from pathlib import Path
import hashlib
import json
import subprocess
import shutil

repo = Path(__file__).resolve().parents[1]
out = repo / 'docs/validation/2026-09-20/hermes-component-repair-v1'
out.mkdir(exist_ok=False)
(out / '.gitattributes').write_bytes(b'** -text -whitespace\n')
for category, version, helper in (
    ('component', 'v2', 'probe_hermes_component_writes_v2.py'),
    ('component', 'v3', 'probe_hermes_component_writes_v3.py'),
    ('hybrid', 'v3', 'probe_hermes_hybrid_copy_v3.py'),
):
    target = out / (category + '-' + version)
    target.mkdir()
    for name in ('result.json', 'candidate.patch'):
        data = subprocess.check_output(['wsl', '-e', 'cat', '/tmp/mnemos-hermes-root-' + category + '-audit-' + version + '/' + name])
        (target / name).write_bytes(data)
    shutil.copyfile(repo / '.venv' / helper, target / 'probe.py.txt')
    result = json.loads((target / 'result.json').read_bytes())
    digest = hashlib.sha256((target / 'candidate.patch').read_bytes()).hexdigest()
    assert digest == result.get('patch_sha256', result.get('patch_sha256_before'))
    if category == 'hybrid':
        assert result['stable_patch'] and result['patch_sha256_after'] == digest
component = json.loads((out / 'component-v3/result.json').read_bytes())
assert all(c['status'] == 'completed' and c['write_result'] == 1 for c in component['cases'])
assert component['cases'][0]['evidence'][0]['content'] == 'Preserve this original user statement.'
assert component['cases'][1]['evidence'] == component['cases'][2]['evidence'] == []
assert component['cases'][3]['evidence'][0]['content'] == 'A real user message.'
hybrid = json.loads((out / 'hybrid-v3/result.json').read_bytes())
assert hybrid['cases'][0]['same_origins']
assert hybrid['cases'][1]['error'] == 'derived memory evidence component no longer matches its source'
assert component['patch_sha256'] == hybrid['patch_sha256_before']
shutil.copyfile(Path(__file__), out / 'collector.py.txt')
files = {p.relative_to(out).as_posix(): {'bytes': p.stat().st_size, 'sha256': hashlib.sha256(p.read_bytes()).hexdigest()}
         for p in sorted(out.rglob('*')) if p.is_file()}
(out / 'evidence-index.json').write_text(json.dumps({'files': files}, indent=2) + '\n', encoding='utf-8')
(out / 'README.md').write_text('''# Independent Hermes candidate repair checks

The isolated host candidate now passes the six bounded checks exercised by
the preceding write and hybrid-copy audits. The exact patch hash is
`9f44eb5b193c0e773f962991670420140f59684b949f62ee2c805c4f76ac2d79`.

- Single-message user writes remain in the evidence projection.
- Tool responses save successfully and contribute no user/assistant evidence.
- Assistant tool calls with null content save and contribute no text evidence.
- Multimodal user text saves and projects its actual text with the native ID.
- An unchanged hybrid carrier copied through the real model-facing restore
  path preserves both canonical origins.
- Role/content tampering of a copied summary component is rejected.

The intermediate component-v2 rerun is retained: single-message capture and
null-content writes were fixed, but tool writes still failed and multimodal
text was still omitted. The earlier four original failures and hybrid-copy
identity failure remain in their own archives.

These are independent SessionDB and compressor fixture checks. They do not
establish every lifecycle caller, database upgrade compatibility, a complete
installed integration, actual model compression, or a live host upgrade.
Broader host suites and installed replay/suppression work remain separate.
No live host source, configuration, or memory data changed.

The evidence index binds 11 artifacts. This README is outside the index.
''', encoding='utf-8', newline='\n')
print(json.dumps({'archive': str(out), 'indexed_files': len(files)}))
