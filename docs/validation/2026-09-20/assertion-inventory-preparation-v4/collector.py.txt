from pathlib import Path
import hashlib
import json
import shutil

repo = Path.cwd()
root = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-assertion-inventory-development-v4')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
prep = json.loads((root / 'preparation.json').read_bytes())
assert not (root / 'status.json').exists()
for name, digest in prep['files'].items():
    assert sha(root / name) == digest
report = json.loads((root / 'offline-preflight.json').read_bytes())
assert report['status'] == 'passed' and report['cases'] == 46
assert report['model_generation_calls'] == 0 and len(report['same_prompt_and_schema']) == 46
assert report['preparation_sha256'] == sha(root / 'preparation.json')
assert len(report['launch_checks']) == 3
out = repo / 'docs/validation/2026-09-20/assertion-inventory-preparation-v4'
out.mkdir(exist_ok=False)
for path in root.iterdir():
    if path.is_file():
        shutil.copyfile(path, out / path.name)
shutil.copyfile(repo / '.venv/preflight_assertion_inventory_v4.py', out / 'preflight.py.txt')
shutil.copyfile(__file__, out / 'collector.py.txt')
(out / 'README.md').write_text('''# Source-only inventory thinking-disabled comparison preparation

(out / '.gitattributes').write_bytes(b'** -text -whitespace\n')
