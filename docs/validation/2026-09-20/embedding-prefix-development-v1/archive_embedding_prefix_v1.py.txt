from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
source = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-embedding-prefix-development-v1')
target = repo / 'docs/validation/2026-09-20/embedding-prefix-development-v1'
report = json.loads((source / 'artifact.json').read_text())
verify = json.loads((repo / '.venv/embedding-prefix-verification.json').read_text())
assert report['status'] == 'complete' and verify['status'] == 'verified'
target.mkdir(parents=True, exist_ok=False)
for path in source.rglob('*'):
    if path.is_file():
        dest = target / path.relative_to(source)
        dest.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(path, dest)
for name in ('verify_embedding_prefix_v1.py', 'archive_embedding_prefix_v1.py',
             'embedding-prefix-verification.json', 'embedding-prefix-offline-final.xml',
             'embedding-prefix-live-red.xml', 'embedding-prefix-live-green.xml'):
    shutil.copyfile(repo / '.venv' / name, target / (name + '.txt' if name.endswith('.py') else name))
for relative, digest in report['input_hashes'].items():
    data = (repo / relative).read_bytes()
    assert hashlib.sha256(data).hexdigest() == digest
    dest = target / 'inputs' / (relative + '.txt' if relative.endswith('.py') else relative)
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_bytes(data)
(target / '.gitattributes').write_bytes(b'** -text -whitespace\n')
files = {p.relative_to(target).as_posix(): {'bytes': p.stat().st_size,
          'sha256': hashlib.sha256(p.read_bytes()).hexdigest()}
         for p in sorted(target.rglob('*')) if p.is_file()}
(target / 'evidence-index.json').write_text(json.dumps({'files': files}, indent=2) + '\n')
