from pathlib import Path
import hashlib
import json
import shutil

repo = Path.cwd()
temp = Path('C:/Users/Ant/AppData/Local/Temp')
source = temp / 'mnemos-installed-checks-spreading-budget-v1'
record = json.loads((source / 'installation.json').read_bytes())
contract = json.loads((source / 'contract.json').read_bytes())
assert record['status'] == contract['status'] == 'passed'
assert record['bindings_after']['commit'] == '995d9c27ca71d30e2a55f548f2541d87614f0c08'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
assert record['runner_sha256'] == sha(repo / '.venv/run_combined_installed_check_v54.py')
assert record['bindings_before']['checks_sha256'] == sha(temp / 'mnemos-package-checks-spreading-budget-v1/verification.json')
for command in record['commands']:
    for stream in ('stdout', 'stderr'):
        assert sha(source / (command['name'] + '.' + stream + '.log')) == command[stream + '_sha256']
expected = json.loads((source / 'expected-contract-cases.json').read_bytes())
for module, count in [('test_sleep_support_version', 4), ('test_support_revision_fence', 5),
                      ('test_spreading_visit_budget', 4)]:
    selected = [item for item in expected if item['module'] == module]
    assert len(selected) == count and all(item['outcome'] == 'passed' for item in selected)
for module in ('test_spreading_activation', 'test_spreading_isolation', 'test_spreading_mutual_builder'):
    selected = [item for item in expected if item['module'] == module]
    assert selected and all(item['outcome'] == 'passed' for item in selected)
out = repo / 'docs/validation/2026-09-20/spreading-budget-installed-checkpoint-v1'
out.mkdir(exist_ok=False)
(out / '.gitattributes').write_bytes(b'** -text -whitespace\n')
for path in source.iterdir():
    if path.is_file() and path.suffix in {'.json', '.xml', '.log'}:
        shutil.copyfile(path, out / path.name)
for name in ('run_combined_installed_check_v54.py', 'installed_contract_probe_v4.py', 'verify_claude_stop_replay_v2.py'):
    shutil.copyfile(repo / '.venv' / name, out / (name + '.txt'))
shutil.copyfile(__file__, out / 'collector.py.txt')
summary = record['contract']['junit_summary']
(out / 'README.md').write_text(f'''# Spreading visit-budget installed-package checkpoint

The wheel from frozen source 995d9c2 passes {summary['tests'] - summary['skipped']}
installed cases with {summary['skipped']} skips. Every selected result matches
the frozen source JUnit, including skip reasons. The selection includes all
four new visit-budget and nine support-revision cases, plus standalone spreading,
isolation and mutual-builder tests. Installed module origins, wheel bytes,
source/test hashes and the replay probe pass.

This is a fresh isolated installation. Standalone spreading now uses its
configured visit budget by default; the managed engine already supplied that
budget explicitly. No graph recall improvement follows from this repair.
Semantic extraction, all-module benefit, wider host recovery and daily-use
acceptance remain open. No live installation, migration or publication occurred.
''')
(out / 'evidence-index.json').write_text(json.dumps({'files': {
    p.name: {'bytes': p.stat().st_size, 'sha256': sha(p)} for p in out.iterdir() if p.is_file()
}}, indent=2) + '\n')
print(json.dumps({'archive': str(out), 'junit': summary}))
