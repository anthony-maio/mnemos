from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
parent = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-ollama-reasoning-routes-v2')
root = parent.with_name('mnemos-ollama-reasoning-budget-v1')
assert not root.exists()
inputs = json.loads((parent / 'inputs.json').read_bytes())
inputs['schema_version'] = 'ollama-reasoning-budget-v1'
inputs['timeout'] = 180
requests = []
for original in inputs['requests']:
    if original['case_id'] != 'historical-approval' or not original['think']:
        continue
    for constrained in (True, False):
        request = json.loads(json.dumps(original))
        request['constrained'] = constrained
        request['payload']['options']['num_predict'] = 8192
        if not constrained:
            request['validation_schema'] = request['payload'].pop('format')
        requests.append(request)
inputs['requests'] = requests
inputs['interpretation'] = ('Four diagnostic calls on the previously observed historical-approval case: generate/chat by format present/omitted, '
    'all think=true and 8192 token budget. Only budget differs from matching earlier arm; timeout raised to 180 seconds. '
    'Final-channel and source-bound-schema validity only, no memory commits or acceptance score.')
root.mkdir()
(root / 'inputs.json').write_text(json.dumps(inputs, indent=2) + '\n', encoding='utf-8')
shutil.copyfile(parent / 'runner.py', root / 'runner.py')
shutil.copyfile(Path(__file__), root / 'prepare.py.txt')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
manifest = {'prepared_at': datetime.now(timezone.utc).isoformat(),
            'parent_inputs_sha256': sha(parent / 'inputs.json'),
            'files': {p.name: sha(p) for p in sorted(root.iterdir())}}
(root / 'preparation.json').write_text(json.dumps(manifest, indent=2) + '\n', encoding='utf-8')
print(json.dumps({'root': str(root), 'calls': len(requests)}))
