from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import shutil
from urllib.request import urlopen

repo = Path(__file__).resolve().parents[1]
parent = repo / 'docs/validation/2026-09-20/history-reasoning-development-v1'
root = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-ollama-reasoning-routes-v1')
assert not root.exists()
original = json.loads((parent / 'inputs.json').read_bytes())
rows = [json.loads(line) for line in (parent / 'observations.jsonl').read_text().splitlines()]
selected = ['historical-approval', 'fiction', 'fiction-project']
# Resolve only exact existing IDs before freezing; do not change content.
all_ids = [c['id'] for c in original['cases']]
assert all(x in all_ids for x in selected), all_ids
requests = []
for case_id in selected:
    row = next(r for r in rows if r['case_id'] == case_id and r['repetition'] == 0)
    for think in (True, False):
        for route in ('generate', 'chat'):
            payload = {'model': original['model'], 'stream': False, 'think': think,
                       'options': {'temperature': original['temperature'], 'num_predict': original['max_tokens']},
                       'format': row['trace']['options']['format']}
            if route == 'generate':
                payload['prompt'] = row['trace']['prompt']
            else:
                payload['messages'] = [{'role': 'user', 'content': row['trace']['prompt']}]
            requests.append({'case_id': case_id, 'route': route, 'think': think, 'payload': payload})
with urlopen(original['endpoint'] + '/api/version', timeout=10) as response:
    version = json.loads(response.read())
with urlopen(original['endpoint'] + '/api/tags', timeout=10) as response:
    model = next(m for m in json.loads(response.read())['models'] if m['name'] == original['model'])
assert model['digest'] == json.loads((parent / 'preparation.json').read_bytes())['model']['digest']
root.mkdir()
inputs = {'schema_version': 'ollama-reasoning-routes-v1', 'endpoint': original['endpoint'],
          'model': model, 'server': version, 'timeout': 120, 'requests': requests,
          'interpretation': 'Previously observed authored cases. One call per case/route/think setting. Same prompt and schema, different API wrappers. Runtime channel diagnosis, no fact commits, no general quality score.'}
(root / 'inputs.json').write_text(json.dumps(inputs, indent=2) + '\n', encoding='utf-8')
shutil.copyfile(repo / '.venv/run_ollama_reasoning_routes_v1.py', root / 'runner.py')
shutil.copyfile(Path(__file__), root / 'prepare.py.txt')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
manifest = {'prepared_at': datetime.now(timezone.utc).isoformat(),
            'parent_index_sha256': sha(parent / 'evidence-index.json'),
            'parent_observations_sha256': sha(parent / 'observations.jsonl'),
            'files': {p.name: sha(p) for p in sorted(root.iterdir())}}
(root / 'preparation.json').write_text(json.dumps(manifest, indent=2) + '\n', encoding='utf-8')
print(json.dumps({'root': str(root), 'calls': len(requests), 'server': version}))
