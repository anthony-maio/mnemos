"""Preserve verified source and installed evidence for the client autostart opt-out."""
from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
temp = Path('C:/Users/Ant/AppData/Local/Temp')
checks = temp / 'mnemos-package-checks-autostart-env-v1'
installed = temp / 'mnemos-installed-checks-autostart-env-v1'
output = repo / 'docs/validation/2026-09-20/autostart-environment-checkpoint-v1'
verification = json.loads((checks / 'verification.json').read_text(encoding='utf-8'))
installation = json.loads((installed / 'installation.json').read_text(encoding='utf-8'))
contract = json.loads((installed / 'contract.json').read_text(encoding='utf-8'))
assert verification['status'] == installation['status'] == contract['status'] == 'passed'
commit = verification['source_commit']
assert commit == 'a64afb73d1b99510cfa32e96164bf6c11458b51e'
assert commit == installation['bindings_after']['commit']
assert all(c['exit_code'] == 0 for c in verification['checks'])

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

assert sha(checks / 'verification.json') == installation['bindings_before']['checks_sha256']
assert sha(checks / 'pytest.xml') == installation['successful_pytest_xml_sha256']
for artifact in verification['artifacts']:
    p = Path(artifact['file'])
    assert sha(p) == artifact['sha256'] and p.stat().st_size == artifact['bytes']
for command in installation['commands']:
    for stream in ('stdout', 'stderr'):
        assert sha(installed / (command['name'] + '.' + stream + '.log')) == command[stream + '_sha256']
assert sha(repo / '.venv/final_checkpoint_checks_v3.py') == verification['runner_sha256']
assert sha(repo / '.venv/run_combined_installed_check_v42.py') == installation['runner_sha256']
output.mkdir(exist_ok=False)
(output / '.gitattributes').write_bytes(b'** -text -whitespace\n')

def copy(path, relative):
    target = output / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(path, target)

for kind, folder in (('source-checks', checks), ('installed-checks', installed)):
    for p in folder.iterdir():
        if p.is_file() and p.suffix in {'.json', '.xml', '.log'}:
            copy(p, Path(kind) / p.name)
for name in ('final_checkpoint_checks_v3.py', 'run_combined_installed_check_v42.py', 'installed_contract_probe_v4.py', 'verify_claude_stop_replay_v2.py'):
    copy(repo / '.venv' / name, Path('helpers') / (name + '.txt'))
copy(Path(__file__), Path('helpers') / (Path(__file__).name + '.txt'))
files = {p.relative_to(output).as_posix(): {'bytes': p.stat().st_size, 'sha256': sha(p)}
         for p in sorted(output.rglob('*')) if p.is_file()}
(output / 'evidence-index.json').write_text(json.dumps({'files': files}, indent=2) + '\n', encoding='utf-8')
s = verification['checks'][0]['counts']
i = installation['contract']['junit_summary']
wheel = next(a for a in verification['artifacts'] if a['file'].endswith('.whl'))
(output / 'README.md').write_text(f"""# Autostart environment package checkpoint

Frozen product `{commit}` passes {s['tests'] - s['skipped']:,} source tests
with {s['skipped']} skips, all six source/package checks, {i['tests'] - i['skipped']:,}
isolated installed cases with {i['skipped']} matching skips, and seven replay
assertions. The installed checks compare individual case outcomes, loaded
module origins, and package bytes against the frozen source and wheel.

Wheel: {wheel['bytes']:,} bytes; SHA-256 `{wheel['sha256']}`.

The [development record](../autostart-environment-development-v1/README.md)
records the reproduced client and test-isolation defects and 139 focused
passes with two skips. Default clients now honor `MNEMOS_AUTO_START`, including
the bridge and managed CLI. Explicit library arguments retain precedence;
stopping a service never autostarts one. Disabled clients still queue capture
and can replay to an existing owner. Disposable host phases independently
pin their database and runtime locations.

The [native v1 failure](../native-claude-outage-failure-v1/README.md) is retained.
An unexpected service wrote three synthetic events and three jobs to the
default database, with zero scoped facts observed. Its owner is stopped,
the private backup is preserved, and no cleanup deletion has been performed.
These tests do not turn that failed attempt into outage recovery evidence.

The new outage acceptance helpers and runner postdate this product checkpoint.
They have separate focused checks and a [frozen successor protocol](../../../plans/2026-09-20-native-claude-service-outage-v2.md).
Their additions do not change the product package. Native outage recovery,
full-module quality, Desktop, and the operational pilot remain separate gates.

The evidence index binds the raw files; this README is outside that index.
The earlier failure/development archives had checkout line-ending conversion
corrected in `c9ae826`; their twenty indexed files match the committed bytes.
""", encoding='utf-8', newline='\n')
print(json.dumps({'archive': str(output), 'files': len(files), 'source_commit': commit,
                  'source_passes': s['tests']-s['skipped'], 'source_skips': s['skipped'],
                  'installed_passes': i['tests']-i['skipped'], 'installed_skips': i['skipped'],
                  'wheel': wheel}))
