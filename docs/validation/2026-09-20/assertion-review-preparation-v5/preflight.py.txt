from pathlib import Path
import asyncio
import hashlib
import json
import subprocess
import sys

root = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-assertion-review-development-v5')
inputs = json.loads((root / 'inputs.json').read_bytes())
prep = json.loads((root / 'preparation.json').read_bytes())
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
for name, digest in prep['files'].items():
    assert sha(root / name) == digest, name
sys.path.insert(0, inputs['snapshot'])
sys.path.insert(0, str(root))
from candidate.assertion_inventory import QualifiedInventory
from candidate.assertion_review import review_qualified_inventory
from mnemos.types import EpisodicEvent

async def main():
    checks = []
    for case in inputs['cases']:
        draft = QualifiedInventory.model_validate_json(inputs['drafts'][case['id']]['response'])
        class Capture:
            supports_structured_output = True
            async def predict(self, prompt, **kwargs):
                payload = json.loads(prompt.split('\nSOURCE EVENTS AND UNTRUSTED DRAFT:\n')[1])
                assert payload['draft'] == draft.model_dump(mode='json')
                assert payload['source_events'][0]['id'] == case['source']['id']
                assert kwargs['format']['$defs']['QualifiedAssertion']['properties']['source_id']['enum'] == [case['source']['id']]
                checks.append({'case_id': case['id'], 'prompt_sha256': hashlib.sha256(prompt.encode()).hexdigest(), 'schema_sha256': hashlib.sha256(json.dumps(kwargs['format'], sort_keys=True).encode()).hexdigest(), 'original_runtime_status': inputs['drafts'][case['id']]['status']})
                return '{"assertions": []}'
        await review_qualified_inventory([EpisodicEvent.model_validate(case['source'])], draft, Capture())
    assert len(checks) == 46
    guards = []
    for args in ([], ['wrong-digest']):
        result = subprocess.run([sys.executable, str(root / 'runner.py'), *args], capture_output=True, text=True)
        assert result.returncode != 0 and 'externally supplied matching preparation digest' in result.stderr
        assert not (root / 'status.json').exists()
        guards.append({'arguments': args, 'exit_code': result.returncode})
    (root / 'offline-preflight.json').write_text(json.dumps({'status': 'passed', 'model_calls': 0, 'cases': checks, 'launch_guards': guards, 'limits': 'Captured prompts and mocked empty outputs verify transport/schema binding only, not semantic decisions.'}, indent=2) + '\n')
    print(json.dumps({'offline_cases': len(checks), 'launch_guards': len(guards), 'model_calls': 0}))

asyncio.run(main())
