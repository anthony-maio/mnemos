from pathlib import Path
import hashlib,json,shutil
from collections import Counter
repo=Path.cwd();root=Path("C:/Users/Ant/AppData/Local/Temp/mnemos-assertion-inventory-development-v4")
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
prep=json.loads((root/"preparation.json").read_bytes()); state=json.loads((root/"status.json").read_bytes())
inputs=json.loads((root/"inputs.json").read_bytes())
assert state["status"]=="completed" and state["processed"]==46
assert state["preparation_sha256"]==sha(root/"preparation.json")
assert sha(root/"preparation.json")==sha(repo/"docs/validation/2026-09-20/assertion-inventory-preparation-v4/preparation.json")
assert state["observations_sha256"]==sha(root/"observations.jsonl")
for n,h in prep["files"].items():assert sha(root/n)==h
for n,h in prep["source_files"].items():assert sha(Path(inputs["snapshot"])/n)==h
parent=repo/"docs/validation/2026-09-20/assertion-inventory-terminal-v3"
binding=json.loads((root/"parent-terminal-binding.json").read_bytes())
for key,name in (("preparation_sha256","preparation.json"),("observations_sha256","observations.jsonl"),("status_sha256","status.json")):
    assert binding[key]==sha(parent/name),key
assert inputs["think"] is False
rows=[json.loads(l) for l in (root/"observations.jsonl").read_text().splitlines()]
assert [r["case_id"] for r in rows]==[c["id"] for c in inputs["cases"]]
assert sum(r["status"]=="error" for r in rows)==state["errors"]
out=repo/"docs/validation/2026-09-20/assertion-inventory-terminal-v4";out.mkdir(exist_ok=False)
for p in root.iterdir():
    if p.is_file():shutil.copyfile(p,out/p.name)
shutil.copyfile(__file__,out/"collector.py.txt")
counts=dict(Counter(r["status"] for r in rows))
(out/"runtime-summary.json").write_text(json.dumps({"counts":counts,"semantic_review":"pending; runtime validity is not accuracy"},indent=2)+"\n")
(out/"review-outputs.json").write_text(json.dumps([{k:v for k,v in r.items() if k!="trace"} for r in rows],indent=2)+"\n")
(out/"README.md").write_text(f"""# Qualified assertion inventory terminal evidence

The frozen thinking-disabled follow-up completed all 46 planned development cases.

configuration-and-route comparison, not an isolated thinking-mode experiment.

Provider errors and validator rejections are preserved separately in raw rows.
Validated quotations and IDs do not prove semantic support or complete recall.

This candidate uses no existing facts and cannot select support or correction
targets. No production prompt, service default or memory store changed. Original
rubrics and ambiguity notes remain bound to the preparation. Review every
retained and omitted meaning before any adoption decision. These are reused,
unblinded development cases with one stochastic repetition, not fresh held-out
quality or full-module/native-host acceptance.
""")
(out/".gitattributes").write_bytes(b"** -text -whitespace\n")
(out/"evidence-index.json").write_text(json.dumps({"files":{p.name:{"bytes":p.stat().st_size,"sha256":sha(p)} for p in out.iterdir() if p.is_file()}},indent=2)+"\n")
print(json.dumps({"archive":str(out),"runtime":counts}))
