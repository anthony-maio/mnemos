from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import shutil
import subprocess
from urllib.request import urlopen

repo = Path.cwd()
parent = repo / 'docs/validation/2026-09-20/source-selection-terminal-v6'
root = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-source-selection-development-v7')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
state = json.loads((parent / 'status.json').read_bytes())
assert state['status'] == 'completed' and state['processed'] == 46
assert sha(parent / 'observations.jsonl') == state['observations_sha256']
prep = json.loads((parent / 'preparation.json').read_bytes())
inputs = json.loads((parent / 'inputs.json').read_bytes())
inputs['model'] = 'qwen2.5:14b'
with urlopen(inputs['endpoint'] + '/api/tags', timeout=10) as r:
    model = next(m for m in json.loads(r.read())['models'] if m['name'] == inputs['model'])
assert model['digest'] == '7cdf5a0187d5c58cc5d369b255592f7841d1c4696d45a8c8a9489440385b22f6'
with urlopen(inputs['endpoint'] + '/api/version', timeout=10) as r:
    version = json.loads(r.read())
root.mkdir(exist_ok=False)
for name in ('candidate/__init__.py', 'candidate/assertion_inventory.py', 'candidate/source_selection.py', 'runner.py',
             'durability-rubric.json', 'durability-rubric-review.json', 'history-rubric.json', 'history-rubric-review.json'):
    assert sha(parent / name) == prep['files'][name]
    dst = root / name
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(parent / name, dst)
(root / 'inputs.json').write_text(json.dumps(inputs, indent=2) + '\n')
(root / 'parent-terminal-binding.json').write_text(json.dumps({
    'directory': str(parent), 'preparation_sha256': sha(parent / 'preparation.json'),
    'observations_sha256': sha(parent / 'observations.jsonl'),
    'status_sha256': sha(parent / 'status.json')}, indent=2) + '\n')
(root / 'protocol.json').write_text(json.dumps({
    'status': 'prepared_not_executed', 'calls': 46, 'repetitions': 1,
    'design': 'Only selected model changes from v6 Qwen3.5:9b to locally available Qwen2.5:14b. Same frozen candidate, provider source, cases, rubrics, prompts, schemas and requested generation settings. Model family/version, tokenizer and weight differences are bundled, not isolated parameter-count effects.',
    'acceptance': 'Review all46 outputs against original meanings and ambiguity notes, including omissions, unwanted retention, kinds and qualifier completeness. Exact source slices and runtime validity are not accuracy. Preserve errors without selective retries or rubric changes.',
    'limits': 'One stochastic repetition on reused unblinded development cases. No fresh holdout, target reconciliation, durable writes or production/default change.'
}, indent=2) + '\n')
shutil.copyfile(__file__, root / 'prepare.py.txt')
new_prep = {k: prep[k] for k in ('source_commit', 'source_files', 'candidate_commit')}
new_prep.update(
    prepared_at=datetime.now(timezone.utc).isoformat(), model=model, ollama_version=version,
    hardware_inventory=subprocess.check_output(['nvidia-smi', '--query-gpu=name,memory.total,driver_version', '--format=csv,noheader'], text=True).strip(),
    parent_preparation_sha256=sha(parent / 'preparation.json'),
    files={str(p.relative_to(root)).replace('\\', '/'): sha(p) for p in root.rglob('*') if p.is_file()}
)
(root / 'preparation.json').write_text(json.dumps(new_prep, indent=2) + '\n')
print(json.dumps({'root': str(root), 'preparation_sha256': sha(root / 'preparation.json'), 'started': False}))
