from pathlib import Path
import difflib
import json

repo = Path(__file__).resolve().parents[1]
temp = Path('C:/Users/Ant/AppData/Local/Temp')
verification = json.loads((temp / 'mnemos-package-checks-forget-preconditions-v2/verification.json').read_bytes())
installation = json.loads((temp / 'mnemos-installed-checks-forget-preconditions-v3/installation.json').read_bytes())
assert verification['status'] == installation['status'] == 'passed'
commit = verification['source_commit']
assert commit == 'e6887deb89608dca34f7bbb463b5d0a3ba973011'
assert installation['bindings_after']['commit'] == commit
wheel = next(item for item in verification['artifacts'] if item['file'].endswith('.whl'))
pairs = [
    ('benchmarks/native_hosts/hermes_lifecycle_run_v13.py', 'benchmarks/native_hosts/hermes_lifecycle_run_v14.py', {
        'native-review-rejection-package-lifecycle-v13': 'native-copyable-preconditions-package-lifecycle-v14',
        'db217509477d92b9a82bd7e11d4a4bffb542fa29': commit,
        'Native v13 requires the frozen review-rejection package': 'Native v14 requires the frozen copyable-preconditions package',
        '8b1f560e98b93376fc2aa073be7c1500997d888d09ae87573d5d70ec8694ad00': wheel['sha256'],
        '2026-09-20-hermes-native-lifecycle-protocol-v13.md': '2026-09-20-hermes-native-lifecycle-protocol-v14.md',
    }),
    ('benchmarks/native_hosts/hermes_cohort_run_v11.py', 'benchmarks/native_hosts/hermes_cohort_run_v12.py', {
        'Hermes v13 lifecycles': 'Hermes v14 lifecycles',
        'hermes_lifecycle_run_v13': 'hermes_lifecycle_run_v14',
        '2026-09-20-hermes-native-lifecycle-protocol-v13.md': '2026-09-20-hermes-native-lifecycle-protocol-v14.md',
        'hermes_cohort_run_v11.py': 'hermes_cohort_run_v12.py',
    }),
]
diff = []
for before, after, substitutions in pairs:
    original = (repo / before).read_text(encoding='utf-8')
    updated = original
    for old, new in substitutions.items():
        assert old in updated, old
        updated = updated.replace(old, new)
    restored = updated
    for old, new in reversed(list(substitutions.items())):
        restored = restored.replace(new, old)
    assert restored == original
    assert not (repo / after).exists()
    (repo / after).write_text(updated, encoding='utf-8', newline='\n')
    diff.extend(difflib.unified_diff(original.splitlines(True), updated.splitlines(True), before, after))
(repo / '.venv/hermes-v14-binding-diff.txt').write_text(''.join(diff), encoding='utf-8', newline='\n')
protocol = repo / 'docs/plans/2026-09-20-hermes-native-lifecycle-protocol-v14.md'
