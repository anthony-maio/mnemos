from pathlib import Path
import hashlib,json,shutil,sqlite3
from contextlib import closing
repo=Path.cwd();sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
for version in [1,2]:
 src=Path(f'C:/Users/Ant/AppData/Local/Temp/mnemos-all-module-provider-outage-v{version}')
 record=json.loads((src/'offline-result.json').read_bytes())
 expected='failed' if version==1 else 'offline_restart_passed_recovery_not_run'
 assert record['status']==expected,record['status']
 assert record['descriptor_removed'] and record['verified_modules']==55

 for name,digest in prep['files'].items(): assert sha(src/name)==digest
 out=repo/f'docs/validation/2026-09-20/all-module-provider-outage-v{version}'
 out.mkdir(exist_ok=False)
 for p in src.iterdir():
  if p.suffix in {'.json','.py','.log'}: shutil.copyfile(p,out/p.name)
 with closing(sqlite3.connect((src/'memory.sqlite').as_uri()+'?mode=ro',uri=True)) as source,closing(sqlite3.connect(out/'memory.sqlite')) as target:
  source.backup(target)
  assert target.execute('PRAGMA integrity_check').fetchone()==('ok',)
  assert target.execute('PRAGMA foreign_key_check').fetchall()==[]
 (out/'.gitattributes').write_bytes(b'** -text -whitespace\n')
 result=('The probe failed on its incorrect expectation of exactly one attempt per maintenance call. Both jobs remained pending after three attempts and no facts were committed. Restart was NOT RUN. Both source events and their pending jobs survived unchanged. No facts were committed. All 55 installed Python modules matched the verified wheel.')
 (out/'README.md').write_text(f'''# Installed provider outage, all module flags enabled: v{version}

{result}
