from pathlib import Path
import asyncio, hashlib, json, runpy, sys
root=Path("C:/Users/Ant/AppData/Local/Temp/mnemos-assertion-inventory-development-v1")
sys.path.insert(0,str(root))
ns=runpy.run_path(str(root/"runner.py"),run_name="offline_preflight")
class Offline:
    supports_structured_output=True
    async def predict(self,prompt,**kwargs):
        assert "EXISTING FACTS" not in prompt
        assert "cedar-quartz-118" not in prompt
        assert kwargs["format"]["$defs"]["Assertion"]["properties"]["source_id"]["enum"]==[self.source_id]
        return '{"assertions":[]}'
async def main():
    count=0
    for c in ns["inputs"]["cases"]:
        provider=Offline();provider.source_id=c["source"]["id"]
        result=await ns["inventory_assertions"]([ns["EpisodicEvent"].model_validate(c["source"])],provider)
        assert result.assertions==[]
        count+=1
    assert count==46
    return count
