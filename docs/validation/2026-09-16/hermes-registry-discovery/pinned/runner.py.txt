def _wsl_path(path: Path) -> str:
    resolved = path.resolve()
    drive = resolved.drive.rstrip(":").lower()
    if not drive:
        raise ValueError(f"Expected a Windows drive path: {resolved}")
    tail = str(resolved)[len(resolved.anchor) :].replace("\\", "/").lstrip("/")
    return "/mnt/" + drive + "/" + tail


def _sha256(path: Path, *, normalize: bool = False) -> str:
    data = path.read_bytes()
    if normalize:
        data = data.decode("utf-8").replace("\r\n", "\n").replace("\r", "\n").encode("utf-8")
    return hashlib.sha256(data).hexdigest()


def _clean_snapshot(snapshot: Path, commit: str) -> dict[str, str]:
    actual = subprocess.run(
        ["git", "-C", str(snapshot), "rev-parse", "HEAD"],
        capture_output=True,
        text=True,
        check=True,
    ).stdout.strip()
    if actual != commit:
        raise RuntimeError(f"snapshot HEAD {actual} does not match requested {commit}")
    status = subprocess.run(
        ["git", "-C", str(snapshot), "status", "--porcelain", "--untracked-files=all"],
        capture_output=True,
        text=True,
        check=True,
    ).stdout
    if status.strip():
        raise RuntimeError(f"snapshot is dirty: {status.strip()}")
    adapter = snapshot / "mnemos" / "adapters" / "hermes.py"
    if not adapter.is_file():
        raise RuntimeError(f"snapshot has no adapter: {adapter}")
    return {
        "head": actual,
        "tracked_source_clean": True,
        "adapter_sha256_raw": _sha256(adapter),
        "adapter_sha256_normalized": _sha256(adapter, normalize=True),
    }


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--snapshot", required=True, type=Path)
    parser.add_argument("--commit", required=True)
    parser.add_argument("--output", required=True, type=Path)
    parser.add_argument("--expected", choices=("available", "unavailable"), required=True)
    args = parser.parse_args()
    snapshot = args.snapshot.resolve()
    output = args.output.resolve()
    if output.exists():
        raise SystemExit(f"refusing to overwrite {output}")
    source = _clean_snapshot(snapshot, args.commit)
    output.mkdir(parents=True)
    expected_available = args.expected == "available"
    expected_reason = "" if expected_available else "Mnemos scope registry is not configured"
    payload = (
        PAYLOAD.replace("__SNAPSHOT__", json.dumps(_wsl_path(snapshot)))
        .replace("__EXPECTED_RAW__", json.dumps(source["adapter_sha256_raw"]))
        .replace("__EXPECTED_NORMALIZED__", json.dumps(source["adapter_sha256_normalized"]))
        .replace("__EXPECTED_AVAILABLE__", repr(expected_available))
        .replace("__EXPECTED_REASON__", repr(expected_reason))
    )
    payload_path = output / "payload.py"
    payload_path.write_text(payload, encoding="utf-8")
    started = datetime.now(timezone.utc)
    command = [
        "wsl.exe",
        "-d",
        "Debian",
        "--",
        "env",
        "-i",
        "PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin",
        "HOME=/tmp/mnemos-hermes-pinned-runner",
        WSL_PYTHON,
        "-c",
        payload,
    ]
    result = subprocess.run(command, capture_output=True, text=True, timeout=30, check=False)
    finished = datetime.now(timezone.utc)
    stdout_lines = [line for line in result.stdout.splitlines() if line.strip()]
    child = json.loads(stdout_lines[-1]) if stdout_lines and result.returncode == 0 else None
    observed_available = bool(
        child and child.get("provider_loaded") and child.get("provider_available")
    )
    report = {
        "status": (
            "passed"
            if result.returncode == 0 and observed_available == (args.expected == "available")
            else "failed"
        ),
        "snapshot": str(snapshot),
        "requested_commit": args.commit,
        "expected": args.expected,
        "started_utc": started.isoformat(),
        "finished_utc": finished.isoformat(),
        "source_verification": source,
        "runner": str(Path(__file__).resolve()),
        "runner_sha256": _sha256(Path(__file__).resolve()),
        "command": [*command[:-2], "<payload.py contents>"],
        "payload_file": "payload.py",
        "payload_sha256": _sha256(payload_path),
        "returncode": result.returncode,
        "stdout_log": "stdout.txt",
        "stderr_log": "stderr.txt",
        "child": child,
        "observation": (
            "provider loaded and available"
            if observed_available
            else "provider unavailable or not loaded"
        ),
    }
    (output / "stdout.txt").write_text(result.stdout, encoding="utf-8")
    (output / "stderr.txt").write_text(result.stderr, encoding="utf-8")
    (output / "report.json").write_text(
        json.dumps(report, indent=2, sort_keys=True) + "\n", encoding="utf-8"
    )
    print(json.dumps(report, indent=2, sort_keys=True))
    return 0 if report["status"] == "passed" else 1


if __name__ == "__main__":
    raise SystemExit(main())
