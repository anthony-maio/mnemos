"""Assert durable Stop replay behavior from a clean, explicitly pinned source tree."""

from __future__ import annotations

import argparse
import asyncio
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
from pathlib import Path
import subprocess
import sys
import tempfile
import zipfile


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--snapshot", required=True, type=Path)
    parser.add_argument("--commit", required=True)
    parser.add_argument("--output", required=True, type=Path)
    parser.add_argument("--installed-package", type=Path)
    parser.add_argument("--wheel", type=Path)
    args = parser.parse_args()
    root = args.snapshot.resolve()
    output = args.output.resolve()
    if output.exists():
        raise SystemExit("Refusing to overwrite existing evidence")
    record = {
        "schema_version": 1,
        "probe": "claude_stop_replay_positive_assertions",
        "status": "failed",
        "started_at": datetime.now(timezone.utc).isoformat(),
        "source_commit_expected": args.commit,
        "source_checkout": str(root),
        "providers": {"embedding": "constant", "cognition": "unavailable"},
        "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    }
    try:
        actual = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=root, text=True).strip()
        record["source_commit_actual"] = actual
        if actual != args.commit:
            raise RuntimeError("Source commit differs from explicit pin")
        dirty = subprocess.check_output(
            ["git", "status", "--porcelain", "--untracked-files=no"], cwd=root, text=True
        )
        if dirty.strip():
            raise RuntimeError("Tracked source changes would invalidate the pin")
        # Import the preserved reproducer by path; its main entrypoint is not called.
        original = root / "docs/validation/2026-09-16/claude-stop-replay/runner.py.txt"
        if not original.exists():
            candidates = list(
                (root / "docs/validation/2026-09-16/claude-stop-replay").glob("*.py.txt")
            )
            if len(candidates) != 1:
                raise RuntimeError("Cannot uniquely locate preserved reproducer")
            original = candidates[0]
        from importlib.machinery import SourceFileLoader

        loader = SourceFileLoader("preserved_stop_replay_probe", str(original))
        spec = importlib.util.spec_from_loader(loader.name, loader)
        if spec is None:
            raise RuntimeError("Cannot load reproducer")
        probe = importlib.util.module_from_spec(spec)
        loader.exec_module(probe)
        record["reproducer_path"] = str(original)
        record["reproducer_sha256"] = hashlib.sha256(original.read_bytes()).hexdigest()
        if bool(args.installed_package) != bool(args.wheel):
            raise RuntimeError("Installed validation requires both package path and wheel")
        import_root = args.installed_package.resolve() if args.installed_package else root
        sys.path.insert(0, str(import_root))
        import mnemos

        imported = Path(str(mnemos.__file__)).resolve()
        record["imported_mnemos"] = str(imported)
        record["import_mode"] = "installed_wheel" if args.installed_package else "source_checkout"
        if not imported.is_relative_to(import_root):
            raise RuntimeError("Imported package is outside declared import root")
        if args.installed_package:
            wheel = args.wheel.resolve()
            record["wheel"] = {
                "path": str(wheel),
                "sha256": hashlib.sha256(wheel.read_bytes()).hexdigest(),
            }
            members = []
            with zipfile.ZipFile(wheel) as archive:
                for member in archive.namelist():
                    if member.startswith("mnemos/") and not member.endswith("/"):
                        if (import_root / member).read_bytes() != archive.read(member):
                            raise RuntimeError("Installed package differs from wheel: " + member)
                        members.append(member)
            tracked = subprocess.check_output(
                ["git", "ls-files", "mnemos"], cwd=root, text=True
            ).splitlines()
            for member in tracked:
                if (root / member).is_file():
                    installed = (import_root / member).read_bytes().replace(b"\r\n", b"\n")
                    source = (root / member).read_bytes().replace(b"\r\n", b"\n")
                    if source != installed:
                        raise RuntimeError(
                            "Installed package differs from pinned source: " + member
                        )
            record["package_binding"] = {
                "exact_wheel_members": len(members),
                "source_files_normalized_lf": len(tracked),
            }
        fixture = Path(tempfile.mkdtemp(prefix="mnemos-claude-stop-fixed-", dir=r"D:\WindowsTemp"))
        record["fixture_directory"] = str(fixture)
        result = asyncio.run(probe._run_probe(root, fixture))
        record["reproduction"] = result
        first = result["events_after_first_stop"]
        second = result["events_after_second_stop"]
        assertions = {
            "first_stop_captured_two": result["first_stop"]["captured"] == 2,
            "first_stop_ok": result["first_stop"]["status"] == "ok",
            "replay_captured_zero": result["second_stop_after_restart"]["captured"] == 0,
            "replay_explicitly_degraded_without_turn_id": result["second_stop_after_restart"][
                "status"
            ]
            == "degraded",
            "first_store_has_two_events": len(first) == 2,
            "restarted_store_has_same_two_events": len(second) == 2
            and sorted(first, key=lambda x: x["id"]) == sorted(second, key=lambda x: x["id"]),
            "one_native_assistant": result["assistant_source_refs_after_restart"]
            == ["native-assistant-uuid"],
        }
        record["assertions"] = assertions
        if not all(assertions.values()):
            raise RuntimeError("At least one positive replay assertion failed")
        record["status"] = "passed"
    except Exception as exc:
        record["error"] = type(exc).__name__
        record["error_detail"] = str(exc)
    finally:
        record["finished_at"] = datetime.now(timezone.utc).isoformat()
        output.parent.mkdir(parents=True, exist_ok=True)
        output.write_text(json.dumps(record, indent=2, sort_keys=True) + "\n", encoding="utf-8")
    print(
        json.dumps(
            {"status": record["status"], "output": str(output), "error": record.get("error_detail")}
        )
    )
    return 0 if record["status"] == "passed" else 1


if __name__ == "__main__":
    raise SystemExit(main())
