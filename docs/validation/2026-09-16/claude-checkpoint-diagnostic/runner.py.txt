The helper invokes only the Mnemos adapter against a disposable local service.
It does not invoke Claude, a model, or a network provider.  A child process
keeps per-operation trace records flushed before each operation completes so a
timeout still leaves evidence for the checkpoint write boundary.
"""

def _sha256(value: bytes | str) -> str:
    raw = value.encode("utf-8") if isinstance(value, str) else value
    return hashlib.sha256(raw).hexdigest()


def _pinned_package_provenance(
    package_path: Path, *, wheel_path: Path, expected_wheel_sha256: str
) -> dict[str, Any]:
    listing = subprocess.run(
        [
            "git",
            "-C",
            str(WORKTREE),
            "ls-tree",
            "-r",
            "--name-only",
            SOURCE_COMMIT,
            "mnemos",
        ],
        capture_output=True,
        text=True,
        check=True,
        timeout=10,
    )
    files = [Path(item) for item in listing.stdout.splitlines() if item.strip()]
    if not files:
        raise RuntimeError("pinned source has no mnemos package files")
    mismatches: list[str] = []
    for relative in files:
        expected = subprocess.run(
            ["git", "-C", str(WORKTREE), "show", f"{SOURCE_COMMIT}:{relative.as_posix()}"],
            capture_output=True,
            check=True,
            timeout=10,
        ).stdout.replace(b"\r\n", b"\n")
        installed = (
            (package_path.parent / Path(*relative.parts[1:])).read_bytes().replace(b"\r\n", b"\n")
        )
        if installed != expected:
            mismatches.append(relative.as_posix())
    if mismatches:
        raise RuntimeError("installed package differs from pinned source")
    observed_wheel_sha256 = _sha256(wheel_path.read_bytes())
    if observed_wheel_sha256 != expected_wheel_sha256:
        raise RuntimeError("wheel archive does not match the expected digest")
    wheel_members = 0
    with zipfile.ZipFile(wheel_path) as archive:
        for item in archive.infolist():
            if item.is_dir() or not item.filename.startswith("mnemos/"):
                continue
            wheel_members += 1
            installed = package_path.parent / Path(*Path(item.filename).parts[1:])
            if installed.read_bytes() != archive.read(item):
                raise RuntimeError("installed package bytes differ from wheel")
    return {
        "source_commit": SOURCE_COMMIT,
        "package_files_checked": len(files),
        "wheel_package_members_checked": wheel_members,
        "wheel_path": str(wheel_path),
        "wheel_sha256_expected": expected_wheel_sha256,
        "wheel_sha256_observed": observed_wheel_sha256,
    }


def _private_mkdir(path: Path) -> None:
    from mnemos.service import _protect_private_path

    path.mkdir(parents=True, exist_ok=False)
    _protect_private_path(path)


def _clean_environment(overrides: Mapping[str, str]) -> dict[str, str]:
    environment = dict(os.environ)
    for key in list(environment):
        if key.startswith("MNEMOS_") or key in {"PYTHONPATH", "CLAUDE_CONFIG_DIR"}:
            environment.pop(key, None)
    environment.update(overrides)
    return environment


def _terminate_process_tree(process: subprocess.Popen[str]) -> bool:
    if process.poll() is not None:
        return False
    if os.name == "nt":
        subprocess.run(
            ["taskkill", "/PID", str(process.pid), "/T", "/F"],
            capture_output=True,
            text=True,
            check=False,
            timeout=10,
        )
    else:  # pragma: no cover - the diagnostic runs on Windows
        process.kill()
    try:
        process.wait(timeout=10)
    except subprocess.TimeoutExpired:
        process.kill()
        process.wait(timeout=10)
    return True

def _run_probe(
    *,
    child: Path,
    event: str,
    payload: Mapping[str, Any],
    trace: Path,
    stack: Path,
    cwd: Path,
    environment: Mapping[str, str],
) -> dict[str, Any]:
    started = time.monotonic()
    process = subprocess.Popen(
        [sys.executable, str(child), event, str(trace), str(stack)],
        cwd=str(cwd),
        env=dict(environment),
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8",
        errors="replace",
        shell=False,
        creationflags=getattr(subprocess, "CREATE_NEW_PROCESS_GROUP", 0),
    )
    timed_out = False
    try:
        stdout, stderr = process.communicate(
            input=json.dumps(dict(payload), separators=(",", ":")),
            timeout=CHILD_TIMEOUT_SECONDS,
        )
    except subprocess.TimeoutExpired:
        timed_out = True
        _terminate_process_tree(process)
        stdout, stderr = process.communicate(timeout=15)
    finally:
        if process.poll() is None:
            _terminate_process_tree(process)
    result: dict[str, Any] = {
        "event": event,
        "returncode": process.returncode,
        "timed_out": timed_out,
        "elapsed_ms": round((time.monotonic() - started) * 1000, 1),
        "stdout_sha256": _sha256(stdout),
        "stdout_bytes": len(stdout.encode("utf-8")),
        "stderr_sha256": _sha256(stderr),
        "stderr_bytes": len(stderr.encode("utf-8")),
        "trace_path": str(trace),
        "trace_bytes": trace.stat().st_size if trace.exists() else 0,
        "stack_path": str(stack),
        "stack_bytes": stack.stat().st_size if stack.exists() else 0,
    }
    if stdout.strip():
        try:
            parsed = json.loads(stdout)
            if isinstance(parsed, Mapping):
                result["adapter_result"] = dict(parsed)
        except json.JSONDecodeError:
            result["adapter_result_parse_error"] = True
    return result


def _trace_summary(path: Path) -> dict[str, Any]:
    rows: list[dict[str, Any]] = []
    if path.exists():
        for line in path.read_text(encoding="utf-8").splitlines():
            try:
                row = json.loads(line)
            except json.JSONDecodeError:
                continue
            if isinstance(row, Mapping):
                rows.append(dict(row))
    operations: dict[str, dict[str, int]] = {}
    for row in rows:
        operation = str(row.get("operation"))
        phases = operations.setdefault(operation, {})
        phase = str(row.get("phase"))
        phases[phase] = phases.get(phase, 0) + 1
    return {"path": str(path), "row_count": len(rows), "operations": operations, "rows": rows}


def _checkpoint_summary(directory: Path) -> list[dict[str, Any]]:
    result: list[dict[str, Any]] = []
    for path in sorted([*directory.glob("*.json"), *directory.glob("*.tmp")]):
        try:
            value = json.loads(path.read_text(encoding="utf-8"))
        except (OSError, json.JSONDecodeError):
            result.append({"path": str(path), "json_complete": False})
            continue
        if not isinstance(value, Mapping):
            continue
        scope = value.get("scope")
        result.append(
            {
                "path": str(path),
                "sha256": _sha256(path.read_bytes()),
                "keys": sorted(str(key) for key in value),
                "offset": value.get("offset"),
                "identity_version": value.get("identity_version"),
                "scope_keys": (
                    sorted(str(key) for key in scope) if isinstance(scope, Mapping) else []
                ),
                "scope_host": scope.get("host") if isinstance(scope, Mapping) else None,
            }
        )
    return result


def _parse_args(argv: list[str] | None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Diagnose Mnemos Claude checkpoint latency offline."
    )
    parser.add_argument("--wheel", type=Path, required=True, help="installed wheel archive")
    parser.add_argument("--expected-wheel-sha256", required=True, help="wheel SHA256 digest")
    args = parser.parse_args(argv)
    digest = args.expected_wheel_sha256.lower()
    if len(digest) != 64 or any(char not in "0123456789abcdef" for char in digest):
        parser.error("--expected-wheel-sha256 must be a 64-character hexadecimal digest")
    args.expected_wheel_sha256 = digest
    return args

            }
        )
        if "fixture" not in report:
            report["fixture"] = str(fixture)
        (fixture / "runner-error.txt").write_text(traceback.format_exc(), encoding="utf-8")
    finally:
        if service is not None:
            try:
                service.stop()
            except Exception:
                pass
        if store is not None:
            store.close()
        report["descriptor_absent_after_shutdown"] = not (fixture / "runtime.json").exists()
        report["private_fixture_preserved"] = True
        report["finished_at"] = datetime.now(timezone.utc).isoformat()
        (OUTPUT_DIR / "report.json").write_text(
            json.dumps(report, indent=2) + "\n", encoding="utf-8"
        )
    print(json.dumps({key: report.get(key) for key in ("status", "fixture", "error_type")}))
    return 0 if report["status"] == "diagnostic_complete" else 1


if __name__ == "__main__":
    raise SystemExit(main())
