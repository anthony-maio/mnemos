def _sha256(value: bytes | str) -> str:
    raw = value.encode("utf-8") if isinstance(value, str) else value
    return hashlib.sha256(raw).hexdigest()


def _assert_v8_helper_source() -> str:
    observed = _sha256(V8_HELPER_PATH.read_bytes())
    if observed != V8_HELPER_SHA256:
        raise RuntimeError("The pinned v8 helper source changed")
    return observed


def _pinned_package_provenance(
    package_path: Path, *, wheel_path: Path, expected_wheel_sha256: str
) -> dict[str, Any]:
    """Compare installed package files with the pinned source and wheel."""
    listing = subprocess.run(
        [
            "git",
            "-C",
            str(WORKTREE),
            "ls-tree",
            "-r",
            "--name-only",
            SOURCE_COMMIT,
            "mnemos",
        ],
        capture_output=True,
        text=True,
        check=True,
        timeout=10,
    )
    files = [Path(item) for item in listing.stdout.splitlines() if item.strip()]
    if not files:
        raise RuntimeError("Pinned source has no package files")
    mismatches: list[str] = []
    for relative in files:
        expected = subprocess.run(
            ["git", "-C", str(WORKTREE), "show", f"{SOURCE_COMMIT}:{relative.as_posix()}"],
            capture_output=True,
            check=True,
            timeout=10,
        ).stdout.replace(b"\r\n", b"\n")
        installed = (
            (package_path.parent / Path(*relative.parts[1:])).read_bytes().replace(b"\r\n", b"\n")
        )
        if installed != expected:
            mismatches.append(relative.as_posix())
    wheel_digest = _sha256(wheel_path.read_bytes())
    if wheel_digest != expected_wheel_sha256:
        raise RuntimeError("Wheel archive does not match the expected digest")
    with zipfile.ZipFile(wheel_path) as archive:
        for item in archive.infolist():
            if item.is_dir() or not item.filename.startswith("mnemos/"):
                continue
            installed = package_path.parent / Path(*Path(item.filename).parts[1:])
            if installed.read_bytes() != archive.read(item):
                mismatches.append("wheel:" + item.filename)
    if mismatches:
        raise RuntimeError("Installed package differs from pinned source or wheel")
    return {
        "source_commit": SOURCE_COMMIT,
        "package_files_checked": len(files),
        "mismatches": [],
        "wheel_sha256_expected": expected_wheel_sha256,
        "wheel_sha256_observed": wheel_digest,
        "wheel_path": str(wheel_path),
    }


def _run_local_cli(args: list[str], *, timeout: float = 5.0) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [str(CODEX_EXE), *args],
        capture_output=True,
        text=True,
        encoding="utf-8",
        errors="replace",
        check=False,
        timeout=timeout,
        shell=False,
    )


def _verify_cli() -> dict[str, Any]:
    if os.name != "nt":
        raise RuntimeError("Native Codex lifecycle runner requires Windows")
    if not CODEX_EXE.is_file():
        raise FileNotFoundError(f"Direct Codex executable is missing: {CODEX_EXE}")
    version = _run_local_cli(["--version"])
    top_help = _run_local_cli(["--help"])
    exec_help = _run_local_cli(["exec", "--help"])
    help_text = (exec_help.stdout or "") + "\n" + (exec_help.stderr or "")
    required = {
        "--json",
        "--sandbox",
        "read-only",
        "--cd",
        "--skip-git-repo-check",
        "--ephemeral",
    }
    missing = sorted(flag for flag in required if flag not in help_text)
    if version.returncode or top_help.returncode or exec_help.returncode or missing:
        raise RuntimeError("Codex CLI capability probe failed: " + ", ".join(missing))
    return {
        "path": str(CODEX_EXE),
        "version": (version.stdout or version.stderr).strip(),
        "help_flags_verified": sorted(required),
        "approval_flag_placement": {
            "top_level": "--ask-for-approval" in (top_help.stdout or ""),
            "exec": "--ask-for-approval" in help_text,
            "command_uses": False,
        },
    }

def _clean_environment(overrides: Mapping[str, str]) -> dict[str, str]:
    environment = dict(os.environ)
    for key in list(environment):
        if key.startswith("MNEMOS_") or key in {
            "PYTHONPATH",
            "VIRTUAL_ENV",
            "CODEX_HOME",
        }:
            environment.pop(key, None)
    environment.update(overrides)
    return environment


def _terminate_process_tree(process: subprocess.Popen[str]) -> bool:
    if process.poll() is not None:
        return False
    if os.name == "nt":
        subprocess.run(
            ["taskkill", "/PID", str(process.pid), "/T", "/F"],
            capture_output=True,
            text=True,
            check=False,
            timeout=10,
        )
    else:  # pragma: no cover - the runner is Windows-only
        process.kill()
    try:
        process.wait(timeout=10)
    except subprocess.TimeoutExpired:
        process.kill()
        process.wait(timeout=10)
    return True


def _numeric_usage(value: Any) -> dict[str, int | float]:
    if not isinstance(value, Mapping):
        return {}
    return {
        str(key): item
        for key, item in value.items()
        if isinstance(item, (int, float)) and not isinstance(item, bool)
    }

    unreviewed_errors = 0
    answer_digests: list[str] = []
    for line in stdout.splitlines():
        try:
            value = json.loads(line)
        except json.JSONDecodeError:
            non_json += 1
            continue
        if not isinstance(value, Mapping):
            continue
        item = value.get("item")
        item_type = (
            item.get("type")
            if isinstance(item, Mapping)
            else value.get("type") if isinstance(value.get("type"), str) else None
        )
        name_values: list[Any] = []
        tool_record = item if isinstance(item, Mapping) else value
        if isinstance(tool_record, Mapping):
            name_values.extend(
                [tool_record.get("name"), tool_record.get("tool_name"), tool_record.get("tool")]
            )
            function = tool_record.get("function")
            if isinstance(function, Mapping):
                name_values.append(function.get("name"))
            text_value = tool_record.get("text")
            if isinstance(text_value, str) and item_type == "agent_message":
                answer_digests.append(_sha256(text_value))
        tool_name = next(
            (
                candidate
                for candidate in name_values
                if isinstance(candidate, str) and candidate.startswith("mnemos_")
            ),
            None,
        )
        server = tool_record.get("server") if isinstance(tool_record, Mapping) else None
        is_mnemos_tool_call = (
            item_type == "mcp_tool_call" and server == "mnemos" and tool_name is not None
        )
        if not is_mnemos_tool_call:
            tool_name = None
        else:
            tool_names.add(tool_name)
        if tool_name is not None and isinstance(tool_record, Mapping):
            raw_result = tool_record.get("result")
            if raw_result is None:
                raw_result = tool_record.get("output")
            if raw_result is None:
                raw_result = value.get("result")
            if raw_result is None:
                raw_result = value.get("output")

            if isinstance(raw_result, Mapping):

                result_parseable = True
            elif result_text:
                try:
                    parsed_result = json.loads(result_text)
                except json.JSONDecodeError:
                    parsed_result = None
                if isinstance(parsed_result, Mapping):
                    result_object = parsed_result
                    result_parseable = True
            if result_object is not None and isinstance(result_object.get("content"), list):
                for content_item in result_object["content"]:
                    if not isinstance(content_item, Mapping):
                        continue
                    content_text = content_item.get("text")
                    if not isinstance(content_text, str):
                        continue
                    try:
                        parsed_content = json.loads(content_text)
                    except json.JSONDecodeError:

                        result_parseable = True

            item_status = (
                tool_record.get("status")
                if isinstance(tool_record.get("status"), str)

                    else (
                        "completed"

                            value["type"].lower().endswith((".completed", "_completed"))

            )
            arguments: Mapping[str, Any] = {}
            for candidate in (tool_record.get("arguments"), tool_record.get("input")):

                    arguments = candidate

                if isinstance(candidate, str):

                        parsed_arguments = json.loads(candidate)

                        parsed_arguments = None
                    if isinstance(parsed_arguments, Mapping):
                        arguments = parsed_arguments

            tool_calls.append(

def _successful_call(call: Mapping[str, Any]) -> bool:
    return (
        call.get("item_type") == "mcp_tool_call"
        and call.get("server") == "mnemos"
        and call.get("item_status") in {"completed", "success", "succeeded", "ok"}
        and call.get("result_present") is True
        and call.get("result_parseable") is True
        and call.get("result_error") is False
    )


def _successful_tool(summary: Mapping[str, Any], name: str) -> bool:
    """Require a completed native call with a returned, non-error result."""
    calls = summary.get("tool_calls")
    if not isinstance(calls, list):
        return False
    return any(
        isinstance(call, Mapping) and call.get("name") == name and _successful_call(call)
        for call in calls
    )


def _result_contains(summary: Mapping[str, Any], name: str, label: str) -> bool:
    calls = summary.get("tool_calls")
    if not isinstance(calls, list):
        return False
    return any(
        isinstance(call, Mapping)
        and call.get("name") == name
        and _successful_call(call)
        and isinstance(call.get("result_contains"), Mapping)
        and call["result_contains"].get(label) is True
        for call in calls
    )


def _result_contains_any(summary: Mapping[str, Any], name: str, labels: tuple[str, ...]) -> bool:
    return any(_result_contains(summary, name, label) for label in labels)


def _service_status(summary: Mapping[str, Any], name: str) -> list[str]:
    calls = summary.get("tool_calls")
    if not isinstance(calls, list):
        return []
    return sorted(
        {
            str(call["service_status"])
            for call in calls
            if isinstance(call, Mapping)
            and call.get("name") == name
            and _successful_call(call)
            and isinstance(call.get("service_status"), str)
        }
    )


def _successful_service_status(summary: Mapping[str, Any], name: str, status: str) -> bool:
    calls = summary.get("tool_calls")
    if not isinstance(calls, list):
        return False
    return any(
        isinstance(call, Mapping)
        and call.get("name") == name
        and _successful_call(call)
        and call.get("service_status") == status
        for call in calls
    )


def _successful_calls(summary: Mapping[str, Any], name: str) -> list[Mapping[str, Any]]:
    calls = summary.get("tool_calls")
    if not isinstance(calls, list):
        return []
    return [
        call
        for call in calls
        if isinstance(call, Mapping) and call.get("name") == name and _successful_call(call)
    ]


def _captured_event_summaries(
    store: Any, *, assertion: str, project_id: str, profile: str
) -> list[dict[str, Any]]:
    """Return automatic hook observations without exposing their content."""
    events = store.list_episodic_events(
        scope="project",
        scope_id=project_id,
        user_profile_id=profile,
        limit=10_000,
    )
    observed: list[dict[str, Any]] = []
    for event in events:
        if assertion not in event.content:
            continue
        if event.metadata.get("explicit_remember") is not False:
            continue
        source_ref = event.metadata.get("source_ref")
        if (
            event.source != "codex"
            or event.metadata.get("source_kind") != "codex_hook"
            or not isinstance(source_ref, str)
            or source_ref.partition(":")[0] not in {"UserPromptSubmit", "Stop"}
        ):
            continue
        observed.append(
            {
                "event_id": event.id,
                "event_id_sha256": _sha256(event.id),
                "source_kind": event.metadata.get("source_kind"),
                "hook_event": source_ref.partition(":")[0],
                "role": event.role,
                "status": event.status,
                "content_sha256": _sha256(event.content),
                "content_length": len(event.content),
            }
        )
    return observed


def _captured_fact_summaries(
    store: Any,
    *,
    assertion: str,
    target_fact_id: str,
    project_id: str,
    profile: str,
    observed_event_ids: set[str],
) -> list[dict[str, Any]]:
    """Find accepted automatic facts tied to observed hook events."""
    observed: list[dict[str, Any]] = []
    for fact in store.list_chunks(limit=10_000):
        if fact.id == target_fact_id:
            continue
        if assertion not in fact.content:
            continue
        if (
            fact.metadata.get("source") != "sleep_consolidation"
            or fact.metadata.get("scope") != "project"
            or fact.metadata.get("scope_id") != project_id
            or fact.metadata.get("user_profile_id") != profile
            or fact.metadata.get("source_event_ids") is None
        ):
            continue
        evidence_ids = {
            item.source_event_id
            for item in store.list_fact_evidence(fact.id)
            if item.source_event_id
        }
        tied_ids = evidence_ids & observed_event_ids
        if not tied_ids:
            continue
        observed.append(
            {
                "fact_id": fact.id,
                "fact_id_sha256": _sha256(fact.id),
                "version": fact.version,
                "content_sha256": _sha256(fact.content),
                "content_length": len(fact.content),
                "source_event_id_sha256": sorted(_sha256(item) for item in tied_ids),
                "evidence_count": len(evidence_ids),
            }
        )
    return observed


def _validated_related_selection(
    preview_call: Mapping[str, Any], related_call: Mapping[str, Any], target_id: str
) -> tuple[set[str], set[str], set[str], set[str]]:
    """Validate that the native apply selected only IDs from its preview."""
    preview_candidate_ids = {
        item for item in preview_call.get("preview_candidate_ids", []) if isinstance(item, str)
    }
    preview_source_ids = {
        item for item in preview_call.get("preview_source_event_ids", []) if isinstance(item, str)
    }
    selected_fact_ids = {
        item for item in related_call.get("selected_fact_ids", []) if isinstance(item, str)
    }
    selected_source_ids = {
        item for item in related_call.get("selected_source_event_ids", []) if isinstance(item, str)
    }
    blocked_candidate_ids = {
        item
        for item in preview_call.get("preview_blocked_candidate_ids", [])
        if isinstance(item, str)
    }
    blocked_source_ids = {
        item
        for item in preview_call.get("preview_blocked_source_event_ids", [])
        if isinstance(item, str)
    }
    if blocked_candidate_ids or blocked_source_ids:
        raise RuntimeError("Forget preview contains a blocked candidate or source selection")
    if selected_fact_ids - preview_candidate_ids - {target_id}:
        raise RuntimeError("Related forget selected a fact absent from the preview")
    if selected_source_ids - preview_source_ids:
        raise RuntimeError("Related forget selected a source absent from the preview")
    return preview_candidate_ids, preview_source_ids, selected_fact_ids, selected_source_ids


def _validated_preview_call(
    call: Mapping[str, Any], *, target_id: str, target_version: int, assertion: str, limit: int
) -> str:
    """Require a complete preview bound to the exact target and history."""
    if (
        call.get("preview_complete") is not True
        or call.get("preview_target_fact_id") != target_id
        or call.get("preview_target_version") != target_version
        or _sha256(assertion) not in call.get("preview_historical_assertion_sha256", [])
        or call.get("request_arguments_present") is not True
        or call.get("request_fact_id") != target_id
        or call.get("request_expected_version") != target_version
        or call.get("request_limit") != limit
    ):
        raise RuntimeError("Forget preview was incomplete or lost the exact target history")
    digest = call.get("preview_digest")
    if not isinstance(digest, str) or len(digest) != 64:
        raise RuntimeError("Preview did not return a usable digest")
    return digest


def _start_service(fixture: Path, registry: Any, config: Any) -> tuple[Any, Any]:
    from mnemos.engine import MnemosEngine
    from mnemos.memory_service import LocalMemoryService
    from mnemos.service import MnemosService
    from mnemos.utils.embeddings import SimpleEmbeddingProvider
    from mnemos.utils.llm import UnavailableLLMProvider
    from mnemos.utils.storage import SQLiteStore

    class ConstantFixtureEmbeddings(SimpleEmbeddingProvider):
        """Transport fixture only; no semantic retrieval claim."""

        def embed(self, text: str) -> list[float]:
            del text
            return [1.0] + [0.0] * 63

        def embed_batch(self, texts: list[str]) -> list[list[float]]:
            return [self.embed(text) for text in texts]

    store = SQLiteStore(str(fixture / "memory.sqlite"))
    service: Any | None = None
    try:
        engine = MnemosEngine(
            config=config,
            llm=UnavailableLLMProvider("native lifecycle cognition unavailable"),
            embedder=ConstantFixtureEmbeddings(dim=64),
            store=store,
            scope_registry=registry,
            require_scope_context=True,
        )
        service = MnemosService(
            dispatcher=LocalMemoryService(engine, registry),
            descriptor_path=fixture / "runtime.json",
            lock_path=fixture / "service.lock",
            startup_timeout=5,
            dispatch_timeout=20,
        )
        service.start()
        return service, store
    except Exception:
        cleanup_errors = _close_service(service, store)
        if cleanup_errors:
            raise RuntimeError("Service startup cleanup failed")
        raise


def _close_service(service: Any | None, store: Any | None) -> list[str]:
    errors: list[str] = []
    service_stopped = True
    if service is not None:
        try:
            service.stop()
        except Exception as exc:
            service_stopped = False
            errors.append("service_stop:" + type(exc).__name__)
    # A store belongs to the service owner thread. Do not close it after a
    # failed stop, since that can race an owner still using SQLite.
    if store is not None and service_stopped:
        try:
            store.close()
        except Exception as exc:
            errors.append("store_close:" + type(exc).__name__)
    return errors

def _auth_copy(config_dir: Path) -> Path:
    source = Path(os.environ.get("USERPROFILE", str(Path.home()))) / ".codex" / "auth.json"
    if not source.is_file():
        raise FileNotFoundError("Existing Codex auth.json is unavailable")
    _protect_file(config_dir)
    target = config_dir / "auth.json"
    try:
        shutil.copy2(source, target)
        _protect_file(target)
    except Exception:
        target.unlink(missing_ok=True)
        raise
    return target


def _cleanup_native_resources(
    process: subprocess.Popen[str] | None, auth_copy: Path | None
) -> list[str]:
    """Attempt process and auth cleanup independently, preserving both errors."""
    errors: list[str] = []
    if process is not None and process.poll() is None:
        try:
            _terminate_process_tree(process)
        except Exception as exc:
            errors.append("process_terminate:" + type(exc).__name__)
        if process.poll() is None:
            errors.append("process_still_running")
    if auth_copy is not None:
        try:
            auth_copy.unlink(missing_ok=True)
        except OSError as exc:
            errors.append("auth_remove:" + type(exc).__name__)
        if auth_copy.exists():
            errors.append("auth_copy_remains")
    return errors


def _run_turn(
    *,
    fixture: Path,
    project: Path,
    turn_name: str,
    prompt: str,
    registry_path: Path,
    settings_path: Path,
    runtime_dir: Path,
    routing_dir: Path,
    profile: str,
    target_python: Path,
    expected_strings: Mapping[str, str] | None = None,
) -> dict[str, Any]:
    from mnemos.hosts import apply_host_integration

    turn_dir = fixture / ("turn-" + turn_name)
    _private_mkdir(turn_dir)
    home = turn_dir / "codex-home"
    _private_mkdir(home)
    session_id = "native-codex-lifecycle-" + turn_name + "-" + secrets.token_hex(8)
    selector_env = {
        "MNEMOS_CONFIG_PATH": str(settings_path),
        "MNEMOS_SCOPE_REGISTRY": str(registry_path),
        "MNEMOS_USER_PROFILE_ID": profile,
        "MNEMOS_RUNTIME_DESCRIPTOR": str(fixture / "runtime.json"),
        "MNEMOS_RUNTIME_DIR": str(runtime_dir),
        "MNEMOS_ROUTING_REVIEW_DIR": str(routing_dir),
        "MNEMOS_AUTO_START": "false",
        "MNEMOS_AUTO_RETRIEVE": "true",
        "MNEMOS_PROJECT_ROOT": str(project),
        "MNEMOS_HOST": "codex",
        "MNEMOS_SESSION_ID": session_id,
        "PATH": str(target_python.resolve().parent) + os.pathsep + os.environ.get("PATH", ""),
    }
    generated = apply_host_integration(
        "codex",
        mnemos_config_path=settings_path,
        cwd=project,
        home=home,
        # The host renderer deliberately accepts only selectors safe to write
        # into a config file. The complete selector environment is still
        # inherited by the native process and its MCP child.
        integration_env={
            key: selector_env[key]
            for key in (
                "MNEMOS_CONFIG_PATH",
                "MNEMOS_SCOPE_REGISTRY",
                "MNEMOS_USER_PROFILE_ID",
                "MNEMOS_RUNTIME_DESCRIPTOR",
            )
        },
    )
    config_path = generated.config_path
    if not config_path.is_file():
        raise RuntimeError("Generated Codex config was not written")
    codex_home = config_path.parent
    if codex_home != (home / ".codex").resolve():
        raise RuntimeError("Generated Codex config is outside the requested Codex home")
    generated_config = config_path.read_text(encoding="utf-8")
    hook_command_lines = [
        line for line in generated_config.splitlines() if "mnemos.adapters.codex" in line
    ]
    if len(hook_command_lines) < 6:
        raise RuntimeError("Generated Codex config lacks the complete Mnemos hook set")
    auth_copy: Path | None = None
    process: subprocess.Popen[str] | None = None
    started = time.monotonic()
    native_stdout = ""
    native_stderr = ""
    timed_out = False
    try:
        auth_copy = _auth_copy(config_path.parent)
        environment = _clean_environment({"CODEX_HOME": str(codex_home), **selector_env})
        command = _isolated_codex_command(project, prompt)
        process = subprocess.Popen(
            command,
            cwd=str(project),
            env=environment,
            stdin=subprocess.DEVNULL,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="utf-8",
            errors="replace",
            shell=False,
            creationflags=subprocess.CREATE_NEW_PROCESS_GROUP,
        )
        try:
            native_stdout, native_stderr = process.communicate(timeout=MAX_NATIVE_SECONDS)
        except subprocess.TimeoutExpired:
            timed_out = _terminate_process_tree(process)
            native_stdout, native_stderr = process.communicate(timeout=15)
        summary = _native_event_summary(native_stdout, expected_strings=expected_strings)
        stdout_path = turn_dir / "native-stdout.jsonl"
        stderr_path = turn_dir / "native-stderr.log"
        stdout_path.write_text(native_stdout, encoding="utf-8")
        stderr_path.write_text(native_stderr, encoding="utf-8")
        _protect_file(stdout_path)
        _protect_file(stderr_path)
        return {
            "turn": turn_name,
            "session_id_digest": _sha256(session_id),
            "project_root_digest": _sha256(str(project.resolve())),
            "status": "completed" if process.returncode == 0 and not timed_out else "failed",
            "returncode": process.returncode,
            "timed_out": timed_out,
            "elapsed_ms": round((time.monotonic() - started) * 1000, 1),
            "codex_home": str(codex_home),
            "config_path": str(config_path),
            "config_sha256": _sha256(config_path.read_bytes()),
            "generated_hook_source": {
                "generator": "mnemos.hosts.apply_host_integration",
                "adapter_module": "mnemos.adapters.codex",
                "command_count": len(hook_command_lines),
                "command_sha256": _sha256("\n".join(hook_command_lines)),
            },
            "agents_path": str(project / "AGENTS.md"),
            "agents_sha256": _sha256((project / "AGENTS.md").read_bytes()),
            "command": [
                item if item != prompt else "<private lifecycle prompt>" for item in command
            ],
            "stdout_path": str(stdout_path),
            "stdout_sha256": _sha256(native_stdout),
            "stdout_bytes": len(native_stdout.encode("utf-8")),
            "stderr_path": str(stderr_path),
            "stderr_sha256": _sha256(native_stderr),
            "stderr_bytes": len(native_stderr.encode("utf-8")),
            "native": summary,
        }
    finally:
        cleanup_errors = _cleanup_native_resources(process, auth_copy)
        if cleanup_errors:
            raise RuntimeError("Native cleanup failed: " + ",".join(cleanup_errors))

def _mcp_transport_preflight(
    fixture: Path, project: Path, registry: Any, content: str
) -> dict[str, Any]:
    """Initialize and call the generated MCP server before any native model turn."""
    from mcp import ClientSession, StdioServerParameters
    from mcp.client.stdio import stdio_client
    from mnemos.hosts import apply_host_integration

    version = importlib.metadata.version("mcp")
    registry_path = fixture / "scopes.json"
    registry.save(registry_path)
    settings_path = fixture / "settings.toml"
    settings_path.write_text("# private MCP preflight settings\n", encoding="utf-8")
    selectors = {
        "MNEMOS_CONFIG_PATH": str(settings_path),
        "MNEMOS_SCOPE_REGISTRY": str(registry_path),
        "MNEMOS_USER_PROFILE_ID": PROFILE,
        "MNEMOS_RUNTIME_DESCRIPTOR": str(fixture / "runtime.json"),
    }
    home = fixture / "mcp-client-home"
    _private_mkdir(home)
    generated = apply_host_integration(
        "codex", mnemos_config_path=settings_path, cwd=project, home=home, integration_env=selectors
    )
    config = tomllib.loads(generated.config_path.read_text(encoding="utf-8"))
    server = config["mcp_servers"]["mnemos"]
    environment = _clean_environment(
        {
            **selectors,
            **server.get("env", {}),
            "MNEMOS_PROJECT_ROOT": str(project),
            "MNEMOS_HOST": "codex",
            "MNEMOS_SESSION_ID": "mcp-preflight",
            "MNEMOS_AUTO_START": "false",
            "PATH": str(Path(sys.executable).resolve().parent)
            + os.pathsep
            + os.environ.get("PATH", ""),
        }
    )
    resolved = shutil.which(server["command"], path=environment["PATH"])
    if resolved is None or Path(resolved).resolve().parent != Path(sys.executable).resolve().parent:
        raise RuntimeError("Generated MCP command does not resolve to the verified installation")
    parameters = StdioServerParameters(
        command=resolved, args=server.get("args", []), env=environment, cwd=str(project)
    )
    stderr_path = fixture / "mcp-preflight.stderr.log"

    async def roundtrip() -> dict[str, Any]:
        with stderr_path.open("w", encoding="utf-8") as error_log:
            async with stdio_client(parameters, errlog=error_log) as (reader, writer):
                async with ClientSession(reader, writer) as session:
                    initialized = await session.initialize()
                    listed = await session.list_tools()
                    names = sorted(tool.name for tool in listed.tools)
                    required = {
                        "mnemos_store",
                        "mnemos_retrieve",
                        "mnemos_correct",
                        "mnemos_forget_preview",
                        "mnemos_forget_related",
                    }
                    if not required.issubset(names):
                        raise RuntimeError("Generated MCP server lacks required lifecycle tools")
                    result = await session.call_tool(
                        "mnemos_retrieve", {"query": content, "top_k": 5}
                    )
                    if result.isError:
                        raise RuntimeError("MCP preflight retrieve returned a tool error")
                    values = [
                        json.loads(item.text)
                        for item in result.content
                        if getattr(item, "type", None) == "text"
                    ]
                    if not any(
                        isinstance(value, dict)
                        and any(
                            isinstance(fact, dict) and fact.get("content") == content
                            for fact in value.get("facts", [])
                        )
                        for value in values
                    ):
                        raise RuntimeError("MCP preflight did not retrieve the scoped fixture fact")
                    return {
                        "status": "passed",
                        "mcp_version": version,
                        "generated_command": server["command"],
                        "resolved_command": resolved,
                        "config_sha256": _sha256(generated.config_path.read_bytes()),
                        "server_name": initialized.serverInfo.name,
                        "tool_names": names,
                        "scoped_fact_retrieved": True,
                        "stderr_path": str(stderr_path),
                    }

    async def bounded() -> dict[str, Any]:
        return await asyncio.wait_for(roundtrip(), timeout=10)

    started = time.monotonic()
    result = asyncio.run(bounded())
    result["elapsed_ms"] = round((time.monotonic() - started) * 1000, 1)
    result["stderr_sha256"] = _sha256(stderr_path.read_bytes())
    return result


def _offline_service_api_preflight() -> dict[str, Any]:
    """Exercise the installed service API in a disposable, no-model fixture."""
    from mnemos.config import MnemosConfig
    from mnemos.scopes import ScopeRegistry
    from mnemos.service import _protect_private_path

    fixture = Path(tempfile.mkdtemp(prefix="mnemos-native-codex-api-preflight-", dir=TEMP_PARENT))
    _protect_private_path(fixture)
    service: Any | None = None
    store: Any | None = None
    try:
        project = fixture / "project"
        project.mkdir()
        _protect_private_path(project)
        runtime_dir = fixture / "runtime"
        runtime_dir.mkdir()
        _protect_private_path(runtime_dir)
        registry = ScopeRegistry()
        identity = registry.register_project(project, user_profile_id=PROFILE, enabled=True)
        config = MnemosConfig()
        for name in ("surprisal", "mutable_rag", "affective", "sleep", "spreading"):
            setattr(config.modules, name, False)
        service, store = _start_service(fixture, registry, config)
        context = registry.resolve(
            project,
            user_profile_id=PROFILE,
            session_id="api-preflight-session",
            host="codex",
        )
        content = "Offline API preflight fact."
        captured_content = "Please remember: " + content
        remembered = service.submit(
            "remember",
            {
                "event_id": "api-preflight-remember",
                "context": context.model_dump(mode="json"),
                "content": content,
                "role": "user",
                "scope": "project",
            },
        )
        if remembered.get("status") != "committed":
            raise RuntimeError("Offline remember API preflight failed")
        captured = service.submit(
            "capture",
            {
                "event_id": "api-preflight-codex-capture",
                "context": context.model_dump(mode="json"),
                "occurred_at": datetime.now(timezone.utc).isoformat(),
                "role": "user",
                "content": captured_content,
                "source_kind": "codex_hook",
                "source_ref": "UserPromptSubmit:api-preflight",
            },
        )
        if captured.get("status") not in {"committed", "queued"}:
            raise RuntimeError("Offline automatic capture API preflight failed")
        captured_event_id = captured.get("event_id")
        if not isinstance(captured_event_id, str) or not captured_event_id:
            captured_job_id = captured.get("job_id")
            captured_job = (
                store.get_job(captured_job_id) if isinstance(captured_job_id, str) else None
            )
            captured_payload = captured_job.payload if captured_job is not None else None
            captured_event_id = (
                captured_payload.get("event_id") if isinstance(captured_payload, Mapping) else None
            )
        if not isinstance(captured_event_id, str) or not captured_event_id:
            raise RuntimeError("Offline capture did not expose a persisted source identity")
        maintained = service.submit("maintenance", {})
        if not isinstance(maintained, Mapping) or "maintenance" not in maintained:
            raise RuntimeError("Offline maintenance API preflight failed")
        retrieved = service.submit(
            "retrieve",
            {
                "query": content,
                "context": context.model_dump(mode="json"),
                "allowed_scopes": ["project", "global"],
                "top_k": 5,
                "max_context_tokens": 1200,
                "profile": "baseline",
            },
        )
        facts = retrieved.get("facts") if isinstance(retrieved, Mapping) else None
        if not isinstance(facts, list) or not any(
            isinstance(fact, Mapping) and fact.get("content") == content for fact in facts
        ):
            raise RuntimeError("Offline retrieve API preflight did not return the fact")
        stored_facts = _find_facts(store, content)
        explicit_facts = [
            fact for fact in stored_facts if fact.metadata.get("source") == "explicit_remember"
        ]
        captured_facts = [
            fact
            for fact in _find_facts(store, captured_content)
            if fact.metadata.get("source") == "sleep_consolidation"
        ]
        if len(explicit_facts) != 1 or len(captured_facts) != 1:
            raise RuntimeError("Offline preflight did not produce target and captured facts")
        stored_fact = explicit_facts[0]
        captured_fact = captured_facts[0]
        if (
            stored_fact.metadata.get("source") != "explicit_remember"
            or stored_fact.metadata.get("scope_id") != identity.id
            or stored_fact.metadata.get("user_profile_id") != PROFILE
        ):
            raise RuntimeError("Offline preflight fact provenance is incorrect")
        captured_evidence = store.list_fact_evidence(captured_fact.id)
        if not any(item.source_event_id == captured_event_id for item in captured_evidence):
            raise RuntimeError("Offline captured fact lacks Codex source evidence")
        mcp_handshake = _mcp_transport_preflight(fixture, project, registry, content)
        preview = service.submit(
            "forget_preview",
            {
                "context": context.model_dump(mode="json"),
                "fact_id": stored_fact.id,
                "expected_version": stored_fact.version,
                "limit": 100,
            },
        )
        if preview.get("status") != "preview" or preview.get("complete") is not True:
            raise RuntimeError("Offline related-forget preview API preflight failed")
        candidate = next(
            (
                item
                for item in preview.get("candidates", [])
                if item.get("fact_id") == captured_fact.id
            ),
            None,
        )
        if not isinstance(candidate, Mapping):
            raise RuntimeError("Offline preview omitted the captured fact")
        selected_source_ids = list(candidate.get("source_event_ids", []))
        applied = service.submit(
            "forget_related",
            {
                "context": context.model_dump(mode="json"),
                "fact_id": stored_fact.id,
                "expected_version": stored_fact.version,
                "preview_digest": preview["preview_digest"],
                "preview_limit": preview["limit"],
                "selected_fact_ids": [captured_fact.id],
                "selected_source_event_ids": selected_source_ids,
            },
        )
        if (
            applied.get("status") != "committed"
            or applied.get("forgotten") is not True
            or stored_fact.id not in applied.get("selected_fact_ids", [])
            or captured_fact.id not in applied.get("selected_fact_ids", [])
        ):
            raise RuntimeError("Offline related-forget apply API preflight failed")
        if store.get(stored_fact.id) is not None or store.get(captured_fact.id) is not None:
            raise RuntimeError("Offline related-forget did not erase selected facts")

def _verify_preflight_report(
    path: Path, *, wheel_path: Path, expected_wheel_sha256: str
) -> dict[str, Any]:
    """Bind an execution to a successful, matching, immutable preflight."""
    if not path.is_file():
        raise FileNotFoundError(f"Verification report is missing: {path}")
    try:
        value = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, UnicodeDecodeError, json.JSONDecodeError) as exc:
        raise RuntimeError("Verification report is not valid JSON") from exc
    if not isinstance(value, Mapping):
        raise RuntimeError("Verification report must be a JSON object")
    if value.get("status") != "preflight_passed":
        raise RuntimeError("Verification report is not a passed preflight")
    if value.get("runner_sha256") != _sha256(Path(__file__).read_bytes()):
        raise RuntimeError("Preflight used different runner bytes")
    api = value.get("service_api_preflight", {})
    if not isinstance(api, Mapping) or api.get("mcp_handshake", {}).get("status") != "passed":
        raise RuntimeError("Preflight lacks a passed generated MCP handshake")
    python_info = value.get("python", {})
    if (
        not isinstance(python_info, Mapping)
        or Path(str(python_info.get("executable"))).resolve() != Path(sys.executable).resolve()
    ):
        raise RuntimeError("Preflight used a different Python installation")
    if value.get("source_commit") != SOURCE_COMMIT:
        raise RuntimeError("Verification report source commit does not match")
    if value.get("expected_wheel_sha256") != expected_wheel_sha256:
        raise RuntimeError("Verification report wheel digest does not match")
    provenance = value.get("package_provenance")
    if not isinstance(provenance, Mapping):
        raise RuntimeError("Verification report lacks package provenance")
    if provenance.get("wheel_sha256_observed") != expected_wheel_sha256:
        raise RuntimeError("Verification report wheel artifact digest does not match")
    recorded_path = provenance.get("wheel_path")
    if not isinstance(recorded_path, str) or Path(recorded_path).resolve() != wheel_path.resolve():
        raise RuntimeError("Verification report wheel artifact path does not match")
    python_info = value.get("python")
    package_path = (
        Path(str(python_info.get("mnemos_package"))) if isinstance(python_info, Mapping) else None
    )
    if (
        not isinstance(python_info, Mapping)
        or python_info.get("site_packages_asserted") is not True
        or package_path is None
        or not package_path.is_file()
        or "site-packages" not in {part.lower() for part in package_path.parts}
    ):
        raise RuntimeError("Verification report lacks a valid installed package path")
    return {
        "report_path": str(path.resolve()),
        "report_sha256": _sha256(path.read_bytes()),
        "source_commit": SOURCE_COMMIT,
        "wheel_path": str(wheel_path.resolve()),
        "wheel_sha256": expected_wheel_sha256,
        "installed_package": str(package_path),
    }


def _verify_build_report(path: Path, wheel: Path, digest: str) -> dict[str, Any]:
    value = json.loads(path.read_text(encoding="utf-8"))
    if value.get("status") != "passed" or value.get("source_commit") != SOURCE_COMMIT:
        raise RuntimeError("The exact source must pass full verification before native preflight")
    artifacts = [
        item
        for item in value.get("artifacts", [])
        if Path(item["file"]).resolve() == wheel.resolve() and item.get("sha256") == digest
    ]
    if len(artifacts) != 1 or _sha256(wheel.read_bytes()) != digest:
        raise RuntimeError("Wheel is not the exact artifact of the passed full verification")
    return {
        "report_path": str(path.resolve()),
        "report_sha256": _sha256(path.read_bytes()),
        "source_commit": SOURCE_COMMIT,
        "wheel": artifacts[0],
    }

        ),

                "selected_source_event_ids": ["source-a"],

            related_result,

    )

                    "server": "mnemos",
                    "name": name,
                    "status": "completed",
                    "arguments": arguments,
                    "result": json.dumps(result),
                }
            }
        )
        for name, arguments, result in calls
    )
    summary = _native_event_summary(
        stdout,
        expected_strings={"old": "old assertion"},
    )
    preview_calls = _successful_calls(summary, "mnemos_forget_preview")
    related_calls = _successful_calls(summary, "mnemos_forget_related")
    if len(preview_calls) != 1 or len(related_calls) != 1:
        raise RuntimeError("Offline parser did not recognize successful MCP calls")
    preview_call = preview_calls[0]
    related_call = related_calls[0]
    if not _result_contains(summary, "mnemos_forget_preview", "old"):
        raise RuntimeError("Offline parser lost preview assertion evidence")
    _validated_preview_call(
        preview_calls[0],
        target_id="target",
        target_version=2,
        assertion="old assertion",
        limit=100,
    )
    incomplete = dict(preview_calls[0])
    incomplete["preview_complete"] = False
    try:
        _validated_preview_call(
            incomplete,
            target_id="target",
            target_version=2,
            assertion="old assertion",
            limit=100,
        )
    except RuntimeError:
        incomplete_preview_rejected = True
    else:
        raise RuntimeError("Offline preview gate accepted an incomplete response")
    mismatched_target = dict(preview_calls[0])
    mismatched_target["preview_target_fact_id"] = "other-target"
    try:
        _validated_preview_call(
            mismatched_target,
            target_id="target",
            target_version=2,
            assertion="old assertion",
            limit=100,
        )
    except RuntimeError:
        mismatched_target_rejected = True
    else:
        raise RuntimeError("Offline preview gate accepted a mismatched target")
    mismatched_history = dict(preview_calls[0])
    mismatched_history["preview_historical_assertion_sha256"] = [_sha256("other assertion")]
    try:
        _validated_preview_call(
            mismatched_history,
            target_id="target",
            target_version=2,
            assertion="old assertion",
            limit=100,
        )
    except RuntimeError:
        mismatched_history_rejected = True
    else:
        raise RuntimeError("Offline preview gate accepted mismatched history")
    _, _, selected_fact_ids, selected_source_ids = _validated_related_selection(
        preview_call, related_call, "target"
    )
    mismatched = dict(related_call)
    mismatched["selected_fact_ids"] = ["unreviewed-fact"]
    try:
        _validated_related_selection(preview_call, mismatched, "target")
    except RuntimeError:
        selection_mismatch_rejected = True
    else:
        raise RuntimeError("Offline selection gate accepted an unreviewed ID")
    if related_call.get("result_forgotten") is not True:
        raise RuntimeError("Offline parser lost the committed forget acknowledgement")
    failed = json.dumps(
        {
            "item": {
                "type": "mcp_tool_call",
                "server": "mnemos",
                "name": "mnemos_forget_related",

                "isError": True,

            }
        }
    )

    failed_call = _successful_calls(failed_summary, "mnemos_forget_related")
    if failed_call:
        raise RuntimeError("Offline parser accepted an outer MCP error")
    legacy_attempt = _native_event_summary(
        json.dumps(
            {
                "item": {
                    "type": "mcp_tool_call",
                    "server": "mnemos",
                    "name": "mnemos_forget",
                    "status": "failed",
                    "result": json.dumps({"status": "failed"}),
                }
            }
        )
    )
    if "mnemos_forget" not in legacy_attempt["tool_names"]:
        raise RuntimeError("Offline parser failed to observe a legacy forget attempt")
    cleanup_probe = Path(tempfile.mkdtemp(prefix="mnemos-lifecycle-v2-cleanup-"))
    auth_probe = cleanup_probe / "auth.json"

    try:

def _parse_args(argv: list[str] | None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--wheel", type=Path)
    parser.add_argument("--expected-wheel-sha256")
    parser.add_argument("--build-verification", type=Path)
    parser.add_argument(
        "--source-commit",
        required=False,
        help="Exact source revision named by the passed full verification report",
    )
    parser.add_argument(
        "--offline-check",
        action="store_true",
        help="Run parser and selection checks without package, host, or model access",
    )
    parser.add_argument("--preflight-only", action="store_true")
    parser.add_argument(
        "--verification",
        type=Path,
        help="Passed preflight report required for a native execution",
    )
    parser.add_argument(
        "--output",
        type=Path,
        help="New evidence directory (defaults to the versioned output directory)",
    )
    args = parser.parse_args(argv)
    if args.offline_check:
        return args
    if args.wheel is None or args.build_verification is None or args.source_commit is None:
        parser.error("--wheel, --build-verification, and --source-commit are required")
    if args.expected_wheel_sha256 is None:
        parser.error("--expected-wheel-sha256 is required")
    digest = args.expected_wheel_sha256.lower()
    if len(digest) != 64 or any(char not in "0123456789abcdef" for char in digest):
        parser.error("--expected-wheel-sha256 must be a 64-character hexadecimal digest")
    if len(args.source_commit) != 40 or any(
        char not in "0123456789abcdefABCDEF" for char in args.source_commit
    ):
        parser.error("--source-commit must be a 40-character hexadecimal revision")
    args.expected_wheel_sha256 = digest
    args.source_commit = args.source_commit.lower()
    return args
