def _wsl_path(path: Path) -> str:
    resolved = path.resolve()
    drive = resolved.drive.rstrip(":").lower()
    tail = str(resolved)[len(resolved.anchor) :].replace("\\", "/").lstrip("/")
    return "/mnt/" + drive + "/" + tail


def _sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def _normalized_bytes(data: bytes) -> bytes:
    try:
        return data.decode("utf-8").replace("\r\n", "\n").replace("\r", "\n").encode("utf-8")
    except UnicodeDecodeError:
        return data


def _source_hashes(snapshot: Path, commit: str) -> dict[str, str]:
    listing = subprocess.run(
        ["git", "-C", str(snapshot), "ls-tree", "-r", "--name-only", commit, "mnemos"],
        capture_output=True,
        check=True,
        text=True,
    )
    paths = [item.strip() for item in listing.stdout.splitlines() if item.strip()]
    if len(paths) != 53:
        raise SystemExit(f"expected 53 tracked mnemos files, found {len(paths)}")
    hashes: dict[str, str] = {}
    for relative in paths:
        data = subprocess.run(
            ["git", "-C", str(snapshot), "show", f"{commit}:{relative}"],
            capture_output=True,
            check=True,
        ).stdout
        hashes[relative.replace("\\", "/")] = hashlib.sha256(_normalized_bytes(data)).hexdigest()
    return hashes


def _asset_hashes(snapshot: Path, commit: str) -> dict[str, str]:
    listing = subprocess.run(
        ["git", "-C", str(snapshot), "ls-tree", "-r", "--name-only", commit, "agents"],
        capture_output=True,
        check=True,
        text=True,
    )
    paths = [item.strip() for item in listing.stdout.splitlines() if item.strip()]
    if len(paths) != 3:
        raise SystemExit(f"expected 3 tracked agent assets, found {len(paths)}")
    hashes: dict[str, str] = {}
    for relative in paths:
        data = subprocess.run(
            ["git", "-C", str(snapshot), "show", f"{commit}:{relative}"],
            capture_output=True,
            check=True,
        ).stdout
        target = "mnemos/resources/" + relative.replace("\\", "/")
        checkout_data = (snapshot / relative).read_bytes()
        if _normalized_bytes(checkout_data) != _normalized_bytes(data):
            raise SystemExit(f"asset checkout differs from source commit: {relative}")
        # The wheel contains checkout bytes, including Windows line endings.
        hashes[target] = hashlib.sha256(checkout_data).hexdigest()
    return hashes


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--wheel", required=True, type=Path)
    parser.add_argument("--wheel-sha256", required=True)
    parser.add_argument("--verification", required=True, type=Path)
    parser.add_argument("--output", required=True, type=Path)
    args = parser.parse_args()
    wheel = args.wheel.resolve()
    verification_path = args.verification.resolve()
    output = args.output.resolve()
    if output.exists():
        raise SystemExit(f"refusing to overwrite {output}")
    verification = json.loads(verification_path.read_text(encoding="utf-8"))
    if verification.get("status") != "passed":
        raise SystemExit("verification report is not passed")
    if verification.get("source_commit") != EXPECTED_SOURCE:
        raise SystemExit("verification report is not pinned to the repaired source")
    source_checkout = Path(verification.get("source_checkout", "")).resolve()
    if not source_checkout.is_dir():
        raise SystemExit("verification report source checkout is missing")
    source_head = subprocess.run(
        ["git", "-C", str(source_checkout), "rev-parse", "HEAD"],
        capture_output=True,
        text=True,
        check=True,
    ).stdout.strip()
    if source_head != EXPECTED_SOURCE:
        raise SystemExit("verification source checkout is not at the expected commit")
    if subprocess.run(
        ["git", "-C", str(source_checkout), "status", "--porcelain", "--untracked-files=all"],
        capture_output=True,
        text=True,
        check=True,
    ).stdout.strip():
        raise SystemExit("verification source checkout is dirty")
    source_hashes = _source_hashes(source_checkout, EXPECTED_SOURCE)
    asset_hashes = _asset_hashes(source_checkout, EXPECTED_SOURCE)
    expected_adapter_sha256 = source_hashes["mnemos/adapters/hermes.py"]
    pyproject_path = source_checkout / "pyproject.toml"
    try:
        project_name = str(
            tomllib.loads(pyproject_path.read_text(encoding="utf-8"))["project"]["name"]
        ).strip()
    except (OSError, KeyError, TypeError, tomllib.TOMLDecodeError) as exc:
        raise SystemExit("verified source pyproject has no readable project name") from exc
    if not project_name:
        raise SystemExit("verified source pyproject has an empty project name")
    dist_info_prefix = re.sub(r"[-_.]+", "_", project_name)
    artifacts = verification.get("artifacts")
    if not isinstance(artifacts, list):
        raise SystemExit("full verification report has no artifact records")
    matching_artifacts = []
    for artifact in artifacts:
        if not isinstance(artifact, dict) or not artifact.get("file"):
            continue
        artifact_path = Path(str(artifact["file"]))
        if not artifact_path.is_absolute():
            artifact_path = (verification_path.parent / artifact_path).resolve()
        if artifact_path == wheel:
            matching_artifacts.append(artifact)
    if len(matching_artifacts) != 1:
        raise SystemExit("full verification report does not identify this wheel path exactly once")
    recorded_wheel_sha256 = str(matching_artifacts[0].get("sha256", "")).lower()
    if recorded_wheel_sha256 != args.wheel_sha256.lower():
        raise SystemExit("wheel hash differs from the full verification artifact record")
    if _sha256(wheel).lower() != args.wheel_sha256.lower():
        raise SystemExit("wheel hash does not match --wheel-sha256")
    output.mkdir(parents=True)
    payload = (
        PAYLOAD.replace("__WHEEL__", json.dumps(_wsl_path(wheel)))
        .replace("__PROJECT_NAME__", repr(project_name))
        .replace("__DIST_INFO_PREFIX__", repr(dist_info_prefix))
        .replace("__WHEEL_SHA256__", repr(args.wheel_sha256))
        .replace("__SOURCE_ADAPTER_SHA256__", repr(expected_adapter_sha256))
        .replace("__SOURCE_HASHES__", repr(source_hashes))
        .replace("__ASSET_HASHES__", repr(asset_hashes))
    )
    (output / "payload.py").write_text(payload, encoding="utf-8")
    started = datetime.now(timezone.utc)
    command = [
        "wsl.exe",
        "-d",
        "Debian",
        "--",
        "env",
        "-i",
        "PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin",
        "HOME=/tmp/mnemos-hermes-wheel-runner",
        WSL_PYTHON,
        "-B",
        "-c",
        payload,
    ]
    try:
        result = subprocess.run(command, capture_output=True, text=True, timeout=30, check=False)
    except subprocess.TimeoutExpired as exc:
        finished = datetime.now(timezone.utc)
        stdout = (
            exc.stdout.decode(errors="replace")
            if isinstance(exc.stdout, bytes)
            else (exc.stdout or "")
        )
        stderr = (
            exc.stderr.decode(errors="replace")
            if isinstance(exc.stderr, bytes)
            else (exc.stderr or "")
        )
        timeout_report = {
            "status": "failed",
            "wheel": str(wheel),
            "wheel_sha256": args.wheel_sha256,
            "verification": str(verification_path),
            "verification_commit": EXPECTED_SOURCE,
            "source_checkout": str(source_checkout),
            "source_files_checked": len(source_hashes),
            "packaged_assets_checked": len(asset_hashes),
            "runner": str(Path(__file__).resolve()),
            "runner_sha256": _sha256(Path(__file__).resolve()),
            "started_utc": started.isoformat(),
            "finished_utc": finished.isoformat(),
            "command": [*command[:-1], "<payload.py contents>"],
            "payload_file": "payload.py",
            "payload_sha256": _sha256(output / "payload.py"),
            "returncode": None,
            "timed_out": True,
            "stdout_log": "stdout.txt",
            "stderr_log": "stderr.txt",
            "child": None,
        }
        (output / "stdout.txt").write_text(stdout, encoding="utf-8")
        (output / "stderr.txt").write_text(stderr, encoding="utf-8")
        (output / "report.json").write_text(
            json.dumps(timeout_report, indent=2, sort_keys=True) + "\n", encoding="utf-8"
        )
        raise
    finished = datetime.now(timezone.utc)
    lines = [line for line in result.stdout.splitlines() if line.strip()]
    child = json.loads(lines[-1]) if lines and result.returncode == 0 else None
    report = {
        "status": "passed" if result.returncode == 0 and child else "failed",
        "wheel": str(wheel),
        "wheel_sha256": args.wheel_sha256,
        "verification": str(verification_path),
        "verification_commit": EXPECTED_SOURCE,
        "source_checkout": str(source_checkout),
        "source_files_checked": len(source_hashes),
        "packaged_assets_checked": len(asset_hashes),
        "verification_artifact": matching_artifacts[0],
        "runner": str(Path(__file__).resolve()),
        "runner_sha256": _sha256(Path(__file__).resolve()),
        "started_utc": started.isoformat(),
        "finished_utc": finished.isoformat(),
        "command": [*command[:-1], "<payload.py contents>"],
        "payload_file": "payload.py",
        "payload_sha256": _sha256(output / "payload.py"),
        "returncode": result.returncode,
        "stdout_log": "stdout.txt",
        "stderr_log": "stderr.txt",
        "child": child,
    }
    (output / "stdout.txt").write_text(result.stdout, encoding="utf-8")
    (output / "stderr.txt").write_text(result.stderr, encoding="utf-8")
    (output / "report.json").write_text(
        json.dumps(report, indent=2, sort_keys=True) + "\n", encoding="utf-8"
    )
    print(json.dumps(report, indent=2, sort_keys=True))
    return 0 if report["status"] == "passed" else 1


if __name__ == "__main__":
    raise SystemExit(main())
