from pathlib import Path
from collections import Counter
import hashlib
import json
import subprocess

root = Path(__file__).resolve().parents[1]
path = root / 'docs/validation/2026-09-16/real-model-v3/lifecycle/baseline.json'
raw = path.read_bytes()
report = json.loads(raw)
assert report['status'] == 'completed'
counts = Counter()
scope = Counter()
trace_counts = Counter()
samples = []
for row in report['raw_outputs']:
    data = row['raw_output']
    for judgment in data['extraction_judgements']:
        if judgment.get('accepted'):
            counts['accepted_exact_text_evidence'] += 1
        elif not judgment.get('matched_supported_fact_id'):
            counts['no_exact_supported_text_and_scope_match'] += 1
        else:
            counts['matched_text_but_evidence_check_failed'] += 1
    for operation in data['operations']:
        if operation.get('operation') == 'retrieve':
            validation = operation.get('scope_validation', {})
            for key in ('foreign_count', 'missing_metadata_count', 'malformed_count', 'verified_count'):
                scope[key] += validation.get(key, 0)
    trace = data.get('provider_trace', {})
    if isinstance(trace, dict):
        for key, value in trace.items():
            if isinstance(value, (int, float)) and not isinstance(value, bool):
                trace_counts[key] += value
    if len(samples) < 2:
        samples.append({'session_id': row['session_id'], 'measurement_notes': data.get('measurement_notes'), 'provider_trace_keys': sorted(trace) if isinstance(trace, dict) else type(trace).__name__})
source = 'b7a9d9d4d9d6947564696457b0e2754f7e95a801'
source_hashes = {}
for name in ['benchmarks/local_memory/ollama_adapter.py', 'benchmarks/local_memory/harness.py', 'mnemos/retrieval.py']:
    blob = subprocess.check_output(['git', '-C', str(root), 'show', f'{source}:{name}'])
    source_hashes[name] = hashlib.sha256(blob).hexdigest()
output = {
    'status': 'completed_baseline_interim_diagnostic',
    'source_commit': source,
    'baseline_report_sha256_at_observation': hashlib.sha256(raw).hexdigest(),
    'report_may_gain_paired_comparisons': True,
    'baseline_raw_outputs_sha256': hashlib.sha256(path.with_name('baseline.raw-outputs.jsonl').read_bytes()).hexdigest(),
    'source_blob_sha256': source_hashes,
    'dataset': report['dataset'],
    'sample_counts': report['sample_counts'],
    'gates': report['gates'],
    'measurements': report['measurements'],
    'lifecycle_operations': report['lifecycle_operations'],
    'precision_proxy_breakdown': dict(counts),
    'retrieval_scope_validation_totals': dict(scope),
    'provider_trace_numeric_totals': dict(trace_counts),
    'samples': samples,
    'limitations': [
        'Full and leave-one-module-out profiles remain unfinished.',
        'Baseline has no enabled biomimetic modules; its model-call verification gate is not met.',
        'Supported precision is an exact frozen text and evidence proxy, not adjudicated semantic factuality.',
        'No-answer correctness requires an empty retrieved facts list; it does not measure a downstream assistant answer.',
        'This preserves failed gates and does not tune source, thresholds, or labels against heldout outputs.'
    ]
}
destination = root / '.venv/baseline-diagnosis-v1.json'
assert not destination.exists()
destination.write_text(json.dumps(output, indent=2) + '\n')
print(json.dumps({key: output[key] for key in ['precision_proxy_breakdown', 'retrieval_scope_validation_totals', 'provider_trace_numeric_totals', 'samples']}, indent=2))
