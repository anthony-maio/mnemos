This runner is an experiment harness for one disposable project.  It checks
the installed wheel, exercises the native Claude command with project-only
settings and persistent transcripts, and keeps full hook payloads private to
the fixture.  The fixture uses constant vectors and unavailable cognition,
so the result is transport and durability evidence rather than a model
quality evaluation.

def _sha256(value: bytes | str) -> str:
    raw = value.encode("utf-8") if isinstance(value, str) else value
    return hashlib.sha256(raw).hexdigest()


def _private_mkdir(path: Path) -> None:
    from mnemos.service import _protect_private_path

    path.mkdir(parents=True, exist_ok=False)
    _protect_private_path(path)


def _clean_environment(overrides: Mapping[str, str]) -> dict[str, str]:
    environment = dict(os.environ)
    isolated_auth_keys = {
        "ANTHROPIC_API_KEY",
        "ANTHROPIC_AUTH_TOKEN",
        "CLAUDE_CODE_OAUTH_TOKEN",
        "AWS_PROFILE",
        "AWS_ACCESS_KEY_ID",
        "AWS_SECRET_ACCESS_KEY",
        "AWS_SESSION_TOKEN",
        "GOOGLE_APPLICATION_CREDENTIALS",
    }
    for key in list(environment):
        if (
            key.startswith("MNEMOS_")
            or key
            in {
                "PYTHONPATH",
                "CLAUDE_CONFIG_DIR",
                "CLAUDE_SECURESTORAGE_CONFIG_DIR",
                "ANTHROPIC_CONFIG_DIR",
            }
            or key in isolated_auth_keys
        ):
            environment.pop(key, None)
    environment.update(overrides)
    return environment


def _terminate_process_tree(process: subprocess.Popen[str]) -> bool:
    """Terminate only the native child and descendants, then await it."""
    if process.poll() is not None:
        return False
    if os.name == "nt":
        subprocess.run(
            ["taskkill", "/PID", str(process.pid), "/T", "/F"],
            capture_output=True,
            text=True,
            check=False,
            timeout=10,
        )
    else:  # pragma: no cover - native runner is Windows-only
        process.kill()
    try:
        process.wait(timeout=10)
    except subprocess.TimeoutExpired:
        process.kill()
        process.wait(timeout=10)
    return True


def _pinned_package_provenance(
    package_path: Path, *, wheel_path: Path, expected_wheel_sha256: str
) -> dict[str, Any]:
    """Verify every tracked package file and every package member in the wheel."""
    listing = subprocess.run(
        [
            "git",
            "-C",
            str(WORKTREE),
            "ls-tree",
            "-r",
            "--name-only",
            SOURCE_COMMIT,
            "mnemos",
        ],
        capture_output=True,
        text=True,
        check=True,
        timeout=10,
    )
    files = [Path(item) for item in listing.stdout.splitlines() if item.strip()]
    mismatches: list[str] = []
    checked = 0
    for relative in files:
        expected = subprocess.run(
            ["git", "-C", str(WORKTREE), "show", f"{SOURCE_COMMIT}:{relative.as_posix()}"],
            capture_output=True,
            check=True,
            timeout=10,
        ).stdout.replace(b"\r\n", b"\n")
        installed_path = package_path.parent / Path(*relative.parts[1:])
        installed = installed_path.read_bytes().replace(b"\r\n", b"\n")
        checked += 1
        if installed != expected:
            mismatches.append(relative.as_posix())
    if not files:
        raise RuntimeError("pinned source has no tracked mnemos package files")
    if mismatches:
        raise RuntimeError("installed package differs from pinned source: " + ", ".join(mismatches))
    wheel_digest = _sha256(wheel_path.read_bytes())
    if wheel_digest != expected_wheel_sha256.lower():
        raise RuntimeError("wheel archive does not match the expected digest")
    wheel_members = 0
    with zipfile.ZipFile(wheel_path) as archive:
        for item in archive.infolist():
            if item.is_dir() or not item.filename.startswith("mnemos/"):
                continue
            wheel_members += 1
            installed_path = package_path.parent / Path(*Path(item.filename).parts[1:])
            if installed_path.read_bytes() != archive.read(item):
                raise RuntimeError("installed package bytes differ from wheel")
    return {
        "source_commit": SOURCE_COMMIT,
        "package_files_checked": checked,
        "wheel_package_members_checked": wheel_members,
        "wheel_path": str(wheel_path),
        "wheel_sha256_expected": expected_wheel_sha256.lower(),
        "wheel_sha256_observed": wheel_digest,
    }


def _run_claude(args: list[str], *, timeout: float = 10) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [str(CLAUDE_EXE), *args],
        capture_output=True,
        text=True,
        encoding="utf-8",
        errors="replace",
        check=False,
        timeout=timeout,
        shell=False,
    )


def _verify_claude() -> dict[str, Any]:
    if os.name != "nt":
        raise RuntimeError("native Claude runner requires Windows")
    if not CLAUDE_EXE.is_file():
        raise FileNotFoundError(f"direct Claude executable is missing: {CLAUDE_EXE}")
    result = _run_claude(["--version"])
    if result.returncode != 0:
        raise RuntimeError("Claude version probe failed")
    version = (result.stdout or result.stderr).strip()
    if "2.1.191" not in version:
        raise RuntimeError(f"unexpected Claude version: {version}")
    return {"path": str(CLAUDE_EXE), "version": version}


def _numeric_mapping(value: Any) -> dict[str, int | float]:
    if not isinstance(value, Mapping):
        return {}
    return {
        str(key): item
        for key, item in value.items()
        if isinstance(item, (int, float)) and not isinstance(item, bool)
    }


def _native_stream_summary(stdout: str, expected_canary: str) -> dict[str, Any]:
    rows: list[dict[str, Any]] = []
    hook_rows: list[dict[str, Any]] = []
    costs: list[float] = []
    answer_digests: list[str] = []
    answer_matches = False
    tool_types: set[str] = set()
    non_json = 0

    def inspect_tool_blocks(value: Any) -> None:
        if isinstance(value, Mapping):
            kind = value.get("type")
            if isinstance(kind, str) and (
                "tool" in kind or kind in {"bash", "computer", "text_editor"}
            ):
                tool_types.add(kind)
            for child in value.values():
                inspect_tool_blocks(child)
        elif isinstance(value, list):
            for child in value:
                inspect_tool_blocks(child)

    for line in stdout.splitlines():
        try:
            value = json.loads(line)
        except json.JSONDecodeError:
            non_json += 1
            continue
        if not isinstance(value, Mapping):
            non_json += 1
            continue
        inspect_tool_blocks(value)
        item = value.get("item")
        item_type = item.get("type") if isinstance(item, Mapping) else None
        if isinstance(item_type, str) and item_type in {
            "tool_use",
            "tool_result",
            "mcp_tool_use",
            "mcp_tool_result",
            "server_tool_use",
            "server_tool_result",
            "web_search_tool_result",
            "bash",
            "computer",
            "text_editor",
        }:
            tool_types.add(item_type)
        answer = item.get("text") if isinstance(item, Mapping) else None
        if not isinstance(answer, str) and value.get("type") == "result":
            answer = value.get("result")
        if isinstance(answer, str) and (
            item_type == "agent_message" or value.get("type") == "result"
        ):
            answer_digests.append(_sha256(answer))
            answer_matches |= (
                re.search(
                    r"(?<![A-Za-z0-9-])" + re.escape(expected_canary) + r"(?![A-Za-z0-9-])",
                    answer,
                )
                is not None
            )
        numeric: dict[str, int | float] = {}
        for key in ("total_cost_usd", "cost_usd"):
            raw = value.get(key)
            if isinstance(raw, (int, float)) and not isinstance(raw, bool):
                numeric[key] = raw
                if key == "total_cost_usd":
                    costs.append(float(raw))
        usage = _numeric_mapping(value.get("usage"))
        if str(value.get("subtype", "")).startswith("hook_") or value.get("hook_event"):
            hook_rows.append(
                {
                    "subtype": value.get("subtype"),
                    "hook_event": value.get("hook_event"),
                    "hook_name": value.get("hook_name"),
                    "outcome": value.get("outcome"),
                    "exit_code": value.get("exit_code"),
                    "stdout_sha256": _sha256(str(value.get("stdout", ""))),
                    "stderr_sha256": _sha256(str(value.get("stderr", ""))),
                    "output_sha256": _sha256(str(value.get("output", ""))),
                    "session_id_sha256": (
                        _sha256(value["session_id"])
                        if isinstance(value.get("session_id"), str)
                        else None
                    ),
                }
            )
        rows.append(
            {
                "type": value.get("type"),
                "subtype": value.get("subtype"),
                "item_type": item_type,
                "usage": usage,
                "numeric_cost_fields": numeric,
                "answer_digest": _sha256(answer) if isinstance(answer, str) else None,
            }
        )
    return {
        "record_count": len(rows),
        "non_json_line_count": non_json,
        "rows": rows,
        "hook_rows": hook_rows,
        "answer_text_digests": answer_digests,
        "answer_mentions_random_canary": answer_matches,
        "tool_call_observed": bool(tool_types),
        "tool_item_types": sorted(tool_types),
        "cost_usd_observed": costs,
        "total_cost_usd": sum(costs) if costs else None,
        "usage_observed": any(bool(row["usage"]) for row in rows),
    }


def _hook_record_summary(path: Path) -> dict[str, Any]:
    record = json.loads(path.read_text(encoding="utf-8"))
    payload = record.get("payload") if isinstance(record, Mapping) else None
    if not isinstance(payload, Mapping):
        payload = {}
    transcript = payload.get("transcript_path")
    transcript_state = "absent"
    if isinstance(transcript, str) and transcript.strip():
        # The native transcript belongs to the host.  Do not inspect or copy
        # files outside the disposable fixture; adapter status is retained in
        # the private trace if its own read attempt fails.
        transcript_state = "path_provided_uninspected"
    adapter_stdout = record.get("stdout") if isinstance(record, Mapping) else ""
    adapter_stderr = record.get("stderr") if isinstance(record, Mapping) else ""
    diagnostic_status = None
    diagnostic_match = re.search(r"status=([A-Za-z_]+)", str(adapter_stderr))
    if diagnostic_match:
        diagnostic_status = diagnostic_match.group(1)
    output: Any = None
    try:
        output = json.loads(adapter_stdout) if str(adapter_stdout).strip() else None
    except json.JSONDecodeError:
        output = None
    hook_output = output.get("hookSpecificOutput") if isinstance(output, Mapping) else None
    context = hook_output.get("additionalContext") if isinstance(hook_output, Mapping) else None
    session_id = payload.get("session_id")
    prompt_id = payload.get("prompt_id")
    return {
        "trace_file": str(path),
        "event": record.get("event"),
        "payload_sha256": _sha256(json.dumps(payload, sort_keys=True, separators=(",", ":"))),
        "returncode": record.get("returncode"),
        "stdout_sha256": _sha256(str(adapter_stdout)),
        "stderr_sha256": _sha256(str(adapter_stderr)),
        "diagnostic_status": diagnostic_status,
        "native_payload_fields": sorted(str(key) for key in payload),
        "session_id_sha256": _sha256(session_id) if isinstance(session_id, str) else None,
        "prompt_id_sha256": _sha256(prompt_id) if isinstance(prompt_id, str) else None,
        "cwd_present": isinstance(payload.get("cwd"), str),
        "prompt_present": isinstance(payload.get("prompt"), str),
        "last_assistant_message_present": isinstance(payload.get("last_assistant_message"), str),
        "transcript_state": transcript_state,
        "native_output_context_present": isinstance(context, str) and bool(context),
        "adapter_output_sha256": _sha256(str(adapter_stdout)),
    }


def _copy_private_auth(config_dir: Path) -> Path | None:
    """Copy only the native OAuth file, without reading it in the runner."""
    source = Path.home() / ".claude" / ".credentials.json"
    if not source.is_file():
        return None
    target = config_dir / ".credentials.json"
    from mnemos.service import _protect_private_path

    _protect_private_path(config_dir)
    try:
        shutil.copy2(source, target)
        _protect_private_path(target)
    except Exception:
        target.unlink(missing_ok=True)
        raise
    return target


def _validate_transcript(path: Path, config_dir: Path) -> dict[str, Any]:
    """Validate native JSONL only after proving it is inside the private root."""
    resolved = path.expanduser().resolve(strict=False)
    root = config_dir.resolve(strict=True)
    try:
        resolved.relative_to(root)
    except ValueError as exc:
        raise RuntimeError("native transcript path escaped private CLAUDE_CONFIG_DIR") from exc
    raw = resolved.read_bytes()
    if not raw:
        raise RuntimeError("native transcript is empty")
    roles: list[str] = []
    source_ref_digests: list[str] = []
    line_count = 0
    for line in raw.splitlines():
        line_count += 1
        record = json.loads(line.decode("utf-8"))
        if not isinstance(record, Mapping):
            continue
        message = record.get("message")
        role = message.get("role") if isinstance(message, Mapping) else record.get("role")
        if not isinstance(role, str) and record.get("type") in {"user", "assistant"}:
            role = record["type"]
        if isinstance(role, str):
            roles.append(role)
        source_ref = record.get("uuid") or record.get("id")
        if isinstance(source_ref, str) and source_ref:
            source_ref_digests.append(_sha256(source_ref))
    if not {"user", "assistant"}.issubset(set(roles)):
        raise RuntimeError("native transcript lacks both user and assistant JSONL roles")
    return {
        "path": str(resolved),
        "sha256": _sha256(raw),
        "bytes": len(raw),
        "line_count": line_count,
        "roles": roles,
        "source_ref_sha256": source_ref_digests,
    }


def _validate_checkpoint(checkpoints: Path, transcript: Mapping[str, Any]) -> dict[str, Any]:
    files = sorted(checkpoints.glob("*.json"))
    if not files:
        raise RuntimeError("Claude transcript checkpoint was not written")
    target = files[0]
    checkpoint = json.loads(target.read_text(encoding="utf-8"))
    if not isinstance(checkpoint, Mapping):
        raise RuntimeError("Claude checkpoint is not an object")
    offset = checkpoint.get("offset")
    if type(offset) is not int or offset < 0 or offset > int(transcript["bytes"]):
        raise RuntimeError("Claude checkpoint offset is outside the transcript")
    if checkpoint.get("path") != transcript["path"]:
        raise RuntimeError("Claude checkpoint path does not match native transcript")
    if offset and Path(transcript["path"]).read_bytes()[offset - 1 : offset] != b"\n":
        raise RuntimeError("Claude checkpoint offset is not a complete JSONL boundary")
    scope = checkpoint.get("scope")
    if not isinstance(scope, Mapping) or scope.get("host") != "claude-code":
        raise RuntimeError("Claude checkpoint scope is missing or mismatched")
    if any(
        not isinstance(scope.get(key), str) or not scope.get(key)
        for key in (
            "user_profile_id",
            "project_id",
            "session_id",
        )
    ):
        raise RuntimeError("Claude checkpoint scope is incomplete")
    if checkpoint.get("identity_version") != 2:
        raise RuntimeError("Claude checkpoint identity version is not v2")
    return {
        "path": str(target),
        "sha256": _sha256(target.read_bytes()),
        "offset": offset,
        "transcript_bytes": transcript["bytes"],
        "scope": dict(scope),
        "identity_version": checkpoint.get("identity_version"),
        "pending_dedup_count": (
            len(checkpoint.get("pending_dedup", []))
            if isinstance(checkpoint.get("pending_dedup"), list)
            else 0
        ),
    }


def _start_service(fixture: Path, registry: Any, config: Any) -> tuple[Any, Any, Any]:
    from mnemos.engine import MnemosEngine
    from mnemos.memory_service import LocalMemoryService
    from mnemos.service import MnemosService
    from mnemos.utils.embeddings import SimpleEmbeddingProvider
    from mnemos.utils.llm import UnavailableLLMProvider
    from mnemos.utils.storage import SQLiteStore

    class ConstantFixtureEmbeddings(SimpleEmbeddingProvider):
        """Offline transport fixture: every text maps to the same unit vector."""

        def embed(self, text: str) -> list[float]:
            del text
            return [1.0] + [0.0] * 63

        def embed_batch(self, texts: list[str]) -> list[list[float]]:
            return [self.embed(text) for text in texts]

    store = SQLiteStore(str(fixture / "memory.sqlite"))
    engine = MnemosEngine(
        config=config,
        llm=UnavailableLLMProvider("native Claude scope smoke cognition unavailable"),
        embedder=ConstantFixtureEmbeddings(dim=64),
        store=store,
        scope_registry=registry,
        require_scope_context=True,
    )
    service = MnemosService(
        dispatcher=LocalMemoryService(engine, registry),
        descriptor_path=fixture / "runtime.json",
        lock_path=fixture / "service.lock",
        startup_timeout=5,
        dispatch_timeout=20,
    )
    service.start()
    context = registry.resolve(
        fixture / "project",
        user_profile_id=PROFILE,
        session_id="native-claude-persistent-seed-session",
        host="claude-code",
    )
    return service, store, context

def _parse_args(argv: list[str] | None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Prepare a native Claude scoped-memory smoke.")
    parser.add_argument("--wheel", type=Path, required=True, help="installed wheel archive")
    parser.add_argument("--expected-wheel-sha256", required=True, help="wheel SHA256 digest")
    args = parser.parse_args(argv)
    digest = args.expected_wheel_sha256.lower()
    if len(digest) != 64 or any(char not in "0123456789abcdef" for char in digest):
        parser.error("--expected-wheel-sha256 must be a 64-character hexadecimal digest")
    args.expected_wheel_sha256 = digest
    return args


def main(argv: list[str] | None = None) -> int:
    args = _parse_args(argv)
    if OUTPUT_DIR.exists():
        raise FileExistsError(f"refusing to overwrite evidence directory: {OUTPUT_DIR}")
    if not TEMP_PARENT.is_dir():
        raise FileNotFoundError(f"required private fixture parent is missing: {TEMP_PARENT}")
    OUTPUT_DIR.mkdir(parents=True)
    report: dict[str, Any] = {
        "runner": "native-claude-persistent-v5",
        "source_commit": SOURCE_COMMIT,
        "runner_sha256": _sha256(Path(__file__).read_bytes()),
        "started_at": datetime.now(timezone.utc).isoformat(),
        "status": "not_observed",
        "scope": "native Claude hooks with persistent transcript, scoped seed, durable restart, replay; no cognition quality claim",
        "model_quality_claim": "none",
        "embedding_fixture": "Constant unit vectors for every text; no semantic retrieval claim",
        "cognition": "UnavailableLLMProvider; all modules disabled",
        "max_native_seconds": MAX_NATIVE_SECONDS,
        "hook_timeout_seconds": HOOK_TIMEOUT_SECONDS,
        "wheel_argument": str(args.wheel),
        "expected_wheel_sha256": args.expected_wheel_sha256,
    }
    fixture = Path(tempfile.mkdtemp(prefix="mnemos-native-claude-persistent-v5-", dir=TEMP_PARENT))
    service: Any | None = None
    restart_service: Any | None = None
    process: subprocess.Popen[str] | None = None
    auth_copy: Path | None = None
    try:
        from mnemos import __file__ as mnemos_file
        from mnemos.config import MnemosConfig
        from mnemos.scopes import ScopeRegistry
        from mnemos.service import _protect_private_path

        _protect_private_path(fixture)
        package_path = Path(mnemos_file).resolve()
        if "site-packages" not in {part.lower() for part in package_path.parts}:
            raise RuntimeError(f"installed-wheel isolation failed: imported {package_path}")
        report["python"] = {
            "executable": str(Path(sys.executable).resolve()),
            "mnemos_package": str(package_path),
            "site_packages_asserted": True,
        }
        report["package_provenance"] = _pinned_package_provenance(
            package_path,
            wheel_path=args.wheel,
            expected_wheel_sha256=args.expected_wheel_sha256,
        )
        report["claude"] = _verify_claude()

        project = fixture / "project"
        trace = fixture / "hooks"
        config_dir = fixture / "claude-config"
        runtime = fixture / "runtime"
        routing = fixture / "routing"
        checkpoints = fixture / "checkpoints"
        for directory in (project, trace, config_dir, runtime, routing, checkpoints):
            _private_mkdir(directory)
        registry = ScopeRegistry()
        project_identity = registry.register_project(project, user_profile_id=PROFILE, enabled=True)
        registry.save(fixture / "scopes.json")
        config = MnemosConfig()
        config.modules.surprisal = False
        config.modules.mutable_rag = False
        config.modules.affective = False
        config.modules.sleep = False
        config.modules.spreading = False

        wrapper = fixture / "hook.py"
        _write_hook_wrapper(fixture, wrapper, trace, config_dir)
        hooks = {
            event: [
                {
                    "hooks": [
                        {
                            "type": "command",
                            "command": (
                                f'"{Path(sys.executable).as_posix()}" '
                                f'"{wrapper.as_posix()}" {event} '
                                f'"{config_dir.as_posix()}"'
                            ),
                            "timeout": HOOK_TIMEOUT_SECONDS,
                        }
                    ]
                }
            ]
            for event in ("SessionStart", "UserPromptSubmit", "Stop")
        }
        settings = fixture / "claude-settings.json"
        settings.write_text(json.dumps({"hooks": hooks}, indent=2), encoding="utf-8")
        empty_mcp = fixture / "empty-mcp.json"
        empty_mcp.write_text('{"mcpServers":{}}', encoding="utf-8")
        env = _clean_environment(
            {
                "MNEMOS_RUNTIME_DESCRIPTOR": str(fixture / "runtime.json"),
                "MNEMOS_SCOPE_REGISTRY": str(fixture / "scopes.json"),
                "MNEMOS_USER_PROFILE_ID": PROFILE,
                "MNEMOS_PROJECT_ROOT": str(project),
                "MNEMOS_RUNTIME_DIR": str(runtime),
                "MNEMOS_CLAUDE_CHECKPOINT_DIR": str(checkpoints),
                "MNEMOS_ROUTING_REVIEW_DIR": str(routing),
                "MNEMOS_CONFIG_PATH": str(fixture / "config.toml"),
                "MNEMOS_AUTO_START": "false",
                "DISABLE_AUTOUPDATER": "1",
                "CLAUDE_CONFIG_DIR": str(config_dir),
            }
        )
        auth_copy = _copy_private_auth(config_dir)
        service, store, context = _start_service(fixture, registry, config)
        codename = "mnemos-" + secrets.token_hex(6)
        seed_content = SEED_PREFIX + codename + "."
        seed = service.submit(
            "remember",
            {
                "context": context.model_dump(mode="json"),
                "event_id": "native-claude-persistent-v5-seed-" + codename,
                "content": seed_content,
                "scope": "project",
            },
        )
        maintenance = service.submit("maintenance", {"context": context.model_dump(mode="json")})
        before_ids = {event.id for event in store.list_episodic_events()}
        prompt = (
            "Using only the supplied project memory, answer in one short sentence: "
            "what is the private release codename for this project? Do not use tools."
        )
        command = [
            str(CLAUDE_EXE),
            "--print",
            "--verbose",
            "--output-format",
            "stream-json",
            "--include-hook-events",
            "--setting-sources",
            "project",
            "--settings",
            str(settings),
            "--strict-mcp-config",
            "--mcp-config",
            str(empty_mcp),
            "--tools",
            "",
            "--disable-slash-commands",
            "--permission-mode",
            "plan",
            "--session-id",
            str(uuid4()),
            "--model",
            "claude-sonnet-4-6",
            "--max-budget-usd",
            "0.08",
            "--system-prompt",
            "Answer briefly using only supplied project memory. Do not use tools.",
            prompt,
        ]
        report["command"] = [
            item if item != prompt else "<random-codename prompt>" for item in command
        ]
        report["fixture_layout"] = {
            "project": str(project),
            "claude_config_dir": str(config_dir),
            "private_trace_dir": str(trace),
            "private_native_stdout": str(fixture / "native-stdout.jsonl"),
            "private_native_stderr": str(fixture / "native-stderr.log"),
            "project_runtime_descriptor_present": (project / "runtime.json").exists(),
            "auth_copy_performed": auth_copy is not None,
        }
        started = time.monotonic()
        process = subprocess.Popen(
            command,
            cwd=str(project),
            env=env,
            stdin=subprocess.DEVNULL,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="utf-8",
            errors="replace",
            shell=False,
            creationflags=getattr(subprocess, "CREATE_NEW_PROCESS_GROUP", 0),
        )
        timed_out = False
        try:
            stdout, stderr = process.communicate(timeout=MAX_NATIVE_SECONDS)
        except subprocess.TimeoutExpired:
            timed_out = _terminate_process_tree(process)
            stdout, stderr = process.communicate(timeout=15)
        elapsed_ms = round((time.monotonic() - started) * 1000, 1)
        native_stdout = fixture / "native-stdout.jsonl"
        native_stderr = fixture / "native-stderr.log"
        native_stdout.write_text(stdout, encoding="utf-8")
        native_stderr.write_text(stderr, encoding="utf-8")
        stream = _native_stream_summary(stdout, codename)
        trace_rows = [_hook_record_summary(path) for path in sorted(trace.glob("*.json"))]
        report["native_process"] = {
            "returncode": process.returncode,
            "timed_out": timed_out,
            "elapsed_ms": elapsed_ms,
            "stdout_sha256": _sha256(stdout),
            "stdout_bytes": len(stdout.encode("utf-8")),
            "stderr_sha256": _sha256(stderr),
            "stderr_bytes": len(stderr.encode("utf-8")),
            "private_stdout": str(native_stdout),
            "private_stderr": str(native_stderr),
            "stream": stream,
        }
        stop_paths = [
            path for path in sorted(trace.glob("*.json")) if path.name.startswith("Stop-")
        ]
        if not stop_paths:
            raise RuntimeError("native Claude did not invoke the Stop hook")
        stop_record = json.loads(stop_paths[-1].read_text(encoding="utf-8"))
        stop_payload = stop_record.get("payload") if isinstance(stop_record, Mapping) else None
        if not isinstance(stop_payload, Mapping):
            raise RuntimeError("private Stop trace has no native payload")
        transcript_path_value = stop_payload.get("transcript_path")
        if not isinstance(transcript_path_value, str) or not transcript_path_value.strip():
            raise RuntimeError("native Claude did not provide a persistent transcript path")
        transcript = _validate_transcript(Path(transcript_path_value), config_dir)
        checkpoint = _validate_checkpoint(checkpoints, transcript)
        expected_scope = {
            "user_profile_id": PROFILE,
            "project_id": project_identity.id,
            "workspace_id": None,
            "session_id": stop_payload["session_id"],
            "host": "claude-code",
        }
        if checkpoint["scope"] != expected_scope:
            raise RuntimeError("Native checkpoint scope does not match this fixture")
        events = list(store.list_episodic_events())
        captured = [event for event in events if event.id not in before_ids]
        user_capture = any(
            event.role == "user"
            and event.content == prompt
            and event.metadata.get("source_kind") == "claude_code_transcript"
            for event in captured
        )
        assistant_capture = any(
            event.role == "assistant"
            and event.metadata.get("source_kind") == "claude_code_transcript"
            for event in captured
        )
        event_identity_ok = bool(captured) and all(
            str(event.metadata.get("source_event_id", "")).startswith("claude:v2:")
            for event in captured
        )
        report["seed"] = {
            "project_id": project_identity.id,
            "remember_status": seed.get("status"),
            "remember_committed": seed.get("committed"),
            "maintenance_status": maintenance.get("status"),
            "codename_sha256": _sha256(codename),
            "seed_content_sha256": _sha256(seed_content),
        }
        report["native_hook_trace"] = {
            "available": bool(trace_rows),
            "private_exact_payload_dir": str(trace),
            "records": trace_rows,
            "payload_retention": "exact payloads and adapter stdout/stderr are private fixture files; report stores digests and safe markers",
        }
        report["persistent_transcript"] = transcript
        report["checkpoint"] = checkpoint
        report["captured_roles_before_restart"] = {
            "event_count": len(captured),
            "role_counts": {
                "user": sum(1 for event in captured if event.role == "user"),
                "assistant": sum(1 for event in captured if event.role == "assistant"),
            },
            "single_turn_expected": True,
            "extra_capture_count": max(0, len(captured) - 2),
            "exactly_one_user": sum(1 for event in captured if event.role == "user") == 1,
            "exactly_one_assistant": sum(1 for event in captured if event.role == "assistant") == 1,
            "user_capture": user_capture,
            "assistant_capture": assistant_capture,
            "claude_v2_identity": event_identity_ok,
            "transcript_source_ref_observed": any(
                isinstance(event.metadata.get("source_ref"), str)
                and _sha256(event.metadata["source_ref"]) in transcript["source_ref_sha256"]
                for event in captured
            ),
        }
        report["transcript_coverage"] = {
            "status": "private_persistent_jsonl_verified",
            "payload_without_transcript_path_count": sum(
                row["transcript_state"] == "absent" for row in trace_rows
            ),
            "path_provided_uninspected_count": sum(
                row["transcript_state"] == "path_provided_uninspected" for row in trace_rows
            ),
            "limitation": "The transcript remains private to the disposable CLAUDE_CONFIG_DIR; the report stores its path, digest, JSONL role summary, and source reference digests only.",
        }
        service.stop()
        service = None
        restart_service, restart_store, _ = _start_service(fixture, registry, config)
        restart_events = list(restart_store.list_episodic_events())
        restart_ids = {event.id for event in restart_events}
        replay_before_ids = set(restart_ids)
        replay = subprocess.run(
            [sys.executable, str(wrapper), "Stop", str(config_dir)],
            cwd=str(project),
            env=env,
            input=json.dumps(stop_payload),
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            check=False,
            timeout=30,
        )
        replay_maintenance = restart_service.submit(
            "maintenance", {"context": context.model_dump(mode="json")}
        )
        replay_events = list(restart_store.list_episodic_events())
        replay_ids = {event.id for event in replay_events}
        checkpoint_after_replay = _validate_checkpoint(checkpoints, transcript)
        replay_same_events = replay_ids == replay_before_ids
        report["replay"] = {
            "returncode": replay.returncode,
            "stdout_sha256": _sha256(replay.stdout),
            "stderr_sha256": _sha256(replay.stderr),
            "event_count_before": len(replay_before_ids),
            "event_count_after": len(replay_ids),
            "no_extra_events": replay_same_events,
            "maintenance_status": replay_maintenance.get("status"),
            "checkpoint_offset_after": checkpoint_after_replay["offset"],
            "checkpoint_offset_non_decreasing": checkpoint_after_replay["offset"]
            >= checkpoint["offset"],
        }
        report["restart"] = {
            "events_survive_restart": all(event.id in restart_ids for event in captured),
            "event_count_after_restart": len(restart_events),
            "claude_v2_ids_after_restart": all(
                str(event.metadata.get("source_event_id", "")).startswith("claude:v2:")
                for event in restart_events
                if event.metadata.get("source_kind") == "claude_code_transcript"
            ),
        }
        restart_service.stop()
        restart_service = None
        report["reviewed_synthetic_answer"] = {
            "answer_text_digests": stream["answer_text_digests"],
            "random_canary_exact_match": stream["answer_mentions_random_canary"],
            "quality_claim": "none",
        }
        report["cost"] = {
            "total_cost_usd": stream["total_cost_usd"],
            "observed_cost_values": stream["cost_usd_observed"],
            "source": "native Claude stream result fields; null means the native output did not expose numeric cost",
        }
        report["criteria"] = {
            "native_completed": process.returncode == 0 and not timed_out,
            "hook_payloads_recorded": bool(trace_rows),
            "user_capture": user_capture,
            "assistant_capture": assistant_capture,
            "claude_v2_identity": event_identity_ok,
            "transcript_source_ref_observed": report["captured_roles_before_restart"][
                "transcript_source_ref_observed"
            ],
            "persistent_transcript_valid": transcript["path"].startswith(str(config_dir.resolve())),
            "checkpoint_valid": checkpoint["offset"] <= transcript["bytes"],
            "events_survive_restart": report["restart"]["events_survive_restart"],
            "replay_no_extra_events": replay_same_events and replay.returncode == 0,
            "exactly_two_native_captures": (
                report["captured_roles_before_restart"]["exactly_one_user"]
                and report["captured_roles_before_restart"]["exactly_one_assistant"]
                and len(captured) == 2
            ),
            "checkpoint_offset_non_decreasing": checkpoint_after_replay["offset"]
            >= checkpoint["offset"],
            "exact_canary_answer": stream["answer_mentions_random_canary"],
            "no_tool_turn": (
                stream["record_count"] > 0
                and stream["non_json_line_count"] == 0
                and not stream["tool_call_observed"]
            ),
        }
        report["status"] = "observed" if all(report["criteria"].values()) else "not_observed"
    except Exception as exc:
        (fixture / "runner-error.txt").write_text(traceback.format_exc(), encoding="utf-8")
        report.update(
            {
                "status": "failed",
                "error_type": type(exc).__name__,
                "error_digest": _sha256(str(exc)),
            }
        )
    finally:
        if process is not None and process.poll() is None:
            try:
                _terminate_process_tree(process)
            except Exception:
                process.kill()
                process.wait(timeout=10)
        for active in (service, restart_service):
            if active is not None:
                try:
                    active.stop()
                except Exception:
                    pass
        if auth_copy is not None:
            auth_copy.unlink(missing_ok=True)
        report["fixture"] = str(fixture)
        report["private_fixture_preserved"] = True
        report["auth_copy_removed"] = auth_copy is None or not auth_copy.exists()
        report["runtime_descriptor_absent"] = not (fixture / "runtime.json").exists()
        if not report["runtime_descriptor_absent"]:
            report["status"] = "not_observed"
        report["finished_at"] = datetime.now(timezone.utc).isoformat()
        (OUTPUT_DIR / "report.json").write_text(
            json.dumps(report, indent=2) + "\n", encoding="utf-8"
        )
    print(json.dumps({key: report.get(key) for key in ("status", "fixture", "error_type")}))
    return 0 if report["status"] == "observed" else 1
