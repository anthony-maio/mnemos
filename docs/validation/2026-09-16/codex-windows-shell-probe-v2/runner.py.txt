"""Wheel-isolated offline check for the Codex Windows hook shell boundary."""

from __future__ import annotations

import hashlib
import json
import os
import subprocess
import sys
import tempfile
import time
import zipfile
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

WORKTREE = Path(__file__).resolve().parents[1]
OUTPUT_DIR = WORKTREE / "docs" / "validation" / "2026-09-16" / "codex-windows-shell-probe-v2"
PROFILE = "codex-shell-probe-v2"
WHEEL_BUILD_DIR = Path(r"D:\WindowsTemp\mnemos-shell-wheel-build")
WHEEL_VENV = Path(r"D:\WindowsTemp\wheel-smoke")


def _isolated_env(integration_env: dict[str, str]) -> dict[str, str]:
    env = dict(os.environ)
    for key in list(env):
        if key.startswith("MNEMOS_") or key == "PYTHONPATH":
            env.pop(key, None)
    env.update(integration_env)
    return env


def _run_subprocess(
    command: list[str] | str,
    *,
    payload: dict[str, Any],
    cwd: Path,
    env: dict[str, str],
    transport: str,
) -> dict[str, Any]:
    started = time.perf_counter()
    completed = subprocess.run(
        command,
        input=json.dumps(payload, ensure_ascii=False),
        capture_output=True,
        text=True,
        cwd=str(cwd),
        env=env,
        shell=False,
        timeout=20,
    )
    stdout = completed.stdout.strip()
    parsed: Any = None
    parse_error: str | None = None
    if stdout:
        try:
            parsed = json.loads(stdout)
        except json.JSONDecodeError as exc:
            parse_error = type(exc).__name__
    return {
        "transport": transport,
        "returncode": completed.returncode,
        "elapsed_ms": round((time.perf_counter() - started) * 1000, 2),
        "stdout": parsed,
        "stdout_parse_error": parse_error,
        "stderr": completed.stderr,
    }


def _wheel_info() -> dict[str, str]:
    wheels = sorted(WHEEL_BUILD_DIR.glob("*.whl"), key=lambda path: path.stat().st_mtime)
    if not wheels:
        raise RuntimeError("wheel build output is missing")
    wheel = wheels[-1]
    return {
        "path": str(wheel),
        "sha256": hashlib.sha256(wheel.read_bytes()).hexdigest(),
    }


def _wheel_source_check(wheel_path: Path) -> dict[str, Any]:
    checked = 0
    mismatches: list[str] = []
    with zipfile.ZipFile(wheel_path) as archive:
        for name in archive.namelist():
            if not name.startswith("mnemos/") or not name.endswith(".py"):
                continue
            checked += 1
            source = WORKTREE / name
            if not source.exists() or source.read_bytes() != archive.read(name):
                mismatches.append(name)
    return {"checked_python_files": checked, "mismatches": mismatches}


def _git_checkpoint() -> str:
    completed = subprocess.run(
        ["git", "-C", str(WORKTREE), "rev-parse", "HEAD"],
        capture_output=True,
        text=True,
        check=True,
        timeout=5,
    )
    return completed.stdout.strip()


def main() -> int:
    from importlib.metadata import version

    import mnemos

    package_path = Path(mnemos.__file__).resolve()
    expected_site_packages = WHEEL_VENV / "Lib" / "site-packages"
    if expected_site_packages not in package_path.parents:
        raise RuntimeError(f"wheel isolation failed: imported {package_path}")
    if os.environ.get("PYTHONPATH"):
        raise RuntimeError("probe runner inherited PYTHONPATH")

    from mnemos.config import MnemosConfig
    from mnemos.engine import MnemosEngine
    from mnemos.hosts import _codex_hook_command
    from mnemos.memory_service import LocalMemoryService
    from mnemos.scopes import ScopeRegistry
    from mnemos.service import MnemosService
    from mnemos.utils.embeddings import SimpleEmbeddingProvider
    from mnemos.utils.llm import UnavailableLLMProvider
    from mnemos.utils.storage import SQLiteStore

    fixture_root = Path(
        tempfile.mkdtemp(prefix="mnemos-codex-shell-probe-v2-", dir=r"D:\WindowsTemp")
    )
    project = fixture_root / "project"
    project.mkdir()
    registry_path = fixture_root / "scopes.json"
    descriptor_path = fixture_root / "runtime.json"
    db_path = fixture_root / "memory.sqlite"
    config_path = fixture_root / "mnemos.toml"
    registry = ScopeRegistry()
    project_identity = registry.register_project(project, user_profile_id=PROFILE, enabled=True)
    registry.save(registry_path)
    config = MnemosConfig()
    config.modules.surprisal = False
    config.modules.mutable_rag = False
    config.modules.affective = False
    config.modules.sleep = False
    config.modules.spreading = False
    service: MnemosService | None = None
    service_restart: MnemosService | None = None
    direct: dict[str, Any]
    shell_result: dict[str, Any]
    try:
        store = SQLiteStore(str(db_path))
        engine = MnemosEngine(
            config=config,
            llm=UnavailableLLMProvider("offline shell probe cognition unavailable"),
            embedder=SimpleEmbeddingProvider(dim=64),
            store=store,
            scope_registry=registry,
            require_scope_context=True,
        )
        domain = LocalMemoryService(engine, registry)
        service = MnemosService(
            dispatcher=domain,
            descriptor_path=descriptor_path,
            lock_path=fixture_root / "service.lock",
            startup_timeout=3,
            dispatch_timeout=10,
        )
        service.start()
        context = registry.resolve(
            project,
            user_profile_id=PROFILE,
            session_id="shell-seed-session",
            host="codex",
        )
        integration_env = {
            "MNEMOS_CONFIG_PATH": str(config_path),
            "MNEMOS_RUNTIME_DESCRIPTOR": str(descriptor_path),
            "MNEMOS_SCOPE_REGISTRY": str(registry_path),
            "MNEMOS_USER_PROFILE_ID": PROFILE,
        }
        seed = service.submit(
            "remember",
            {
                "context": context.model_dump(mode="json"),
                "event_id": "codex-shell-v2-seed",
                "content": "The consolidated shell seed fact is durable local state.",
                "scope": "project",
            },
        )
        seed_maintenance = service.submit(
            "maintenance", {"context": context.model_dump(mode="json")}
        )
        direct_payload = {
            "hook_event_name": "UserPromptSubmit",
            "cwd": str(project),
            "session_id": "shell-direct-session",
            "turn_id": "shell-direct-turn",
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "prompt": "Where is the consolidated shell seed fact for the direct path?",
        }
        raw_payload = {
            **direct_payload,
            "session_id": "shell-raw-session",
            "turn_id": "shell-raw-turn",
            "prompt": "Where is the consolidated shell seed fact for the raw path?",
        }
        child_env = _isolated_env(integration_env)
        direct = _run_subprocess(
            [str(WHEEL_VENV / "Scripts" / "python.exe"), "-m", "mnemos.adapters.codex"],
            payload=direct_payload,
            cwd=project,
            env=child_env,
            transport="direct_python",
        )
        hook_command = _codex_hook_command(integration_env)
        comspec = os.environ.get("COMSPEC", r"C:\Windows\System32\cmd.exe")
        raw_commandline = f'{comspec} /d /c "{hook_command}"'
        shell_result = _run_subprocess(
            raw_commandline,
            payload=raw_payload,
            cwd=project,
            env=child_env,
            transport="tagged_raw_cmd",
        )
        seed_text = "The consolidated shell seed fact is durable local state."
        for label, invocation in (("direct Python", direct), ("raw cmd", shell_result)):
            output = invocation.get("stdout")
            context_text = ""
            if isinstance(output, dict):
                hook_output = output.get("hookSpecificOutput")
                if isinstance(hook_output, dict):
                    context_text = str(hook_output.get("additionalContext") or "")
            if seed_text not in context_text:
                raise AssertionError(f"{label} adapter output must contain consolidated seed")
        capture_maintenance = service.submit(
            "maintenance", {"context": context.model_dump(mode="json")}
        )
        before_restart = service.submit(
            "list",
            {"context": context.model_dump(mode="json"), "allowed_scopes": ["project"], "limit": 50},
        )
        before_fact_contents = [
            item.get("content")
            for item in before_restart.get("facts", [])
            if isinstance(item, dict)
        ]
        expected_direct = direct_payload["prompt"]
        expected_raw = raw_payload["prompt"]
        if expected_direct not in before_fact_contents or expected_raw not in before_fact_contents:
            raise AssertionError("both adapter prompts must persist before restart")
        direct["stderr"] = direct["stderr"].replace(str(fixture_root), "<fixture-root>")
        shell_result["stderr"] = shell_result["stderr"].replace(str(fixture_root), "<fixture-root>")
        service.stop()
        service = None

        registry_restart = ScopeRegistry.load(registry_path)
        restart_store = SQLiteStore(str(db_path))
        restart_engine = MnemosEngine(
            config=config,
            llm=UnavailableLLMProvider("offline shell probe cognition unavailable"),
            embedder=SimpleEmbeddingProvider(dim=64),
            store=restart_store,
            scope_registry=registry_restart,
            require_scope_context=True,
        )
        restart_domain = LocalMemoryService(restart_engine, registry_restart)
        service_restart = MnemosService(
            dispatcher=restart_domain,
            descriptor_path=fixture_root / "runtime-restart.json",
            lock_path=fixture_root / "service-restart.lock",
            startup_timeout=3,
            dispatch_timeout=10,
        )
        service_restart.start()
        restart_context = registry_restart.resolve(
            project,
            user_profile_id=PROFILE,
            session_id="shell-restart-session",
            host="codex",
        )
        after_restart = service_restart.submit(
            "list",
            {
                "context": restart_context.model_dump(mode="json"),
                "allowed_scopes": ["project"],
                "limit": 50,
            },
        )
        restart_events = restart_store.list_episodic_events(
            user_profile_id=PROFILE,
            newest_first=True,
            limit=50,
        )
        after_fact_contents = [
            item.get("content")
            for item in after_restart.get("facts", [])
            if isinstance(item, dict)
        ]
        if expected_direct not in after_fact_contents or expected_raw not in after_fact_contents:
            raise AssertionError("both adapter prompts must persist after restart")
        wheel_path = Path(_wheel_info()["path"])
        result = {
            "probe": "codex-windows-shell-probe-v2",
            "host_calls": False,
            "model_calls": False,
            "package": {
                "version": version("mnemos-memory"),
                "module_path": str(package_path),
                "wheel": _wheel_info(),
                "source_checkpoint": _git_checkpoint(),
                "source_comparison": _wheel_source_check(wheel_path),
                "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
            },
            "python": {"executable": str(Path(sys.executable).resolve()), "version": sys.version.split()[0]},
            "pythonpath": os.environ.get("PYTHONPATH"),
            "fixture": {
                "root": str(fixture_root),
                "project": str(project),
                "database": str(db_path),
                "registry": str(registry_path),
                "descriptor": str(descriptor_path),
                "retained": True,
            },
            "project_identity": project_identity.id,
            "integration_env_keys": sorted(integration_env),
            "generated_hook_command": hook_command,
            "tagged_windows_commandline": raw_commandline,
            "seed": seed,
            "seed_maintenance": seed_maintenance,
            "capture_maintenance": capture_maintenance,
            "direct_python": direct,
            "raw_cmd": shell_result,
            "before_restart": {
                "fact_contents": [
                    item.get("content")
                    for item in before_restart.get("facts", [])
                    if isinstance(item, dict)
                ],
                "job_summary": before_restart.get("jobs"),
            },
            "after_restart": {
                "fact_contents": after_fact_contents,
                "episode_count": len(restart_events),
                "episode_roles": [event.role for event in restart_events],
            },
            "episode_survival_checks": {
                "direct_prompt": {
                    "expected": expected_direct,
                    "before_restart": expected_direct in before_fact_contents,
                    "after_restart": expected_direct in after_fact_contents,
                },
                "raw_prompt": {
                    "expected": expected_raw,
                    "before_restart": expected_raw in before_fact_contents,
                    "after_restart": expected_raw in after_fact_contents,
                },
            },
        }
    finally:
        if service_restart is not None:
            service_restart.stop()
        if service is not None:
            service.stop()

    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    (OUTPUT_DIR / "result.json").write_text(
        json.dumps(result, indent=2, ensure_ascii=False) + "\n", encoding="utf-8"
    )
    print(json.dumps({"status": "complete", "result": str(OUTPUT_DIR / "result.json")}))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
