import json

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

root = Path(__file__).resolve().parents[1]
output = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-native-crosshost-v1')
report_path = output/'report.json'
report = json.loads(report_path.read_text(encoding='utf-8'))
fixture = Path(report['fixture'])
target = report['target']
assert report['status'] == 'observed_partial'
assert report['source_commit'] == '1068a629887ef9a5d267a19ead6eb64b2f125757'
assert not report['cleanup']['errors']
assert all(v for k,v in report['cleanup'].items() if k != 'errors')

    assert not path.exists()

assert not archive.exists()
selected_codex = json.loads((output/'codex-mcp-completions.json').read_text(encoding='utf-8'))
assert len(selected_codex) == 2
codex_raw = []
codex_payloads = []
for selected, expected in zip(selected_codex, ('mnemos_store','mnemos_consolidate')):
    source = fixture/'codex-home/.codex'/selected['source']
    raw = source.read_bytes().splitlines(keepends=True)[selected['line']-1]
    assert json.loads(raw) == selected['record']
    p = selected['record']['payload']
    assert p['invocation']['server'] == 'mnemos' and p['invocation']['tool'] == expected
    assert set(p['result']) == {'Ok'} and p['result']['Ok']['isError'] is False
    result = p['result']['Ok']
    decoded = json.loads(result['structuredContent']['result'])
    assert decoded == json.loads(result['content'][0]['text'])
    codex_payloads.append(decoded)
    codex_raw.append(raw)
assert selected_codex[0]['record']['payload']['invocation']['arguments'] == {
    'content':target['content'],'role':'user','scope':'project'}
stored, consolidated = codex_payloads
assert stored['status'] == 'committed' and stored['event_committed'] is True
assert stored['scope_id'] == target['project_id'] and stored['scope'] == 'project'
assert consolidated['maintenance']['failed_count'] == 0 and consolidated['completed'] > 0

selected_claude = []
decoded_phases = {}
for phase, key, names in (
    ('crosshost-read','claude_read',['mnemos_retrieve']),
    ('crosshost-correct','claude_correction',['mnemos_correct','mnemos_consolidate','mnemos_retrieve'])):
    raw = (fixture/f'native-{phase}-stdout.jsonl').read_bytes()
    # Runner writes UTF-8 text with Windows newline translation.
    text = raw.decode('utf-8').replace('\r\n','\n')
    assert sha(text.encode()) == report[key]['stdout_sha256']
    assert report[key]['returncode'] == 0 and report[key]['timed_out'] is False
    calls, results = [], []
    for number, line in enumerate(raw.splitlines(keepends=True),1):
        record = json.loads(line)
        for node in record.get('message',{}).get('content',[]):
            if not isinstance(node,dict) or node.get('type') not in ('tool_use','tool_result'):
                continue
            selected_claude.append({'phase':phase,'line':number,'line_sha256':sha(line),'node':node})
            (calls if node['type']=='tool_use' else results).append(node)
    assert [c['name'].split('__')[-1] for c in calls] == names
    assert [c['id'] for c in calls] == [r['tool_use_id'] for r in results]
    assert len(set(c['id'] for c in calls)) == len(calls)
    payloads = []
    for result, observation in zip(results, report[key]['results']):
        assert not result.get('is_error',False)
        assert sha(json.dumps(result,sort_keys=True,separators=(',',':')).encode()) == observation['sha256']
        decoded = json.loads(result['content'])
        if 'result' in decoded:
            decoded = json.loads(decoded['result'])
        assert decoded.get('status') != 'error' and 'error' not in decoded
        payloads.append(decoded)
    decoded_phases[phase] = {'calls':calls,'payloads':payloads}

def target_fact(payload, version, content):
    assert target['foreign_fact_id'] not in [f['id'] for f in payload['facts']]
    matches = [f for f in payload['facts'] if f['id']==target['fact_id']]
    assert len(matches)==1
    fact=matches[0]
    assert fact['version']==version and fact['content']==content
    assert fact['scope']=='project' and fact['scope_id']==target['project_id']
    return fact

    revisions=[dict(r) for r in db.execute('select fact_id,version,content from fact_revisions where fact_id=? order by version',(target['fact_id'],))]
