def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def load(name):
    path = ROOT / '.venv' / name
    if digest(path.read_bytes()) != PINS[name]:
        raise RuntimeError('helper pin mismatch: ' + name)
    spec = importlib.util.spec_from_file_location(name.replace('.', '_'), path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def codex_completions(home):
    """Only actual Mnemos completion records; no auth, logs, or full session copy."""
    selected = []
    for path in home.rglob('*.jsonl'):
        for number, line in enumerate(path.read_text(encoding='utf-8').splitlines(), 1):
            value = json.loads(line)
            payload = value.get('payload', {})
            if payload.get('type') == 'mcp_tool_call_end' and payload.get('invocation', {}).get('server') == 'mnemos':
                selected.append({'source': path.relative_to(home).as_posix(), 'line': number, 'record': value})
    return selected


def decode_codex(selected, expected):
    assert len(selected) == len(expected)
    decoded = []
    for item, tool in zip(selected, expected):
        payload = item['record']['payload']
        assert payload['invocation']['server'] == 'mnemos'
        assert payload['invocation']['tool'] == tool
        assert set(payload['result']) == {'Ok'}
        result = payload['result']['Ok']
        assert result['isError'] is False
        value = json.loads(result['structuredContent']['result'])
        assert value == json.loads(result['content'][0]['text'])
        assert value.get('status') != 'error' and 'error' not in value
        decoded.append(value)
    return decoded


def force_unelevated(path, lifecycle_module):
    text = path.read_text(encoding='utf-8')
    if '[windows]' not in text:
        text = text.rstrip() + '\n\n[windows]\nsandbox = "unelevated"\n'
        path.write_text(text, encoding='utf-8')
    else:
        text = text.replace('sandbox = "workspace-write"', 'sandbox = "unelevated"')
        path.write_text(text, encoding='utf-8')
    lifecycle_module._protect_file(path)
    assert tomllib.loads(path.read_text(encoding='utf-8'))['windows']['sandbox'] == 'unelevated'

            and native_forget_calls[1]['input'].get('query') == CONTROL_QUERY)
        checkpoint()
        assert all(report['forget_contract'].values())
        assert forget.get('returncode') == 0 and forget.get('timed_out') is False
        assert forget_payloads and any(item.get('forgotten') is True for item in forget_payloads)
        forget_facts = [fact for item in forget_retrieve_payloads for fact in item.get('facts', [])]
        assert target_id not in {fact.get('id') for fact in forget_facts}
        assert unrelated_id in {fact.get('id') for fact in forget_facts}
        report['claude_forget'] = claude._safe_phase_summary(forget)
        report['claude_forget_result'] = forget_results
        checkpoint()

        close()
        start(registry, config)
        assert store.get(target_id) is None
        assert store.get(unrelated_id) is not None

        # Replay exact source identifiers through the service boundary. The
        # teaching event should be suppressed by its forget tombstone. The
        # original correction request cannot target an erased chunk and must
        # not recreate it.
        replay_results = []
        for source in original_sources:
            replay_payload = {
                'event_id': source['event_id'], 'context': source['context'],
                'content': source['content'], 'role': source['role'],
                'scope': source['scope'], 'source_ref': source['source_ref'],
            }
            if source['source_kind'] == 'explicit_correction':
                replay_payload.update({'fact_id': target_id, 'expected_version': 1})
                try:
                    replay_results.append(service.submit('correct', replay_payload))
                except Exception as exc:
                    assert type(exc).__name__ == 'KeyError'
                    replay_results.append({'status': 'rejected', 'error_type': 'KeyError'})
                else:
                    raise RuntimeError('correction replay did not reject the erased target')
            else:
                replay_result = service.submit('remember', replay_payload)
                assert replay_result.get('processing_status') == 'suppressed'
                replay_results.append(replay_result)
        report['direct_replay'] = replay_results
        replay_maintenance = service.submit('maintenance', {})
        assert replay_maintenance['maintenance']['failed_count'] == 0
        assert store.get(target_id) is None
        assert not any(f.content in {CONTENT, REPLACEMENT} and f.metadata.get('assertion_status') == 'asserted' for f in store.get_all())
        report['direct_replay_maintenance'] = {'target_present': store.get(target_id) is not None,
                                               'unrelated_present': store.get(unrelated_id) is not None}
        checkpoint()
        close()
        start(registry, config)
        assert store.get(target_id) is None and store.get(unrelated_id) is not None

        # A fresh native Codex session must retain the unrelated project fact,
        # exclude the foreign canary, and omit the forgotten target assertion.
        final_home = fixture / 'codex-final-home'
        lifecycle._private_mkdir(final_home)
        final_generated = apply_host_integration('codex', mnemos_config_path=settings, cwd=project, home=final_home,
            integration_env={key: selectors[key] for key in ('MNEMOS_CONFIG_PATH','MNEMOS_SCOPE_REGISTRY','MNEMOS_USER_PROFILE_ID','MNEMOS_RUNTIME_DESCRIPTOR')})
        final_config = final_generated.config_path
        force_unelevated(final_config, lifecycle)
        final_auth = lifecycle._auth_copy(final_config.parent)
        final_env = dict(env, CODEX_HOME=str(final_config.parent), MNEMOS_SESSION_ID='native-crosshost-codex-final')
        report['codex_final_home'] = str(final_config.parent.relative_to(fixture))
        report['codex_final_entrypoint'] = wrapper._validate_mcp_entrypoint(lifecycle, final_config, final_env)
        report['codex_final_config_sha256'] = digest(final_config.read_bytes())
        final_prompt = ('Use only Mnemos MCP tools. Call mnemos_retrieve exactly twice in this order. '
                        'First arguments: ' + json.dumps({'query': TARGET_QUERY, 'current_scope': 'project', 'allowed_scopes': ['project']})
                        + '. Second arguments: ' + json.dumps({'query': CONTROL_QUERY, 'current_scope': 'project', 'allowed_scopes': ['project']})
                        + '. Use those exact query strings. Report each result separately. Do not call any other tool.')
        final_command = native._native_command(lifecycle, project, final_prompt)
        print(json.dumps({'stage': 'native_codex_recall_after_forget_restart_and_replay'}), flush=True)
        checkpoint()
        started = time.monotonic()
        process = subprocess.Popen(final_command, cwd=project, env=final_env, shell=False,
            creationflags=getattr(subprocess, 'CREATE_NEW_PROCESS_GROUP', 0))
        try:
            process.wait(timeout=180)
            final_timed_out = False
        except subprocess.TimeoutExpired:
            lifecycle._terminate_process_tree(process)
            final_timed_out = True
        report['codex_final_process'] = {'returncode': process.returncode, 'timed_out': final_timed_out,
                                         'elapsed_ms': round((time.monotonic()-started)*1000, 1)}
        final_selected = codex_completions(final_config.parent)
        (args.output.parent / 'codex-final-completions.json').write_text(json.dumps(final_selected, indent=2)+'\n', encoding='utf-8')
        final_result, control_result = decode_codex(final_selected, ('mnemos_retrieve', 'mnemos_retrieve'))
        assert [item['record']['payload']['invocation']['arguments']['query'] for item in final_selected] == [TARGET_QUERY, CONTROL_QUERY]
        final_facts = final_result.get('facts', [])
        control_facts = control_result.get('facts', [])
        assert target_id not in {fact.get('id') for fact in final_facts}
        assert foreign_id not in {fact.get('id') for fact in final_facts}
        assert unrelated_id in {fact.get('id') for fact in control_facts}
        assert target_id not in {fact.get('id') for fact in control_facts}
        assert foreign_id not in {fact.get('id') for fact in control_facts}
        assert not any(fact.get('content') in {CONTENT, REPLACEMENT} and fact.get('assertion_status') == 'asserted' for fact in final_facts)
        mentions = [fact for fact in final_facts if any(term in fact.get('content', '') for term in ('cedar-otter', 'maple-heron'))]
        report['final_mentions'] = [
            {'id': fact.get('id'), 'version': fact.get('version'), 'content': fact.get('content'),
             'assertion_status': fact.get('assertion_status'), 'scope': fact.get('scope'),
             'scope_id': fact.get('scope_id'),
             'qualified_in_context': any(fact['id'] in line and 'captured conversation; unverified, may be outdated' in line for line in final_result['context'].splitlines())}
            for fact in mentions
        ]
        report['criteria']['native_claude_forget_target_absent'] = target_id not in {fact.get('id') for fact in forget_facts}
        report['criteria']['direct_replay_does_not_resurrect_target'] = store.get(target_id) is None
        report['criteria']['native_codex_final_excludes_target_and_foreign'] = target_id not in {fact.get('id') for fact in final_facts} and foreign_id not in {fact.get('id') for fact in final_facts}
        report['criteria']['native_codex_final_retains_unrelated'] = unrelated_id in {fact.get('id') for fact in control_facts}
        report['status'] = 'observed_partial'
        report['remaining'] = ['independent artifact verification', 'broader reviewed related-content forgetting', 'native hook payload replay', 'Hermes native continuity', 'Codex shutdown acceptance', 'complete capture and model quality']
    except Exception as exc:
        report['error_type'] = type(exc).__name__
        report['error_digest'] = digest(str(exc).encode('utf-8'))
        # Fixture-only traceback is not copied into the public artifact.
        import traceback
        (fixture/'runner-error.txt').write_text(traceback.format_exc(), encoding='utf-8')
    finally:
        report['dispatch_traces'] = traces
        native._restore_dispatch(binding)
        errors = lifecycle._cleanup_native_resources(process, auth)
        errors += lifecycle._cleanup_native_resources(None, final_auth)
        errors += lifecycle._close_service(service, store)
        if claude_auth is not None:
            claude_auth.unlink(missing_ok=True)
        if reverse_auth is not None:
            reverse_auth.unlink(missing_ok=True)
        if forget_auth is not None:
            forget_auth.unlink(missing_ok=True)
        report['cleanup'] = {'errors': errors, 'codex_auth_absent': auth is None or not auth.exists(),
                             'claude_auth_absent': claude_auth is None or not claude_auth.exists(),
                             'codex_reverse_auth_absent': reverse_auth is None or not reverse_auth.exists(),
                             'claude_forget_auth_absent': forget_auth is None or not forget_auth.exists(),
                             'codex_final_auth_absent': final_auth is None or not final_auth.exists(),
                             'descriptor_absent': not (fixture/'runtime.json').exists()}
        if errors or not all(v for k,v in report['cleanup'].items() if k != 'errors'):
            report['status'] = 'failed'
        report['finished_at'] = datetime.now(timezone.utc).isoformat()
        checkpoint()
    print(json.dumps({'status': report['status'], 'report': str(args.output)}), flush=True)
    return 0 if report['status'] == 'observed_partial' else 1
