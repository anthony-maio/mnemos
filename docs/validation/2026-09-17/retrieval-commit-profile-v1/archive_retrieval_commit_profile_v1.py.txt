"""Archive the completed retrieval commit profile without copying its fixture."""

from __future__ import annotations

from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import shutil

REPO = Path(__file__).resolve().parents[1]
DEST = REPO / "docs" / "validation" / "2026-09-17" / "retrieval-commit-profile-v1"
RUN = Path(r"C:\Users\Ant\AppData\Local\Temp\mnemos-profile-retrieval-commit-v1")
REPORT = RUN / "report.json"
PROFILE_HELPER = REPO / ".venv" / "profile_retrieval_commit_v1.py"
ARCHIVER = Path(__file__).resolve()
PERFORMANCE_REPORT = (
    REPO
    / "docs"
    / "validation"
    / "2026-09-16"
    / "real-model-v4"
    / "performance-10k"
    / "performance-10k.json"
)
SOURCE_COMMIT = "b752ffbe07e2e32f5ddbf54cc689b8dbbad8c9e0"
PROFILE_HELPER_SHA256 = "b24e5033d94672ff3f5145b2c889a90b831472fcd68b30d2be96aa494c0d2d78"


def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def load_json(path: Path) -> dict[str, object]:
    value = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(value, dict):
        raise TypeError(f"expected object JSON: {path}")
    return value


def validate_report(report: dict[str, object]) -> None:
    if report.get("status") != "completed":
        raise RuntimeError("profiler report is not completed")
    provenance = report.get("provenance")
    if not isinstance(provenance, dict):
        raise RuntimeError("profiler report has no provenance")
    if provenance.get("source_commit") != SOURCE_COMMIT:
        raise RuntimeError("profiler source commit mismatch")
    if provenance.get("helper_sha256") != PROFILE_HELPER_SHA256:
        raise RuntimeError("profiler helper hash mismatch")
    if provenance.get("checkout_status") != "clean":
        raise RuntimeError("profiler did not run from a clean checkout")
    source_files = provenance.get("source_files")
    if not isinstance(source_files, dict) or not source_files:
        raise RuntimeError("profiler source file provenance is missing")
    for details in source_files.values():
        if not isinstance(details, dict) or details.get("checkout_matches_commit") is not True:
            raise RuntimeError("profiler source file does not match its commit")
    before = report.get("sqlite_before")
    after = report.get("sqlite_after")
    if not isinstance(before, dict) or not isinstance(after, dict):
        raise RuntimeError("SQLite settings are missing")
    for settings in (before, after):
        if settings.get("journal_mode") != "wal" or settings.get("synchronous") != 2:
            raise RuntimeError("profiler did not use WAL and synchronous FULL")
    counts = report.get("counts_after_reopen")
    if not isinstance(counts, dict):
        raise RuntimeError("post-reopen validation is missing")
    if counts.get("memory_chunks") != 10000 or counts.get("access_rows") != 10000:
        raise RuntimeError("post-reopen fixture counts are wrong")
    if counts.get("updated_access_counts") != [33, 33, 33, 33, 33]:
        raise RuntimeError("post-reopen updated access counts are wrong")
    measurement = report.get("measurement")
    if not isinstance(measurement, dict):
        raise RuntimeError("measurement is missing")
    rows = measurement.get("rows")
    summary = measurement.get("summary")
    if not isinstance(rows, list) or len(rows) != 33 or not isinstance(summary, dict):
        raise RuntimeError("measurement does not contain 33 calls")
    if summary.get("all_calls_have_one_insert_one_update_one_commit") is not True:
        raise RuntimeError("stage call-count validation failed")
    if summary.get("all_rowcounts_are_five") is not True:
        raise RuntimeError("five-ID rowcount validation failed")


def performance_reference(path: Path) -> dict[str, object]:
    report = load_json(path)
    retrieval = report.get("retrieval")
    if not isinstance(retrieval, dict):
        raise RuntimeError("original performance report has no retrieval object")
    warm = retrieval.get("warm_recall")
    actual = retrieval.get("actual_store_stage_latency")
    raw = retrieval.get("raw_queries")
    if not isinstance(warm, dict) or not isinstance(actual, dict) or not isinstance(raw, list):
        raise RuntimeError("original performance timing fields are incomplete")
    row = next((item for item in raw if isinstance(item, dict) and item.get("index") == 12), None)
    if not isinstance(row, dict):
        raise RuntimeError("original p95 query row is missing")
    if warm.get("p95_ms") != 1088.094 or retrieval.get("recall_gate") != "FAIL":
        raise RuntimeError("original performance gate no longer matches expected evidence")
    return {
        "path": str(path),
        "sha256": sha256(path),
        "status": report.get("status"),
        "schema_version": report.get("schema_version"),
        "retrieval": {
            "query_count": retrieval.get("query_count"),
            "rows_indexed": retrieval.get("rows_indexed"),
            "warm_recall": warm,
            "recall_gate_ms": retrieval.get("recall_gate_ms"),
            "recall_gate": retrieval.get("recall_gate"),
            "actual_store_stage_latency": actual,
            "p95_query_index": row.get("index"),
            "p95_query_timing": {
                key: row.get(key)
                for key in (
                    "latency_ms",
                    "embedding_ms",
                    "scoped_sql_ranking_ms",
                    "lexical_candidate_ms",
                    "mark_and_refresh_ms",
                )
            },
        },
        "copied": False,
    }


def derived_stats(report: dict[str, object]) -> dict[str, object]:
    measurement = report["measurement"]
    assert isinstance(measurement, dict)
    summary = measurement["summary"]
    assert isinstance(summary, dict)
    stages = summary["stage_timings_ms"]
    assert isinstance(stages, dict)
    return {
        "schema_version": 1,
        "source_commit": SOURCE_COMMIT,
        "profile_helper_sha256": PROFILE_HELPER_SHA256,
        "call_count": len(measurement["rows"]),
        "stage_timings_ms": stages,
        "event_counts": summary["event_counts"],
        "interpretation": {
            "result": "The isolated 33-call write profile completed without reproducing the original approximately 796 ms metadata tail.",
            "commit": "The measured commit p95 is 86.213 ms and dominates this isolated fixture's 86.372 ms total p95.",
            "limits": "This is not an end-to-end retrieval measurement, a 500 ms gate, or proof of a product speedup. It does not identify filesystem, WAL, scheduler, or cross-process causes.",
        },
    }


def main() -> None:
    if DEST.exists():
        raise SystemExit(f"refusing to overwrite existing archive: {DEST}")
    if not REPORT.is_file() or not PROFILE_HELPER.is_file() or not PERFORMANCE_REPORT.is_file():
        raise FileNotFoundError("required run, helper, or performance report is missing")
    if sha256(PROFILE_HELPER) != PROFILE_HELPER_SHA256:
        raise RuntimeError("current profile helper hash changed")
    report = load_json(REPORT)
    validate_report(report)
    reference = performance_reference(PERFORMANCE_REPORT)
    DEST.mkdir(parents=True)
    shutil.copyfile(REPORT, DEST / "profile-report.json")
    shutil.copyfile(PROFILE_HELPER, DEST / "profile_retrieval_commit_v1.py.txt")
    shutil.copyfile(ARCHIVER, DEST / "archive_retrieval_commit_profile_v1.py.txt")
    (DEST / "derived-stats.json").write_text(
        json.dumps(derived_stats(report), ensure_ascii=True, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
        newline="\n",
    )
    (DEST / "performance-reference.json").write_text(
        json.dumps(reference, ensure_ascii=True, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
        newline="\n",
    )
    readme = f"""# Retrieval commit profile

This archive preserves one completed, source-bound SQLite write-stage profile.
It created a disposable synthetic fixture with 10,000 production-schema rows,
then measured 33 `mark_retrieved_many` calls with five IDs per call. The
production SQLite settings remained WAL with `synchronous=FULL`; no explicit
checkpoint was issued. Each call recorded exactly one access INSERT, one access
UPDATE, and one commit. The five updated access counts were 33 after closing
and reopening the store.

The isolated profile's total p95 was 86.372 ms. Its commit p95 was 86.213 ms,
while access INSERT p95 was 0.134 ms, access UPDATE p95 was 0.042 ms, and
lock-wait p95 was 0.001 ms. This run did not reproduce the approximately 796
ms metadata tail from the original end-to-end report. That observation does
not establish a fix or eliminate environmental, WAL, filesystem, scheduler,
or cross-process causes.

The original end-to-end report remains referenced in `performance-reference.json`
without copying lifecycle or held-out quality artifacts. It recorded 33 warm
queries, a recall p95 of 1088.094 ms, and a failed 500 ms gate. Its p95 row
was query index 12, with 220.545 ms embedding, 63.409 ms vector SQL, 0.416 ms
lexical work, and 796.927 ms metadata update plus refresh. The end-to-end gate
therefore remains failed; this archive is a write-stage diagnostic only.

Source commit: `{SOURCE_COMMIT}`. The executed profiler helper and both JSON
reports are copied byte-for-byte. The disposable fixture at
`D:\\WindowsTemp\\mnemos-profile-retrieval-commit-v1` is intentionally not
archived.
"""
    (DEST / "README.md").write_text(readme, encoding="utf-8", newline="\n")
    (DEST / ".gitattributes").write_text(
        "* -text\n*.json -whitespace\n", encoding="utf-8", newline="\n"
    )
    files: dict[str, dict[str, object]] = {}
    for path in sorted(DEST.iterdir()):
        if path.name == "evidence-index.json":
            continue
        files[path.name] = {"bytes": path.stat().st_size, "sha256": sha256(path)}
    (DEST / "evidence-index.json").write_text(
        json.dumps({"files": files}, ensure_ascii=True, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
        newline="\n",
    )
    print(json.dumps({"archive": str(DEST), "files": files}, sort_keys=True))


if __name__ == "__main__":
    main()
