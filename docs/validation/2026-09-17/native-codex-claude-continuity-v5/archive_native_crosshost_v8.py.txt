def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def read_json(path: Path):
    return json.loads(path.read_text(encoding="utf-8"))


def selected_codex(home: Path, selected_path: Path, tools: tuple[str, ...]):
    selected = read_json(selected_path)
    if len(selected) != len(tools):
        raise AssertionError(f"{selected_path.name}: completion count differs")
    values = []
    for item, expected_tool in zip(selected, tools):
        source = (home / item["source"]).resolve()
        if not source.is_relative_to(home.resolve()):
            raise AssertionError("selected Codex source escaped its home")
        lines = source.read_bytes().splitlines(keepends=True)
        raw = lines[item["line"] - 1]
        if json.loads(raw) != item["record"]:
            raise AssertionError("selected Codex line changed")
        payload = item["record"]["payload"]
        invocation = payload["invocation"]
        if invocation.get("server") != "mnemos" or invocation.get("tool") != expected_tool:
            raise AssertionError("unexpected selected Codex tool")
        result = payload["result"].get("Ok")
        if not result or result.get("isError") is not False:
            raise AssertionError("selected Codex result was not successful")
        structured = json.loads(result["structuredContent"]["result"])
        content = json.loads(result["content"][0]["text"])
        if structured != content or structured.get("status") == "error" or "error" in structured:
            raise AssertionError("selected Codex payload is inconsistent")
        values.append(structured)
    return selected, values


def selected_claude(phase: str, report: dict):
    fixture = Path(report["fixture"])
    raw_path = fixture / f"native-{phase}-stdout.jsonl"
    if not raw_path.exists():
        return [], [], [], {"present": False}
    raw = raw_path.read_bytes()
    summary = report.get({
        "crosshost-read": "claude_read",
        "crosshost-correct": "claude_correction",
        "crosshost-forget": "claude_forget",
    }[phase], {})
    normalized_hash = sha(raw.replace(b"\r\n", b"\n"))
    summary_hash = summary.get("stdout_sha256")
    if summary_hash and normalized_hash != summary_hash:
        raise AssertionError(f"Claude {phase} stdout hash mismatch")
    selected, calls, results = [], [], []
    for number, line in enumerate(raw.splitlines(keepends=True), 1):
        record = json.loads(line)
        for node in record.get("message", {}).get("content", []):
            if not isinstance(node, dict) or node.get("type") not in ("tool_use", "tool_result"):
                continue
            selected.append({"phase": phase, "line": number, "line_sha256": sha(line), "node": node})
            (calls if node["type"] == "tool_use" else results).append(node)
    if [c.get("id") for c in calls] != [r.get("tool_use_id") for r in results]:
        raise AssertionError(f"Claude {phase} call/result IDs do not align")
    observations = summary.get("results", [])
    if observations and len(observations) != len(results):
        raise AssertionError(f"Claude {phase} result count differs")
    for result, observation in zip(results, observations):
        if result.get("is_error", False):
            raise AssertionError(f"Claude {phase} tool returned error")
        if observation.get("sha256") != sha(json.dumps(result, sort_keys=True, separators=(",", ":")).encode()):
            raise AssertionError(f"Claude {phase} selected result hash differs")
    decoded = []
    for result in results:
        value = json.loads(result['content'])
        if 'result' in value:
            value = json.loads(value['result'])
        assert value.get('status') != 'error' and 'error' not in value
        decoded.append(value)
    assert summary['returncode'] == 0 and summary['timed_out'] is False
    return selected, calls, decoded, {"present": True, "stdout_sha256": normalized_hash}


def table_rows(db, query, params=()):
    return [dict(row) for row in db.execute(query, params)]
