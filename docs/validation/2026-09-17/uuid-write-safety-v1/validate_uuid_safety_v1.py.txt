"""Run source-bound static and full-suite checks for UUID write safety repair."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys
from datetime import datetime, timezone

ROOT = Path(__file__).resolve().parents[1]
OUTPUT = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-uuid-safety-v1')
if OUTPUT.exists():
    raise FileExistsError(OUTPUT)
OUTPUT.mkdir()
sys.path.insert(0, str(ROOT))
import mnemos
assert Path(mnemos.__file__).resolve() == ROOT / 'mnemos/__init__.py'
paths = sorted((ROOT / 'mnemos').rglob('*.py')) + sorted((ROOT / 'tests').rglob('*.py'))
digest = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
initial = {p.relative_to(ROOT).as_posix(): digest(p) for p in paths}
patch = subprocess.check_output(['git', 'diff', '--binary', 'HEAD'], cwd=ROOT)
(OUTPUT / 'tracked-source.patch').write_bytes(patch)
report = {
    'status': 'running',
    'started_at': datetime.now(timezone.utc).isoformat(),
    'base_commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip(),
    'package_origin': str(Path(mnemos.__file__).resolve()),
    'source_files_sha256': initial,
    'runner_sha256': digest(Path(__file__)),
    'checks': [],
}
save = lambda: (OUTPUT / 'report.json').write_text(json.dumps(report, indent=2) + '\n', encoding='utf-8')
save()
python = str(ROOT / '.venv/Scripts/python.exe')
checks = [
    ('black', [python, '-m', 'black', '--check', 'mnemos', 'tests/test_adapter_context_authority.py', 'tests/test_cli_context_authority.py', 'tests/test_uuid_write_safety.py']),
    ('mypy', [str(ROOT / '.venv/compat-py310/Scripts/python.exe'), '-m', 'mypy', 'mnemos']),
    ('pytest', [python, '-m', 'pytest', '-q', 'tests', '--junitxml=' + str(OUTPUT / 'pytest.xml')]),
]
for name, command in checks:
    with (OUTPUT / (name + '.stdout.log')).open('wb') as out, (OUTPUT / (name + '.stderr.log')).open('wb') as err:
        completed = subprocess.run(command, cwd=ROOT, stdout=out, stderr=err)
    report['checks'].append({'name': name, 'command': command, 'returncode': completed.returncode})
    save()
    print(name, completed.returncode, flush=True)
    if completed.returncode:
        break
current_paths = sorted((ROOT / 'mnemos').rglob('*.py')) + sorted((ROOT / 'tests').rglob('*.py'))
final = {p.relative_to(ROOT).as_posix(): digest(p) for p in current_paths}
report['source_unchanged'] = initial == final
report['status'] = 'passed' if len(report['checks']) == len(checks) and all(c['returncode'] == 0 for c in report['checks']) and initial == final else 'failed'
report['finished_at'] = datetime.now(timezone.utc).isoformat()
save()
print(report['status'], str(OUTPUT), flush=True)
sys.exit(0 if report['status'] == 'passed' else 1)
