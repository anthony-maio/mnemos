"""

def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()

def decode_codex_file(home: Path, selected: list[dict], expected_tools: tuple[str, ...]) -> list[dict]:
    assert len(selected) == len(expected_tools)
    decoded = []
    for item, expected in zip(selected, expected_tools):
        path = (home / item["source"]).resolve()
        assert path.is_relative_to(home.resolve())
        raw_lines = path.read_bytes().splitlines(keepends=True)
        raw = raw_lines[item["line"] - 1]
        assert json.loads(raw) == item["record"]
        payload = item["record"]["payload"]
        assert payload["invocation"]["server"] == "mnemos"
        assert payload["invocation"]["tool"] == expected
        assert set(payload["result"]) == {"Ok"}
        result = payload["result"]["Ok"]
        assert result["isError"] is False
        value = json.loads(result["structuredContent"]["result"])
        assert value == json.loads(result["content"][0]["text"])
        assert value.get("status") != "error" and "error" not in value
        decoded.append(value)
    return decoded


def selected_claude(phase: str, report: dict) -> tuple[list[dict], list[dict], list[dict]]:
    fixture = Path(report["fixture"])
    raw = (fixture / f"native-{phase}-stdout.jsonl").read_bytes()
    summary = report["claude_read" if phase == "crosshost-read" else "claude_correction"]
    assert sha(raw.replace(b"\r\n", b"\n")) == summary["stdout_sha256"]
    calls, results, selected = [], [], []
    for number, line in enumerate(raw.splitlines(keepends=True), 1):
        record = json.loads(line)
        for node in record.get("message", {}).get("content", []):
            if not isinstance(node, dict) or node.get("type") not in ("tool_use", "tool_result"):
                continue
            selected.append({"phase": phase, "line": number, "line_sha256": sha(line), "node": node})
            (calls if node["type"] == "tool_use" else results).append(node)
    assert [c["id"] for c in calls] == [r["tool_use_id"] for r in results]
    assert len({c["id"] for c in calls}) == len(calls)
    assert len(results) == len(summary["results"])
    for result, observation in zip(results, summary["results"]):
        assert not result.get("is_error", False)
        assert sha(json.dumps(result, sort_keys=True, separators=(",", ":")).encode()) == observation["sha256"]
    decoded = []
    for result in results:
        value = json.loads(result["content"])
        if "result" in value:
            value = json.loads(value["result"])
        assert value.get("status") != "error" and "error" not in value
        decoded.append(value)
    assert summary["returncode"] == 0 and summary["timed_out"] is False
    return selected, calls, decoded
