"""Archive the verified Hermes reviewed-copy API checkpoint and its checks."""
from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
temp = Path("C:/Users/Ant/AppData/Local/Temp")
checks = temp / "mnemos-package-checks-hermes-reviewed-forget-v1"
installed = temp / "mnemos-installed-checks-hermes-reviewed-forget-v1"
output = repo / "docs/validation/2026-09-19/hermes-reviewed-forget-checkpoint-v1"
verification = json.loads((checks / "verification.json").read_text())
installation = json.loads((installed / "installation.json").read_text())
contract = json.loads((installed / "contract.json").read_text())
assert verification["status"] == installation["status"] == contract["status"] == "passed"
assert verification["source_commit"] == installation["bindings_after"]["commit"] == "db932befcd4d0aac9220f9f1e73f043ccd00093c"
assert all(row["exit_code"] == 0 for row in verification["checks"])

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

assert sha(checks / "verification.json") == installation["bindings_before"]["checks_sha256"]
assert sha(checks / "pytest.xml") == installation["successful_pytest_xml_sha256"]
for artifact in verification["artifacts"]:
    path = Path(artifact["file"])
    assert sha(path) == artifact["sha256"]
    assert path.stat().st_size == artifact["bytes"]
for command in installation["commands"]:
    for stream in ("stdout", "stderr"):
        assert sha(installed / (command["name"] + "." + stream + ".log")) == command[stream + "_sha256"]
assert sha(repo / ".venv/final_checkpoint_checks_v3.py") == verification["runner_sha256"]
assert sha(repo / ".venv/run_combined_installed_check_v35.py") == installation["runner_sha256"]
output.mkdir(exist_ok=False)
(output / ".gitattributes").write_bytes(b"** -text -whitespace\n")

def copy(source, relative):
    destination = output / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, destination)

for kind, folder in (("source-checks", checks), ("installed-checks", installed)):
    for path in folder.iterdir():
        if path.is_file() and path.suffix in {".json", ".xml", ".log"}:
            copy(path, Path(kind) / path.name)
for name in ("final_checkpoint_checks_v3.py", "run_combined_installed_check_v35.py", "installed_contract_probe_v4.py", "verify_claude_stop_replay_v2.py"):
    copy(repo / ".venv" / name, Path("helpers") / (name + ".txt"))
for name in ("hermes-related-tools-red-v1.stdout", "hermes-related-tools-red-v2.stdout", "hermes-related-tools-green-v8.stdout"):
    copy(repo / ".venv" / name, Path("development-checks") / (name + ".log"))
copy(Path(__file__), Path("helpers") / (Path(__file__).name + ".txt"))
files = {
    path.relative_to(output).as_posix(): {"bytes": path.stat().st_size, "sha256": sha(path)}
    for path in sorted(output.rglob("*")) if path.is_file()
}
(output / "evidence-index.json").write_text(json.dumps({"files": files}, indent=2) + "\n", encoding="utf-8")
s = verification["checks"][0]["counts"]
i = installation["contract"]["junit_summary"]
wheel = next(item for item in verification["artifacts"] if item["file"].endswith(".whl"))
(output / "README.md").write_text(f"""# Hermes reviewed-copy tool checkpoint

Frozen source `db932befcd4d0aac9220f9f1e73f043ccd00093c` passes
{s['tests'] - s['skipped']:,} source tests with {s['skipped']} skips, all six source/package checks,
{i['tests'] - i['skipped']:,} installed cases with {i['skipped']} matching skips, and seven replay assertions.
The isolated installed run checks each selected case and outcome, imported
module origins, and package bytes against the frozen source and built wheel.

Wheel: {wheel['bytes']:,} bytes; SHA-256
`{wheel['sha256']}`.

print(json.dumps({"archive": str(output), "indexed_files": len(files), "source": s, "installed": i, "wheel": wheel}))
