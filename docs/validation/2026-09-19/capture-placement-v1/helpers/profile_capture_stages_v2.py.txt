events = []
guard = threading.Lock()
local = threading.local()
originals = []


def record(name, started, error):
    item = {"stage": name, "capture_id": getattr(local, "capture_id", None),
            "thread_id": threading.get_ident(), "milliseconds": (time.perf_counter() - started) * 1000,
            "error_type": error}
    with guard:
        events.append(item)


def wrap_sync(owner, method, *, label=False):
    original = getattr(owner, method)
    name = owner.__name__ + "." + method
    @functools.wraps(original)
    def wrapped(*args, **kwargs):
        previous = getattr(local, "capture_id", None)
        if label:
            local.capture_id = args[1].event_id
        started = time.perf_counter()
        error = None
        try:
            return original(*args, **kwargs)
        except BaseException as exc:
            error = type(exc).__name__
            raise
        finally:
            record(name, started, error)
            local.capture_id = previous
    originals.append((owner, method, original))
    setattr(owner, method, wrapped)


def wrap_async(owner, method, *, label=False):
    original = getattr(owner, method)
    name = owner.__name__ + "." + method
    @functools.wraps(original)
    async def wrapped(*args, **kwargs):
        previous = getattr(local, "capture_id", None)
        if label:
            local.capture_id = args[1].get("event_id")
        started = time.perf_counter()
        error = None
        try:
            return await original(*args, **kwargs)
        except BaseException as exc:
            error = type(exc).__name__
            raise
        finally:
            record(name, started, error)
            local.capture_id = previous
    originals.append((owner, method, original))
    setattr(owner, method, wrapped)


for owner, method in (

    (client, "_protect_private_path"), (service, "_protect_private_path"),
    (os, "fsync"), (os, "replace"),
):
