"""Preserve one verified native-failure repair package and regression evidence."""
from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
temp = Path("C:/Users/Ant/AppData/Local/Temp")
checks = temp / "mnemos-package-checks-native-repairs-v1"
installed = temp / "mnemos-installed-checks-native-repairs-v1"
output = repo / "docs/validation/2026-09-19/native-repairs-checkpoint-v1"
verification = json.loads((checks / "verification.json").read_text())
installation = json.loads((installed / "installation.json").read_text())
contract = json.loads((installed / "contract.json").read_text())
assert verification["status"] == installation["status"] == contract["status"] == "passed"
assert verification["source_commit"] == installation["bindings_after"]["commit"] == "fd83fb27042c05897f29dba952b0924b15889aad"
assert all(row["exit_code"] == 0 for row in verification["checks"])

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

assert sha(checks / "verification.json") == installation["bindings_before"]["checks_sha256"]
assert sha(checks / "pytest.xml") == installation["successful_pytest_xml_sha256"]
for artifact in verification["artifacts"]:
    path = Path(artifact["file"])
    assert sha(path) == artifact["sha256"]
    assert path.stat().st_size == artifact["bytes"]
for command in installation["commands"]:
    for stream in ("stdout", "stderr"):
        assert sha(installed / (command["name"] + "." + stream + ".log")) == command[stream + "_sha256"]
assert sha(repo / ".venv/final_checkpoint_checks_v3.py") == verification["runner_sha256"]
assert sha(repo / ".venv/run_combined_installed_check_v34.py") == installation["runner_sha256"]
output.mkdir(exist_ok=False)
(output / ".gitattributes").write_bytes(b"** -text -whitespace\n")

def copy(source, relative):
    destination = output / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, destination)

for kind, folder in (("source-checks", checks), ("installed-checks", installed)):
    for path in folder.iterdir():
        if path.is_file() and path.suffix in {".json", ".xml", ".log"}:
            copy(path, Path(kind) / path.name)
for name in ("mnemos-fact-error-prior-regression-v1", "mnemos-bridge-prior-regression-v2", "mnemos-forgotten-id-prior-regression-v1"):
    for path in (temp / name).iterdir():
        if path.is_file() and path.suffix in {".json", ".xml", ".log"}:
            copy(path, Path("prior-source-regressions") / name / path.name)
for name in ("verify_fact_error_prior_regression_v1.py", "verify_bridge_prior_regression_v2.py", "verify_forgotten_id_prior_regression_v1.py", "final_checkpoint_checks_v3.py", "run_combined_installed_check_v34.py", "installed_contract_probe_v4.py", "verify_claude_stop_replay_v2.py"):
    copy(repo / ".venv" / name, Path("helpers") / (name + ".txt"))
copy(Path(__file__), Path("helpers") / (Path(__file__).name + ".txt"))
files = {
    path.relative_to(output).as_posix(): {"bytes": path.stat().st_size, "sha256": sha(path)}
    for path in sorted(output.rglob("*")) if path.is_file()
}
(output / "evidence-index.json").write_text(json.dumps({"files": files}, indent=2) + "\n", encoding="utf-8")
source_counts = verification["checks"][0]["counts"]
installed_counts = installation["contract"]["junit_summary"]
wheel = next(item for item in verification["artifacts"] if item["file"].endswith(".whl"))
(output / "README.md").write_text(f"""# Native failure repair package checkpoint

Frozen source `fd83fb27042c05897f29dba952b0924b15889aad` passes
{source_counts['tests'] - source_counts['skipped']:,} source tests with {source_counts['skipped']} skips, all six source/package checks,
{installed_counts['tests'] - installed_counts['skipped']:,} installed cases with {installed_counts['skipped']} matching skips, and seven installed replay
assertions. The installed run checks each selected case/outcome, package bytes,
and imported module origins against the frozen source and built wheel.

Wheel: {wheel['bytes']:,} bytes; SHA-256
`{wheel['sha256']}`.

## Repairs

- The bridge writes every protocol response as explicit UTF-8 bytes. Real
  subprocess/service tests cover cp1252, ASCII, and UTF-16 text-stream settings,
  Unicode content, oversized requests, and malformed requests.
- Inspect, correct, and forget return the same `404/fact_not_found` for absent,
  deleted, and inaccessible fact IDs. Direct callers retain KeyError compatibility.
  Unexpected internal KeyError still returns a generic 500. This change does
  not alter the separate related-forget error contract.
- Automatic assistant captures quoting an explicitly forgotten fact ID are
  suppressed before persistence. Atomic fact commit rechecks source and candidate
  text to prevent a capture-before-forget/commit-after-forget race from producing
  new facts. Batched exact-token lookups use a non-destructive SQLite index.

The end-to-end fixture teaches, corrects, forgets, delivers delayed assistant
narration, reopens the database, and replays that narration. Separate project
and profile controls survive; explicit reteaching creates a new fact ID.

## Limits and prior evidence

This is a bounded exact-ID safeguard. IDs composed of ASCII letters, digits,
underscores, and hyphens, including native UUIDs, are supported. An assistant
event mixing an erased ID with unrelated notes is conservatively suppressed
as a whole. User captures and explicit writes remain eligible. A source already
captured before forgetting can remain in raw storage even when its later
fact extraction is suppressed. This does not prove arbitrary paraphrase
erasure, general raw-source erasure, or native-model reliability.

The [v4 native cohort](../hermes-native-cohort-v4/README.md) remains zero of
three full lifecycle passes on its earlier `8f8f63f` runtime. This package has
not had a new native-model lifecycle run. Model correction omission, complete
capture, three-host continuity, module-quality acceptance, and the pilot remain
open.

The preserved pre-fix `8f8f63f` source was checked retrospectively with the new
regressions: six fact-error assertions fail (three controls pass); four bridge
cases fail (two pass); the delayed-narration lifecycle fails because the old
source accepts the capture as pending. Reports bind imported paths, old source,
test hashes, commands, and outcomes. These archived runs are post-fix regression
proof, not a claim that every behavioral failure was recorded before coding.

The evidence index binds {len(files)} files: source/installed checks, regression
reports, and exact executed helpers. This README is outside the raw index.
Live host configuration and live memory data were unchanged.
""", encoding="utf-8")
print(json.dumps({"archive": str(output), "indexed_files": len(files)}))
