from agent.memory_manager import MemoryManager
from mnemos.adapters.hermes import MnemosHermesProvider
from mnemos.client import Outbox

class LocalOutboxClient:
    def __init__(self, path):
        self.outbox = Outbox(path)
        self.calls = []
    async def capture(self, payload):
        payload = self.outbox.sanitize(payload)
        event_id = self.outbox.append(payload, event_id=payload.get("event_id"))
        self.calls.append({"entry_event_id": event_id, "payload": payload})
        self.outbox.acknowledge(event_id)
        return {"schema_version": 1, "status": "committed", "committed": True, "event_id": event_id}

client = LocalOutboxClient(root / "outbox.json")
provider = MnemosHermesProvider(client=client)
manager = MemoryManager(); manager.add_provider(provider)
manager.initialize_all("rewind-diagnostic", cwd=cfg["project"], hermes_home=cfg["home"])

def flush():
    assert manager.flush_pending(timeout=10)
    assert provider._flush(timeout=10)

def snapshot(label):
    rows = client.calls[:]
    return {"label": label, "calls": len(rows), "events": [
        {"event_id": r["payload"].get("event_id"), "occurred_at": r["payload"].get("occurred_at"),
         "role": r["payload"].get("role"), "source_ref": r["payload"].get("source_ref")}
        for r in rows
    ]}

manager.sync_all(messages[0]["content"], messages[1]["content"], session_id="rewind-diagnostic", messages=messages); flush()
