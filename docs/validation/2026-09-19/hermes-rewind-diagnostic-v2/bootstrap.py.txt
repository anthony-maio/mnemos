class LocalOutboxClient:
    def __init__(self, path):
        self.outbox = Outbox(path)
        self.calls = []
        self.phase = "startup"
    async def capture(self, payload):
        payload = self.outbox.sanitize(payload)
        transport_id = self.outbox.append(payload, event_id=payload.get("event_id"))
        self.calls.append({"phase": self.phase, "transport_event_id": transport_id, "payload": payload})
        return {"schema_version": 1, "status": "queued", "committed": False, "degraded": True, "event_id": transport_id}

def snapshot(label, before_calls):
    pending = client.outbox.pending()
    return {"label": label, "request_count_delta": len(client.calls) - before_calls,
            "pending_entry_count": len(pending),
            "pending_payload_event_ids": sorted(item["payload"].get("event_id") for item in pending),
            "calls": [{"phase": item["phase"], "payload_event_id": item["payload"].get("event_id"),
                       "occurred_at": item["payload"].get("occurred_at"), "role": item["payload"].get("role"),
                       "source_ref": item["payload"].get("source_ref")} for item in client.calls[before_calls:]]}
