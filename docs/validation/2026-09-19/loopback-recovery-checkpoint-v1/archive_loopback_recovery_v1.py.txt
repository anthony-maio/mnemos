"""Preserve failed and repaired recovery evidence with cognition setup measurements."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import shutil

ROOT = Path(__file__).resolve().parent.parent
TEMP = Path('C:/Users/Ant/AppData/Local/Temp')
DEST = ROOT / 'docs/validation/2026-09-19/loopback-recovery-checkpoint-v1'
COMMIT = 'de6cdcf1c05fdad1bb19ed8f52638eff5ed67ae4'

INSTALLED = TEMP / 'mnemos-installed-checks-loopback-recovery-v2'

def read(path):
    return json.loads(path.read_text(encoding='utf-8'))

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

checks = read(CHECKS / 'verification.json')
installed = read(INSTALLED / 'installation.json')
failed = read(TEMP / 'mnemos-package-checks-cognition-context-v1/verification.json')
before = read(TEMP / 'mnemos-recovery-readiness-v1/report.json')
after = read(TEMP / 'mnemos-recovery-readiness-v2/report.json')
cognition = read(TEMP / 'mnemos-cognition-context-comparison-v1/report.json')
assert checks['status'] == installed['status'] == 'passed'
assert checks['source_commit'] == installed['commit'] == COMMIT
assert after['source_commit'] == 'a6c275bc052a83eefa088452d54090c2f0e80675'
assert sha(TEMP / 'mnemos-source-loopback-recovery-v1/mnemos/client.py') == sha(TEMP / 'mnemos-source-loopback-recovery-v2/mnemos/client.py')
assert len(checks['checks']) == 6 and all(c['status'] == 'passed' for c in checks['checks'])
assert failed['status'] == 'failed' and failed['checks'][0]['counts']['failures'] == 1
assert failed['source_commit'] == before['source_commit'] == cognition['expected_commits']['candidate']
assert [r['exit_code'] for r in before['runs']] == [1, 1, 0, 0, 0, 1]
assert [r['exit_code'] for r in after['runs']] == [0] * 6
assert cognition['status'] == 'completed'
for arm in cognition['arms']:
    assert arm['source_tree_sha256_before'] == arm['source_tree_sha256_after']
    assert arm['server']['accepted_connections'] == arm['server']['finished_connections'] == 13
    assert not arm['server']['invalid_requests']
helpers = ['final_checkpoint_checks_v3.py', 'run_combined_installed_check_v27.py',
           'installed_contract_probe_v4.py', 'verify_claude_stop_replay_v2.py',
           'profile_recovery_readiness_v1.py', 'profile_recovery_readiness_v2.py',
           'measure_cognition_context_reuse_v1.py', 'profile_cognition_setup_v1.py', Path(__file__).name]
for name, recorded in [('run_combined_installed_check_v27.py', installed['runner_sha256']),
                       ('profile_recovery_readiness_v1.py', before['helper_sha256']),
                       ('profile_recovery_readiness_v2.py', after['helper_sha256']),
                       ('measure_cognition_context_reuse_v1.py', cognition['helper_sha256'])]:
    assert sha(ROOT / '.venv' / name) == recorded
DEST.mkdir(exist_ok=False)

def copy(source, target):
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, target)
    assert sha(source) == sha(target)

for label, source in [('source-checks', CHECKS), ('installed-checks', INSTALLED),
                      ('failed-source-checks', TEMP / 'mnemos-package-checks-cognition-context-v1'),
                      ('failed-sharing-source-checks', TEMP / 'mnemos-package-checks-loopback-recovery-v1'),
                      ('cognition-comparison', TEMP / 'mnemos-cognition-context-comparison-v1'),
                      ('cognition-prototype', TEMP / 'mnemos-cognition-setup-v1')]:
    for file in source.iterdir():
        if file.is_file():
            copy(file, DEST / label / file.name)
for version in (1, 2):
    source = TEMP / f'mnemos-recovery-readiness-v{version}'
    for file in source.rglob('*'):
        if file.is_file() and ('fixtures' not in file.relative_to(source).parts):
            copy(file, DEST / f'readiness-v{version}' / file.relative_to(source))
for name in helpers:
    copy(ROOT / '.venv' / name, DEST / (name + '.txt'))
counts = checks['checks'][0]['counts']
outcomes = read(INSTALLED / 'contract.json')['case_summary']['actual_outcomes']
assert len(installed['probe']['assertions']) == 7 and all(installed['probe']['assertions'].values())
binding = {'source_commit': COMMIT, 'runtime_repair_commits': ['93a109a', 'de6cdcf'],
           'archived_at': datetime.now(timezone.utc).isoformat(),
           'focused_tests': {'loopback': {'passed': 51, 'skipped': 2}, 'windows_sharing': {'passed': 44, 'skipped': 0}},
           'regression_red': 'Three failures: certificate loading in both routes and premature failure after competing launcher exit.',
           'extra_check_note': 'Explicitly checking the normally excluded test file with mypy produced private-import export errors, including existing subprocess references. The configured full-project mypy check is recorded separately.',
           'source_verification': str(CHECKS / 'verification.json'),
           'installed_verification': str(INSTALLED / 'installation.json'),
           'limitations': 'Synthetic transport and recovery evidence, not end-to-end model latency, host capture, or lifecycle quality acceptance.'}
(DEST / 'checkpoint-binding.json').write_text(json.dumps(binding, indent=2) + '\n', encoding='utf-8')
