from pathlib import Path
import hashlib
import json

repo = Path(__file__).resolve().parent.parent
temporary = Path("C:/Users/Ant/AppData/Local/Temp")
output = repo / "docs/validation/2026-09-19/hermes-continuity-checkpoint-v1"
assert not output.exists()
checks = temporary / "mnemos-package-checks-hermes-continuity-v1"
installed = temporary / "mnemos-installed-checks-hermes-continuity-v1"
verification = json.loads((checks / "verification.json").read_text())
installation = json.loads((installed / "installation.json").read_text())
assert verification["status"] == installation["status"] == "passed"
assert verification["source_commit"] == installation["commit"] == "78c63ed36942848c95c1acd93b805fe0e0d13a67"
output.mkdir()
for source, label in ((checks, "source-checks"), (installed, "installed-checks")):
    dest = output / label
    dest.mkdir()
    for path in source.iterdir():
        if path.is_file() and path.suffix in {".json", ".log", ".xml"}:
            (dest / path.name).write_bytes(path.read_bytes())
(output / "prefetch-stale-red.log").write_bytes((temporary / "mnemos-prefetch-stale-red-v1/result.log").read_bytes())
(output / "prefetch-stale-red-test.py.txt").write_bytes((temporary / "mnemos-prefetch-stale-red-v1/test_hermes_adapter.py").read_bytes())
helpers = output / "helpers"
helpers.mkdir()
for name in ("final_checkpoint_checks_v3.py", "run_combined_installed_check_v27.py", "installed_contract_probe_v4.py", "verify_claude_stop_replay_v2.py", Path(__file__).name):
    (helpers / (name + ".txt")).write_bytes((repo / ".venv" / name).read_bytes())
(output / ".gitattributes").write_bytes(b"* -text -whitespace\n")
index = {"files": {}}
for path in sorted(output.rglob("*")):
    if path.is_file():
        raw = path.read_bytes()
        index["files"][path.relative_to(output).as_posix()] = {"bytes": len(raw), "sha256": hashlib.sha256(raw).hexdigest()}
(output / "evidence-index.json").write_bytes((json.dumps(index, indent=2) + "\n").encode())
print(len(index["files"]), "archived files")
