"""Preserve source and installed evidence for the narrow source-selection error repair."""
from pathlib import Path
import hashlib
import json
import shutil

repo = Path(__file__).resolve().parents[1]
temp = Path('C:/Users/Ant/AppData/Local/Temp')
checks = temp / 'mnemos-package-checks-related-forget-error-v1'
installed = temp / 'mnemos-installed-checks-related-forget-error-v1'
output = repo / 'docs/validation/2026-09-19/related-forget-error-checkpoint-v1'
verification = json.loads((checks / 'verification.json').read_text())
installation = json.loads((installed / 'installation.json').read_text())
contract = json.loads((installed / 'contract.json').read_text())
assert verification['status'] == installation['status'] == contract['status'] == 'passed'
commit = verification['source_commit']
assert commit == installation['bindings_after']['commit']
assert all(c['exit_code'] == 0 for c in verification['checks'])

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

assert sha(checks / 'verification.json') == installation['bindings_before']['checks_sha256']
assert sha(checks / 'pytest.xml') == installation['successful_pytest_xml_sha256']
for artifact in verification['artifacts']:
    p = Path(artifact['file'])
    assert sha(p) == artifact['sha256'] and p.stat().st_size == artifact['bytes']
for command in installation['commands']:
    for stream in ('stdout', 'stderr'):
        assert sha(installed / (command['name'] + '.' + stream + '.log')) == command[stream + '_sha256']
assert sha(repo / '.venv/final_checkpoint_checks_v3.py') == verification['runner_sha256']
assert sha(repo / '.venv/run_combined_installed_check_v36.py') == installation['runner_sha256']
output.mkdir(exist_ok=False)
(output / '.gitattributes').write_bytes(b'** -text -whitespace\n')

def copy(path, relative):
    target = output / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(path, target)

for kind, folder in (('source-checks', checks), ('installed-checks', installed)):
    for p in folder.iterdir():
        if p.is_file() and p.suffix in {'.json', '.xml', '.log'}:
            copy(p, Path(kind) / p.name)
for name in ('final_checkpoint_checks_v3.py', 'run_combined_installed_check_v36.py', 'installed_contract_probe_v4.py', 'verify_claude_stop_replay_v2.py'):
    copy(repo / '.venv' / name, Path('helpers') / (name + '.txt'))
for p in sorted((repo / '.venv').glob('related-forget-service-errors-*.stdout')):
    copy(p, Path('development-checks') / (p.name + '.log'))
copy(Path(__file__), Path('helpers') / (Path(__file__).name + '.txt'))
files = {p.relative_to(output).as_posix(): {'bytes': p.stat().st_size, 'sha256': sha(p)}
         for p in sorted(output.rglob('*')) if p.is_file()}
(output / 'evidence-index.json').write_text(json.dumps({'files': files}, indent=2) + '\n', encoding='utf-8')
s = verification['checks'][0]['counts']
i = installation['contract']['junit_summary']
wheel = next(a for a in verification['artifacts'] if a['file'].endswith('.whl'))
(output / 'README.md').write_text(f"""# Related-forget source-selection error checkpoint

Frozen source `{commit}` passes {s['tests'] - s['skipped']:,} source tests
with {s['skipped']} skips, all six source/package checks, {i['tests'] - i['skipped']:,} installed
cases with {i['skipped']} matching skips, and seven replay assertions. The isolated
installed run checks individual case outcomes, module origins, and package
bytes against the frozen source and wheel.

Wheel: {wheel['bytes']:,} bytes; SHA-256
`{wheel['sha256']}`.

## Observed defect and narrow repair

The [native v6 cohort](../hermes-native-cohort-v6/README.md) failed all three
attempts. In attempt 3, the model selected additional facts but omitted their
required source-event IDs. Storage correctly rejected the operation and
rolled back, but its ordinary ValueError became HTTP 500 at the service.

That specific domain rejection now has a typed ValueError subclass. The
authenticated service returns HTTP 409, status rejected, committed false,
and a fixed source-selection error code and message. The client accepts only
the strict known contract for forget_related and returns it without replaying
the outbox. Hermes receives the actionable rejection. The change neither
selects sources for the model nor retries the invalid request.

The original storage error text remains compatible. Unexpected ValueError
exceptions, malformed or unknown HTTP 409 bodies, and unrelated operations
keep their prior generic failure behavior. No internal exception text,
memory content, or selected IDs are added to the rejection response. Other
related-forget validation errors are outside this narrow classification fix.

Regression evidence covers the real service/client/Hermes path, transaction
rollback, explicit valid source selection, safe error handling, and outbox
preservation. The red-v1 development log contains one genuine missing-409
regression plus an unrelated test assumption error: the old success schema
does not include a committed boolean. The latter assertion was corrected to
use the existing status and forgotten fields. Do not count it as a product
defect. The green-v3 log records a collection error from a nonexistent test
path, not a test result; green-v4 uses the actual test paths. Frozen source
and installed reports establish the final results.

This is fixture, service, and installed-package evidence. No native model
has run this new package; the v6 failures remain unchanged. Selecting all
requested conversational copies, cross-host continuity, complete capture,
model-quality acceptance, Desktop, and the pilot remain open. Live host
configuration and live memory were unchanged.

The evidence index binds {len(files)} raw files. This interpretive README is
outside the raw index.
""", encoding='utf-8')
print(json.dumps({'archive': str(output), 'indexed_files': len(files), 'source': s, 'installed': i, 'wheel': wheel}))
