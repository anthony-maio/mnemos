def save(name, value):
    (ROOT / name).write_text(json.dumps(value, indent=2, default=str) + "\n", encoding="utf-8")

def wsl(path):
    value = str(path).replace("\\", "/")
    return "/mnt/" + value[0].lower() + value[2:]

def api(route, payload=None, timeout=15):
    data = None if payload is None else json.dumps(payload).encode()
    request = urllib.request.Request("http://127.0.0.1:11434/api/" + route, data=data, headers={"Content-Type": "application/json"})
    with urllib.request.urlopen(request, timeout=timeout) as response:
        return json.load(response)

        from mnemos.config import MnemosConfig, ModuleConfig
        from mnemos.engine import MnemosEngine
        from mnemos.memory_service import LocalMemoryService
        from mnemos.scopes import ScopeRegistry
        from mnemos.service import MnemosService
        from mnemos.utils.embeddings import OllamaEmbeddingProvider
        from mnemos.utils.llm import UnavailableLLMProvider
        from mnemos.utils.storage import SQLiteStore

        registry = ScopeRegistry()
        contexts = {}
        for label in ("project", "foreign-project"):
            directory = ROOT / label
            directory.mkdir()
            project = registry.register_project(directory, user_profile_id="native-fixture", enabled=True)
            project.aliases.append(wsl(directory))
            contexts[label] = registry.resolve(directory, user_profile_id="native-fixture", session_id="native-seed", host="manual").model_dump(mode="json")
        registry.save(ROOT / "scopes.json")
        config = MnemosConfig(modules=ModuleConfig(**{name: False for name in ModuleConfig.model_fields}))
        config.surprisal.embedding_dim = 768
        def factory():
            embedder = OllamaEmbeddingProvider(base_url="http://127.0.0.1:11434", timeout=30)
            engine = MnemosEngine(config=config, llm=UnavailableLLMProvider(), embedder=embedder,
                                  store=SQLiteStore(str(ROOT / "memory.sqlite")), scope_registry=registry,
                                  require_scope_context=True, owns_providers=True)
            return LocalMemoryService(engine, registry)
        service = MnemosService(dispatcher_factory=factory, descriptor_path=ROOT / "runtime.json", startup_timeout=10, dispatch_timeout=40)
        service.start()
        client = MnemosClient(descriptor_path=ROOT / "runtime.json", outbox_path=ROOT / "outbox.json", auto_start=False, request_timeout=40)

        }
        report["status"] = "passed" if all(report["checks"].values()) else "failed"
    except BaseException as exc:

    finally:

            service.stop()
        report["finished_at"] = datetime.now(timezone.utc).isoformat()
