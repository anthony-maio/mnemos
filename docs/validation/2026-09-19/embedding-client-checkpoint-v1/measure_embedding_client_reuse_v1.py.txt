"""Compare client setup against a disposable loopback fixture, without inference."""
import argparse
from datetime import datetime, timezone
import hashlib
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
import json
import os
from pathlib import Path
import statistics
import subprocess
import sys
import threading
import time


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def worker(source):
    sys.path.insert(0, str(source))
    import httpx
    from mnemos.utils.embeddings import OllamaEmbeddingProvider

    class Handler(BaseHTTPRequestHandler):
        protocol_version = "HTTP/1.0"

        def do_POST(self):
            data = json.loads(self.rfile.read(int(self.headers["Content-Length"])))
            assert self.path == "/api/embed"
            assert data["model"] == "fixture"
            response = json.dumps({"embeddings": [[1.0, 0.0, 0.0]]}).encode()
            self.send_response(200)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(response)))
            self.send_header("Connection", "close")
            self.end_headers()
            self.wfile.write(response)

        def log_message(self, *args):
            pass

    before = sha(source / "mnemos/utils/embeddings.py")
    server = ThreadingHTTPServer(("127.0.0.1", 0), Handler)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    provider = OllamaEmbeddingProvider(model="fixture", base_url=f"http://127.0.0.1:{server.server_port}")
    values = []
    try:
        for _ in range(13):
            start = time.perf_counter()
            assert provider.embed("constant local fixture") == [1.0, 0.0, 0.0]
            values.append((time.perf_counter() - start) * 1000)
    finally:
        close = getattr(provider, "close", None)
        if callable(close):
            close()
        server.shutdown()
        server.server_close()
        thread.join()
    assert sha(source / "mnemos/utils/embeddings.py") == before
    print(json.dumps({
        "source": str(source), "embedding_source_sha256": before,
        "source_commit": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=source, text=True).strip(),
        "source_dirty": bool(subprocess.check_output(["git", "diff", "--name-only"], cwd=source, text=True).strip()),
        "httpx_version": httpx.__version__, "cold_ms": values[0], "warm_ms": values[1:],
        "warm_median_ms": statistics.median(values[1:]), "warm_max_ms": max(values[1:]),
    }))


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--worker", type=Path)
    parser.add_argument("--baseline", type=Path)
    parser.add_argument("--candidate", type=Path)
    parser.add_argument("--output", type=Path)
    args = parser.parse_args()
    if args.worker:
        worker(args.worker.resolve())
        return
    assert args.baseline and args.candidate and args.output
    assert not args.output.exists()
    env = {k: v for k, v in os.environ.items() if not k.startswith("MNEMOS_") and k != "PYTHONPATH"}
    arms = []
    for name, source in (("baseline", args.baseline), ("candidate", args.candidate)):
        result = subprocess.run([sys.executable, str(Path(__file__).resolve()), "--worker", str(source)],
                                cwd=source, env=env, capture_output=True, text=True, timeout=120)
        assert result.returncode == 0, (result.stdout, result.stderr)
        arms.append({"name": name, **json.loads(result.stdout)})
    record = {"scope": "Disposable HTTP/1.0 loopback fixture; no Ollama calls or model inference; not a recall/capture acceptance result",
              "observed_at": datetime.now(timezone.utc).isoformat(), "samples": "1 cold plus 12 warm per arm",
              "helper_sha256": sha(Path(__file__)), "arms": arms}
    with args.output.open("x", encoding="utf-8") as output:
        output.write(json.dumps(record, indent=2) + "\n")
    print(json.dumps(record))


if __name__ == "__main__":
    main()
