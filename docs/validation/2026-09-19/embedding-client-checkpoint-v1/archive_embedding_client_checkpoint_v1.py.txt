"""Preserve checked source/package evidence and separate negative benchmarks."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import shutil

ROOT = Path(__file__).resolve().parents[1]
TEMP = Path("C:/Users/Ant/AppData/Local/Temp")
SOURCE = TEMP / "mnemos-package-checks-embedding-client-v2"
INSTALLED = TEMP / "mnemos-installed-checks-embedding-client-v1"
OUTPUT = ROOT / "docs/validation/2026-09-19/embedding-client-checkpoint-v1"
COMMIT = "39b9c799020a58ea2ac3d798189474ed10389bd0"


def sha(data):
    return hashlib.sha256(data).hexdigest()


source = json.loads((SOURCE / "verification.json").read_text())
installed = json.loads((INSTALLED / "installation.json").read_text())
assert source["source_commit"] == COMMIT and source["status"] == "passed"
assert {row["name"] for row in source["checks"]} == {"pytest", "black", "mypy", "javascript", "build", "twine"}
assert all(row["exit_code"] == 0 for row in source["checks"])
assert installed["status"] == "passed"
assert installed["bindings_after"]["commit"] == COMMIT
assert installed["bindings_before"] == installed["bindings_after"]
for artifact in source["artifacts"]:
    assert sha(Path(artifact["file"]).read_bytes()) == artifact["sha256"]
diagnostic = json.loads((TEMP / "mnemos-embedding-client-diagnostic-v1.json").read_text())
assert diagnostic["arms"][1]["source_commit"] == COMMIT
assert not any(arm["source_dirty"] for arm in diagnostic["arms"])
OUTPUT.mkdir(exist_ok=False)
for name, directory in (("source-checks", SOURCE), ("installed-checks", INSTALLED)):
    target = OUTPUT / name
    target.mkdir()
    for path in directory.iterdir():
        if path.is_file() and path.suffix in {".json", ".log", ".xml"}:
            shutil.copyfile(path, target / path.name)
files = {
    "client-reuse-diagnostic.json": TEMP / "mnemos-embedding-client-diagnostic-v1.json",
    "benchmark-candidate.json": TEMP / "mnemos-benchmark-gate-embedding-client-v2.json",
    "benchmark-candidate.log": TEMP / "mnemos-benchmark-gate-embedding-client-v2.log",
    "benchmark-baseline.json": TEMP / "mnemos-benchmark-gate-baseline-v1.json",
    "benchmark-baseline.log": TEMP / "mnemos-benchmark-gate-baseline-v1.log",
    "benchmark-inherited-config-failure.log": TEMP / "mnemos-benchmark-gate-embedding-client-v1.log",
}
for name, path in files.items():
    shutil.copyfile(path, OUTPUT / name)
for name in ("final_checkpoint_checks_v3.py", "run_combined_installed_check_v25.py",
             "installed_contract_probe_v4.py", "verify_claude_stop_replay_v2.py",
             "measure_embedding_client_reuse_v1.py", "archive_embedding_client_checkpoint_v1.py"):
    shutil.copyfile(ROOT / ".venv" / name, OUTPUT / (name + ".txt"))
baseline = json.loads((OUTPUT / "benchmark-baseline.json").read_text())
candidate = json.loads((OUTPUT / "benchmark-candidate.json").read_text())
assert baseline["summary"]["failed_runs"] == candidate["summary"]["failed_runs"] == 2
record = {
    "archived_at": datetime.now(timezone.utc).isoformat(), "source_commit": COMMIT,
    "baseline_commit": "fc72cc251aeafa161a1a390269b8dfa34d83cf2b",
    "source_verification": str(SOURCE / "verification.json"),
    "installed_verification": str(INSTALLED / "installation.json"),
    "benchmark_exit_codes": {"inherited_config_attempt": 1, "isolated_candidate": 2, "isolated_baseline": 2},
    "benchmark_args": ["--stores", "memory,sqlite", "--retrievers", "baseline,engine", "--dataset-pack", "claim-driving", "--top-k", "1", "--repetitions", "2", "--enforce-production-gate"],
    "benchmark_environment": {"MNEMOS_CONFIG_PATH": str(TEMP / "mnemos-offline-benchmark-missing.toml"), "MNEMOS_EMBEDDING_PROVIDER": "simple"},
    "source_preflight_failure": {"exit_code": 1, "message": "Verification snapshot must be clean", "detail": "First verification started while worktree checkout was still running. It rejected the incomplete snapshot before any check. After checkout exited zero and the snapshot was confirmed clean, verification used fresh v2 output and temp directories."},
    "helper_sha256": sha(Path(__file__).read_bytes()),
}
(OUTPUT / "checkpoint-binding.json").write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
counts = next(row["counts"] for row in source["checks"] if row["name"] == "pytest")
source_passes = counts["tests"] - counts["skipped"] - counts["failures"] - counts["errors"]
before, after = (arm["warm_median_ms"] for arm in diagnostic["arms"])
(OUTPUT / "README.md").write_text(f"""# Embedding client reuse and shutdown checkpoint

(OUTPUT / ".gitattributes").write_text("* -text\n*.log -whitespace\n*.txt -whitespace\n", encoding="utf-8")

         for p in OUTPUT.rglob("*") if p.is_file()}
(OUTPUT / "evidence-index.json").write_text(json.dumps({"files": index}, indent=2) + "\n", encoding="utf-8")
print(json.dumps({"archive": str(OUTPUT), "files": len(index), "source_passes": source_passes, "installed_summary": installed["contract"]["junit_summary"]}))
