"""Offline prospective degree diagnostic on saved development vectors only."""
from pathlib import Path
import hashlib
import importlib.util
import json
import time
import traceback

repo=Path(__file__).resolve().parent.parent
temp=Path('C:/Users/Ant/AppData/Local/Temp')
snapshot=temp/'mnemos-source-hermes-provisional-v1'
fixture=temp/'mnemos-real-model-local-v1/mechanism-development/frozen-vectors.json'
manifest=repo/'docs/validation/2026-09-19/real-model-local-v1/mechanism-development/manifest.json'
protocol=repo/'docs/plans/2026-09-19-graph-degree-development-protocol-v1.md'
work=temp/'mnemos-graph-degree-development-v1'
output=repo/'docs/validation/2026-09-19/graph-degree-development-v1'
assert not work.exists() and not output.exists()
sha=lambda path:hashlib.sha256(path.read_bytes()).hexdigest()
prior=json.loads(manifest.read_text())
assert sha(fixture)==prior['frozen_vectors_sha256']=='938537dc4b615358a36c748a31667d477a6aa556f7e62137ed281a08caaa7698'
assert prior['split']=='development' and prior['status']=='complete'
spec=importlib.util.spec_from_file_location('pinned_preflight',repo/'.venv/graph_eligibility_preflight_v3.py')
preflight=importlib.util.module_from_spec(spec)
spec.loader.exec_module(preflight)
commit='5e4b057b16906c2ec7dd302ae80748ef1ac5eddc'
verified=preflight._verify_snapshot(snapshot,commit)
measurement,mechanism,imported=preflight._import_snapshot(snapshot)
dataset=mechanism.load_mechanism_dataset(fixture)
assert len(dataset.graph_cases)==5 and all(c.split=='development' for c in dataset.graph_cases)
output.mkdir(); work.mkdir()
(output/'.gitattributes').write_bytes(b'* -text -whitespace\n')
original_factory=mechanism._engine_for
report={'status':'running','source':verified,'imported':imported,'source_commit':commit,
        'fixture_sha256':sha(fixture),'manifest_sha256':sha(manifest),
        'helper_sha256':sha(Path(__file__)),'protocol_sha256':sha(protocol),
        'preflight_helper_sha256':sha(repo/'.venv/graph_eligibility_preflight_v3.py'),
        'model_calls':0,'network_calls':0,'budgets':[4,8,16],'observations':[],
        'effective_configurations':{},'scope':'development only; one template; no production selection'}
def save():
    (output/'report.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
try:
    for degree in report['budgets']:
        def factory(*args, **kwargs):
            engine,store=original_factory(*args,**kwargs)
            assert engine.spreading_activation._config is engine.config.spreading
            engine.config.spreading.max_neighbors_per_node=degree
            report['effective_configurations'][str(degree)]=engine.config.model_dump(mode='json')
            return engine,store
        mechanism._engine_for=factory
        for case in sorted(dataset.graph_cases,key=lambda c:c.id):
            started=time.monotonic()
            row=measurement.measure_graph_repetition(case,work_dir=work/f'degree-{degree}',repetition=0)
            (output/f'degree-{degree}-{case.id}.json').write_text(json.dumps(row,indent=2)+'\n',encoding='utf-8')
            p=row['production']
            item={'degree':degree,'case_id':case.id,'template_id':case.template_id,
                  'eligible':row['eligible'],'success':row['success'],
                  'selected_seed':row['actual_selected_seed_id'],
                  'target_id':case.target_id,'target_in_direct_pool':row['target_in_direct_pool'],
                  'target_in_graph_results':row['target_in_graph_results'],
                  'target_rank':p['graph_result_ids'].index(case.target_id)+1 if case.target_id in p['graph_result_ids'] else None,
                  'baseline_result_ids':p['baseline_result_ids'],'graph_result_ids':p['graph_result_ids'],
                  'structural_valid_path':p['structural_valid_path'],
                  'activation_valid_path':p['activation_valid_path'],
                  'target_energy':p['target_energy'],'edge_count':p['graph_edge_count'],
                  'elapsed_seconds':time.monotonic()-started,
                  'raw_file':f'degree-{degree}-{case.id}.json'}
            report['observations'].append(item)
            save()
            controls=row['correctness_controls']
            assert controls['activation_vs_independent_path']['matches']
            assert controls['insertion_order']['edges_equal'] and controls['insertion_order']['results_equal']
            assert not p['scope_control_ids_forbidden_observed']
            assert p['target_absent_after_edge_cut']
    report['summary']={str(degree):{
        'cases':sum(r['degree']==degree for r in report['observations']),
        'eligible':sum(r['degree']==degree and r['eligible'] for r in report['observations']),
        'target_returned':sum(r['degree']==degree and r['target_in_graph_results'] for r in report['observations'])}
        for degree in report['budgets']}
    assert preflight._verify_snapshot(snapshot,commit)==verified
    report['status']='diagnostic_complete'
except BaseException as exc:
    report['status']='failed'
    report['error']=str(exc)
    (output/'failure.txt').write_text(traceback.format_exc(),encoding='utf-8')
finally:
    mechanism._engine_for=original_factory
    save()
    for source,name in [(Path(__file__),'graph_degree_development_v1.py.txt'),
                        (repo/'.venv/graph_eligibility_preflight_v3.py','graph_eligibility_preflight_v3.py.txt'),
                        (protocol,'protocol.md.txt'),(manifest,'development-manifest.json')]:
        (output/name).write_bytes(source.read_bytes())
    files={p.name:{'sha256':sha(p),'bytes':p.stat().st_size} for p in output.iterdir() if p.is_file()}
    (output/'evidence-index.json').write_text(json.dumps({'files':files},indent=2)+'\n',encoding='utf-8')
print(json.dumps({k:v for k,v in report.items() if k in {'status','error','summary'}},indent=2))
raise SystemExit(0 if report['status']=='diagnostic_complete' else 1)
