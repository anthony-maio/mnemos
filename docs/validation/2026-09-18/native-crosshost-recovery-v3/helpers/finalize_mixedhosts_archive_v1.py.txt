"""Attach bounded failure results and index the completed archive."""
from pathlib import Path
import hashlib
import json
import shutil

root = Path(__file__).resolve().parents[1]
out = root / 'docs/validation/2026-09-18/native-crosshost-recovery-v3'
def sha(raw):
    return hashlib.sha256(raw).hexdigest()
def dump(path, value):
    path.write_text(json.dumps(value,indent=2)+'\n',encoding='utf-8')
for version in ('v1','v2'):
    worker = json.loads((out / version / 'outage-b.json').read_text())
    source = Path(worker['home']).parents[1] / 'outage-b-stdout.jsonl'
    raw = source.read_bytes()
    rows = [json.loads(line) for line in raw.decode('utf-8').splitlines() if line.startswith('{')]
    rows = [row for row in rows if row.get('type') == 'result']
    assert len(rows) == 1
    result = {key:rows[0][key] for key in ('type','subtype','is_error','result','total_cost_usd','num_turns') if key in rows[0]}
    if version == 'v1':
        assert result['subtype'] == 'error_max_budget_usd' and result['total_cost_usd'] == 0.370554
    else:
        assert result['subtype'] == 'success' and result['result'] == 'outage cobaltfox'
    destination = out / version / 'claude-result.selected.json'
    assert not destination.exists()
    dump(destination, {'source_sha256':sha(raw),'result':result})
shutil.copyfile(__file__, out / 'helpers' / (Path(__file__).name + '.txt'))
# The successful verifier preserved all observation helpers. Only documentation
# and these bounded result summaries are added before the final index is staged.
files = {p.relative_to(out).as_posix():{'bytes':p.stat().st_size,'sha256':sha(p.read_bytes())}
         for p in sorted(out.rglob('*')) if p.is_file() and p.name != 'evidence-index.json'}
dump(out / 'evidence-index.json',{'files':files})
print(json.dumps({'indexed_files':len(files)}))
