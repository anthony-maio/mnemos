"""Run the ten comparator service cases against the verified installed wheel."""
from pathlib import Path
import hashlib
import json
import os
import shutil
import subprocess
import xml.etree.ElementTree as ET
import zipfile

ROOT = Path(__file__).resolve().parents[1]
TEMP = Path('C:/Users/Ant/AppData/Local/Temp')
SNAPSHOT = TEMP / 'mnemos-source-encoding-comparison-v1'
CHECKS = TEMP / 'mnemos-package-checks-encoding-comparison-v1'
PRIOR = TEMP / 'mnemos-installed-checks-encoding-comparison-v2/installation.json'
ENV = TEMP / 'mnemos-installed-encoding-comparison-v1'
SITE = ENV / 'Lib/site-packages'
PYTHON = ENV / 'Scripts/python.exe'
OUT = TEMP / 'mnemos-installed-encoding-cases-v1'
TEST = Path('tests/test_encoding_comparator.py')
COMMIT = 'aefc338c159c1e00d76d40c1947e7a3f11a6c538'
WHEEL = CHECKS / 'dist/mnemos_memory-0.7.0-py3-none-any.whl'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def installed_hashes():
    result = {}
    with zipfile.ZipFile(WHEEL) as wheel:
        for name in wheel.namelist():
            if name.startswith('mnemos/') and not name.endswith('/'):
                digest = hashlib.sha256(wheel.read(name)).hexdigest()
                assert sha(SITE / name) == digest, name
                result[name] = digest
    assert result
    return result


prior = json.loads(PRIOR.read_text(encoding='utf-8'))
assert prior['status'] == 'passed' and prior['commit'] == COMMIT
assert prior['bindings_after']['wheel_sha256'] == sha(WHEEL)
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=SNAPSHOT, text=True).strip() == COMMIT
assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=SNAPSHOT, text=True).strip()
OUT.mkdir(exist_ok=False)
fixture = OUT / 'fixture'
(fixture / 'tests').mkdir(parents=True)
(fixture / 'tests/__init__.py').write_text('', encoding='utf-8')
(fixture / 'benchmarks').mkdir()
(fixture / 'benchmarks/__init__.py').write_text('', encoding='utf-8')
shutil.copyfile(SNAPSHOT / TEST, fixture / TEST)
shutil.copytree(SNAPSHOT / 'benchmarks/local_memory', fixture / 'benchmarks/local_memory',
                ignore=shutil.ignore_patterns('__pycache__'))
inputs = {p.relative_to(fixture).as_posix(): sha(p) for p in fixture.rglob('*') if p.is_file()}
for name, digest in inputs.items():
    if name not in {'tests/__init__.py', 'benchmarks/__init__.py'}:
        assert sha(SNAPSHOT / name) == digest
env = {key: value for key, value in os.environ.items()
       if not key.startswith('MNEMOS_') and key not in {'PYTHONPATH', 'VIRTUAL_ENV'}}
env['TEMP'] = env['TMP'] = env['TMPDIR'] = str(OUT)
record = {'source_commit': COMMIT, 'status': 'running', 'runner_sha256': sha(Path(__file__)),
          'wheel_sha256': sha(WHEEL), 'prior_installation_sha256': sha(PRIOR),
          'installed_files_before': installed_hashes(), 'fixture_inputs': inputs}


def save():
    (OUT / 'verification.json').write_text(json.dumps(record, indent=2) + '\n', encoding='utf-8')


save()
try:
    origin = subprocess.check_output([str(PYTHON), '-c',
        'import mnemos.memory_service as m; print(m.__file__)'], cwd=fixture, env=env, text=True).strip()
    assert Path(origin).resolve() == (SITE / 'mnemos/memory_service.py').resolve()
    record['installed_module_origin'] = origin
    command = [str(PYTHON), '-m', 'pytest', '-q', str(TEST), '--basetemp', str(OUT / 'pytest-temp'),
               '--junitxml', str(OUT / 'installed.xml')]
    record['command'] = command
    with (OUT / 'pytest.log').open('w', encoding='utf-8') as log:
        result = subprocess.run(command, cwd=fixture, env=env, stdout=log, stderr=subprocess.STDOUT,
                                creationflags=getattr(subprocess, 'CREATE_NO_WINDOW', 0))
    record['exit_code'] = result.returncode
    assert result.returncode == 0
    selected = []
    for path in (CHECKS / 'pytest.xml', OUT / 'installed.xml'):
        cases = [c for c in ET.parse(path).getroot().iter('testcase')
                 if c.attrib.get('classname', '').endswith('test_encoding_comparator')]
        assert len(cases) == 10
        assert all(all(c.find(tag) is None for tag in ('failure', 'error', 'skipped')) for c in cases)
        selected.append(sorted((c.attrib['classname'], c.attrib['name']) for c in cases))
    assert selected[0] == selected[1]
    record['matched_cases'] = selected[0]
    record['installed_files_after'] = installed_hashes()
    assert record['installed_files_after'] == record['installed_files_before']
    assert all(sha(fixture / name) == digest for name, digest in inputs.items())
    record['status'] = 'passed'
except BaseException as error:
    record.update(status='failed', error=repr(error))
    raise
finally:
    save()
print(json.dumps({'status': record['status'], 'matched_cases': len(record['matched_cases'])}))
