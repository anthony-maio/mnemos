def _nested_mappings(value):
    """Yield parsed mapping payloads without retaining native transcript text."""
    if isinstance(value, Mapping):
        yield value
        for child in value.values():
            yield from _nested_mappings(child)
    elif isinstance(value, list):
        for child in value:
            yield from _nested_mappings(child)
    elif isinstance(value, str):
        stripped = value.strip()
        if stripped.startswith("{") or stripped.startswith("["):
            try:
                yield from _nested_mappings(json.loads(stripped))
            except json.JSONDecodeError:
                return


def _preview_payload(results):
    """Return the authoritative preview object from a native MCP result."""
    for result in results:
        for payload in result.get("payload", []):
            for mapping in _nested_mappings(payload):
                if isinstance(mapping.get("preview_digest"), str) and isinstance(
                    mapping.get("candidates"), list
                ):
                    return mapping
    return None


def _selection_payload(results):
    """Return the authoritative related-forget result from a native MCP result."""
    for result in results:
        for payload in result.get("payload", []):
            for mapping in _nested_mappings(payload):
                if any(
                    key in mapping for key in ("selected_fact_ids", "forgotten", "deleted_fact_ids")
                ):
                    return mapping
    return None


def _safe_preview(preview):
    """Record reviewable IDs and content digests, not arbitrary native payloads."""
    candidates = []
    for item in preview.get("candidates", []):
        if not isinstance(item, dict):
            continue
        content = item.get("content")
        candidates.append(
            {
                "fact_id": item.get("fact_id"),
                "match_type": item.get("match_type"),
                "similarity": item.get("similarity"),
                "matched_assertions": item.get("matched_assertions", []),
                "required_source_event_ids": item.get("required_source_event_ids", []),
                "content_sha256": (
                    digest(content.encode("utf-8")) if isinstance(content, str) else None
                ),
                "content": content if isinstance(content, str) else None,
            }
        )
    return {
        "status": preview.get("status"),
        "complete": preview.get("complete"),
        "preview_digest": preview.get("preview_digest"),
        "fact_id": preview.get("target", {}).get("fact_id"),
        "expected_version": preview.get("target", {}).get("version"),
        "limit": preview.get("limit"),
        "candidates": candidates,
        "similarity_search": preview.get("similarity_search"),
        "source_event_ids": [
            item.get("source_event_id")
            for item in preview.get("source_events", [])
            if isinstance(item, dict) and isinstance(item.get("source_event_id"), str)
        ],
    }


def _reviewable_candidate(item):
    """Require the reviewed copy itself to mention the current or old convention."""
    content = item.get("content")
    if not isinstance(content, str):
        return False
    lowered = content.lower()
    return ("cedar-otter" in lowered or "maple-heron" in lowered) and "release label" in lowered

def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def load(name):
    path = ROOT / ".venv" / name
    if digest(path.read_bytes()) != PINS[name]:
        raise RuntimeError("helper pin mismatch: " + name)
    spec = importlib.util.spec_from_file_location(name.replace(".", "_"), path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def codex_completions(home):
    """Only actual Mnemos completion records; no auth, logs, or full session copy."""
    selected = []
    for path in home.rglob("*.jsonl"):
        for number, line in enumerate(path.read_text(encoding="utf-8").splitlines(), 1):
            value = json.loads(line)
            payload = value.get("payload", {})
            if (
                payload.get("type") == "mcp_tool_call_end"
                and payload.get("invocation", {}).get("server") == "mnemos"
            ):
                selected.append(
                    {"source": path.relative_to(home).as_posix(), "line": number, "record": value}
                )
    return selected


def codex_assistant_answers(home):
    """Keep final assistant text with line and digest provenance."""
    selected = []
    for path in home.rglob("*.jsonl"):
        source_file_sha256 = digest(path.read_bytes())
        for number, raw_line in enumerate(path.read_bytes().splitlines(keepends=True), 1):
            line = raw_line.decode("utf-8", errors="replace")
            try:
                value = json.loads(line)
            except json.JSONDecodeError:
                continue
            payload = value.get("payload", {})
            message = value.get("message", {})
            text = None
            if isinstance(payload, dict) and payload.get("type") in {
                "assistant_message",
                "agent_message",
            }:
                text = payload.get("text", payload.get("message"))
            elif isinstance(message, dict) and message.get("role") == "assistant":
                content = message.get("content")
                if isinstance(content, str):
                    text = content
                elif isinstance(content, list):
                    text = "\n".join(
                        item.get("text", "")
                        for item in content
                        if isinstance(item, dict) and isinstance(item.get("text"), str)
                    )
            if isinstance(text, str) and text.strip():
                selected.append(
                    {
                        "source": path.relative_to(home).as_posix(),
                        "line": number,
                        "line_sha256": digest(raw_line),
                        "source_file_sha256": source_file_sha256,
                        "phase": payload.get("phase") if isinstance(payload, dict) else None,
                        "text_sha256": digest(text.encode("utf-8")),
                        "text": text,
                    }
                )
    return selected


def decode_codex(selected, expected):
    assert len(selected) == len(expected)
    decoded = []
    for item, tool in zip(selected, expected):
        payload = item["record"]["payload"]
        assert payload["invocation"]["server"] == "mnemos"
        assert payload["invocation"]["tool"] == tool
        assert set(payload["result"]) == {"Ok"}
        result = payload["result"]["Ok"]
        assert result["isError"] is False
        value = json.loads(result["structuredContent"]["result"])
        assert value == json.loads(result["content"][0]["text"])
        assert value.get("status") != "error" and "error" not in value
        decoded.append(value)
    return decoded


def force_unelevated(path, lifecycle_module):
    text = path.read_text(encoding="utf-8")
    if "[windows]" not in text:
        text = text.rstrip() + '\n\n[windows]\nsandbox = "unelevated"\n'
        path.write_text(text, encoding="utf-8")
    else:
        text = text.replace('sandbox = "workspace-write"', 'sandbox = "unelevated"')
        path.write_text(text, encoding="utf-8")
    lifecycle_module._protect_file(path)
    assert tomllib.loads(path.read_text(encoding="utf-8"))["windows"]["sandbox"] == "unelevated"

            "claude_forget_auth_absent": forget_auth is None or not forget_auth.exists(),
            "codex_final_auth_absent": final_auth is None or not final_auth.exists(),
            "descriptor_absent": not (fixture / "runtime.json").exists(),
        }
        if errors or not all(v for k, v in report["cleanup"].items() if k != "errors"):
            report["status"] = "failed"
        report["finished_at"] = datetime.now(timezone.utc).isoformat()
        checkpoint()
    print(json.dumps({"status": report["status"], "report": str(args.output)}), flush=True)
    return 0 if report["status"] == "observed_partial" else 1
