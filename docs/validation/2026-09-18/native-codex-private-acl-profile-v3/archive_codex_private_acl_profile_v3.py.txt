def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def read_json(path: Path) -> Any:
    return json.loads(path.read_text(encoding="utf-8"))


def load_previous_helpers():
    path = ROOT / ".venv/archive_codex_stop_profile_v2.py"
    spec = importlib.util.spec_from_file_location("codex_stop_profile_v2_archive", path)
    if spec is None or spec.loader is None:
        raise ImportError(path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    module.REPORT = REPORT
    module.FIXTURE = FIXTURE
    module.RUNNER = RUNNER
    module.SOURCE = SOURCE
    module.WHEEL = WHEEL
    module.PROFILE = PROFILE
    return module


def acl_observation() -> dict[str, Any]:
    paths = {"fixture_root": FIXTURE, "outbox_lock": FIXTURE / "outbox.json.lock"}
    raw_observations: dict[str, dict[str, Any]] = {}
    for label, path in paths.items():
        if not path.exists():
            raise AssertionError(f"ACL path is missing: {path}")
        escaped = str(path).replace("'", "''")
        command = (
            "$a=Get-Acl -LiteralPath '"
            + escaped
            + "'; $rules=@($a.Access); "
            + "$ownerSid=(New-Object System.Security.Principal.NTAccount($a.Owner)).Translate([System.Security.Principal.SecurityIdentifier]).Value; "
            + "$ruleSids=@($rules | ForEach-Object { $_.IdentityReference.Translate([System.Security.Principal.SecurityIdentifier]).Value }); "
            + "[pscustomobject]@{ owner_sid=$ownerSid; protected=$a.AreAccessRulesProtected; "
            + "rule_count=$rules.Count; inherited_count=@($rules | Where-Object IsInherited).Count; "
            + "rule_sids=$ruleSids; rule_types=@($rules | ForEach-Object {$_.AccessControlType.ToString()}); "
            + "rule_rights=@($rules | ForEach-Object {$_.FileSystemRights.ToString()}) } | ConvertTo-Json -Compress"
        )
        completed = subprocess.run(
            ["pwsh", "-NoProfile", "-Command", command],
            capture_output=True,
            text=True,
            check=True,
        )
        raw = json.loads(completed.stdout)
        rule_sids = raw["rule_sids"] if isinstance(raw["rule_sids"], list) else [raw["rule_sids"]]
        if raw["protected"] is not True or raw["rule_count"] != 1 or raw["inherited_count"] != 0:
            raise AssertionError(f"ACL is not protected owner-only state: {label}")
        if rule_sids != [raw["owner_sid"]]:
            raise AssertionError(f"ACL rule is not exactly the current owner SID: {label}")
        if raw["rule_types"] != ["Allow"] or "FullControl" not in raw["rule_rights"][0]:
            raise AssertionError(f"ACL rule is not owner FullControl: {label}")
        raw_observations[label] = {
            "path": str(path),
            "owner_sid_sha256": sha(raw["owner_sid"].encode("utf-8")),
            "rule_sid_sha256": sha(rule_sids[0].encode("utf-8")),
            "owner_rule_same_sid": True,
            "access_rules_protected": True,
            "rule_count": 1,
            "inherited_rule_count": 0,
            "rule_type": "Allow",
            "rule_rights": raw["rule_rights"],
        }
    return {
        "status": "owner_only_protected",
        "observations": raw_observations,
        "privacy": "Account names and raw SIDs are omitted; hashes bind the owner and rule identities.",
    }


def strict_capture(report: dict[str, Any], capture: dict[str, Any]) -> dict[str, Any]:
    import sqlite3

    target = report["target"]
    expected = {item["id"]: item for item in report["capture_events"]}
    db_path = FIXTURE / "memory.sqlite"
    rows_out = []
    with sqlite3.connect(db_path.as_uri() + "?mode=ro", uri=True) as db:
        db.row_factory = sqlite3.Row
        rows = db.execute(
            "select id, content, role, source, scope, scope_id, session_id, metadata "
            "from episodic_events where id in (?, ?)",
            tuple(expected),
        ).fetchall()
        if len(rows) != 2:
            raise AssertionError("SQLite reopen did not find both capture rows")
        for row in rows:
            metadata = json.loads(row["metadata"])
            item = {
                "id": row["id"],
                "role": row["role"],
                "source": row["source"],
                "source_kind": metadata.get("source_kind"),
                "host": metadata.get("host"),
                "scope": row["scope"],
                "scope_id": row["scope_id"],
                "project_id": metadata.get("project_id"),
                "session_id": row["session_id"],
                "profile": metadata.get("user_profile_id"),
                "content_sha256": sha(row["content"].encode("utf-8")),
            }
            exp = expected[row["id"]]
            if item["content_sha256"] != exp["content_sha256"] or item["role"] != exp["role"]:
                raise AssertionError("SQLite capture content or role differs")
            if item["source"] != "codex" or item["source_kind"] != "codex_hook":
                raise AssertionError("SQLite source metadata differs")
            if item["host"] != "codex" or item["scope"] != "project":
                raise AssertionError("SQLite host or scope differs")
            if item["scope_id"] != target["scope_id"] or item["project_id"] != target["scope_id"]:
                raise AssertionError("SQLite project scope differs")
            if item["session_id"] != exp["session_id"] or item["profile"] != PROFILE:
                raise AssertionError("SQLite session or profile differs")
            rows_out.append(item)
    capture["strict_sqlite_rows"] = rows_out
    return capture


def main() -> None:
    if ARCHIVE.exists():
        raise FileExistsError(ARCHIVE)
    report = read_json(REPORT)
    if report.get("status") != "observed_transport" or report.get("source_commit") != SOURCE:
        raise AssertionError("unexpected status or source commit")
    if report.get("runner_sha256") != sha(RUNNER.read_bytes()) or report.get("wheel_sha256") != WHEEL:
        raise AssertionError("runner or wheel binding differs")
    if report["process"]["returncode"] != 0 or report["process"]["timed_out"]:
        raise AssertionError("native process did not complete")
    if report["selected_completion"]["target_verified"] is not True:
        raise AssertionError("native completion target was not verified")
    helpers = load_previous_helpers()
    selected = helpers.selected_completion(report)
    messages = helpers.selected_messages()
    capture = strict_capture(report, helpers.sqlite_capture_summary(report, selected))
    expected_events = {item["role"]: item for item in report["capture_events"]}
    by_role = {item["role"]: item for item in messages}
    if set(by_role) != {"user", "assistant"}:
        raise AssertionError("prompt and assistant records are incomplete")
    for role in by_role:
        if by_role[role]["text_sha256"] != expected_events[role]["content_sha256"]:
            raise AssertionError(f"native {role} text differs from persisted capture")
    capture["prompt_source"] = by_role["user"]
    capture["final_answer_source"] = by_role["assistant"]
    capture["prompt_sha256"] = by_role["user"]["text_sha256"]
    capture["final_answer_sha256"] = by_role["assistant"]["text_sha256"]
    marker_data, profile_data = helpers.marker_and_profile(report)
    acl = acl_observation()

    ARCHIVE.mkdir(parents=True)
    (ARCHIVE / "report.json").write_bytes(REPORT.read_bytes())
    (ARCHIVE / "archive_codex_private_acl_profile_v3.py.txt").write_bytes(Path(__file__).read_bytes())
    (ARCHIVE / "native_codex_private_acl_profile_v3.py.txt").write_bytes(RUNNER.read_bytes())
    for name, expected_hash in report["helper_pins"].items():
        path = ROOT / ".venv" / name
        if not path.exists() or sha(path.read_bytes()) != expected_hash:
            raise AssertionError(f"helper pin mismatch: {name}")
        (ARCHIVE / f"{Path(name).name}.txt").write_bytes(path.read_bytes())
    (ARCHIVE / "stop-boundary-bootstrap.py.txt").write_bytes(
        (FIXTURE / "stop-boundary-bootstrap.py").read_bytes()
    )
    (ARCHIVE / "stop-python-markers.jsonl").write_text(
        "".join(json.dumps(row, separators=(",", ":")) + "\n" for row in marker_data["marker_rows"]),
        encoding="utf-8",
    )
    (ARCHIVE / "stop-shell.txt").write_text(report["stop-shell.txt"]["text"], encoding="utf-8")
    (ARCHIVE / "selected-mcp-completion.json").write_text(
        json.dumps(selected, indent=2) + "\n", encoding="utf-8"
    )
    (ARCHIVE / "selected-prompt-answer.json").write_text(
        json.dumps(messages, indent=2) + "\n", encoding="utf-8"
    )
    (ARCHIVE / "sqlite-capture-summary.json").write_text(
        json.dumps(capture, indent=2) + "\n", encoding="utf-8"
    )
    timing = {
        **marker_data["summary"],
        "report_process_elapsed_ms": report["process"]["elapsed_ms"],
        "limits": report["stop_instrumentation"]["limits"],
    }
    (ARCHIVE / "timing-summary.json").write_text(json.dumps(timing, indent=2) + "\n", encoding="utf-8")
    (ARCHIVE / "profile-top60.json").write_text(
        json.dumps(profile_data, indent=2) + "\n", encoding="utf-8"
    )
    (ARCHIVE / "acl-observation.json").write_text(json.dumps(acl, indent=2) + "\n", encoding="utf-8")
    (ARCHIVE / ".gitattributes").write_text("* -text\n*.jsonl -whitespace\n", encoding="utf-8")
    readme = (
        "# Native Codex private ACL profile v3\n\n"
        "This archive preserves one successful instrumented Codex Stop observation against the installed `5d8cacf` package. The native MCP retrieve used the project scope and returned the seeded river-cedar fact while excluding the foreign project. A read-only SQLite reopen matched the user and assistant capture rows by content hash, role, source, host, session, project, scope, profile, and native evidence.\n\n"
        "The marker and timing files preserve shell elapsed time, shell-to-Python, adapter import, handler, and service capture dispatch observations. The bounded profile contains the 60 emitted records. The fixture used a fixed three-second hook deadline. File tracing and the installed adapter bootstrap add overhead, and host dispatch to shell entry is not measured. These values are not an uninstrumented latency, speedup, or general deadline acceptance result.\n\n"
        "The fresh installed package observation also verifies protected owner-only ACLs on the fixture root and `outbox.json.lock`: each has one Allow FullControl rule for the current owner SID, no inherited rules, and protected access rules. Raw account names and SIDs are omitted. This is one instrumented Windows fixture observation and does not establish complete conversation capture, model quality, or broader host reliability. Credentials, full native sessions, and the SQLite database are not archived.\n"
    )
    (ARCHIVE / "README.md").write_text(readme, encoding="utf-8")
    files = {}
    for path in sorted(ARCHIVE.iterdir()):
        if path.name == "evidence-index.json" or not path.is_file():
            continue
        files[path.name] = {"bytes": path.stat().st_size, "sha256": sha(path.read_bytes())}
    (ARCHIVE / "evidence-index.json").write_text(
        json.dumps({"archive": str(ARCHIVE), "files": files}, indent=2) + "\n", encoding="utf-8"
    )
    print(json.dumps({"status": "passed", "archive": str(ARCHIVE), "files": len(files)}))


if __name__ == "__main__":
    main()
