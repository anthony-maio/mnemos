def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def read_json(path: Path) -> Any:
    return json.loads(path.read_text(encoding="utf-8"))


def selected_completion(report: dict[str, Any]) -> dict[str, Any]:
    home = FIXTURE / "codex-home/.codex"
    found: list[dict[str, Any]] = []
    for source in sorted((home / "sessions").rglob("*.jsonl")):
        raw = source.read_bytes()
        source_hash = sha(raw)
        for line_number, raw_line in enumerate(raw.splitlines(keepends=True), 1):
            try:
                record = json.loads(raw_line)
            except json.JSONDecodeError:
                continue
            payload = record.get("payload", {})
            invocation = payload.get("invocation")
            if not isinstance(invocation, dict):
                continue
            if invocation.get("server") != "mnemos":
                continue
            if invocation.get("tool") != report["selected_completion"]["tool"]:
                continue
            result = payload.get("result", {})
            ok = result.get("Ok") if isinstance(result, dict) else None
            if not isinstance(ok, dict) or ok.get("isError") is not False:
                raise AssertionError("selected MCP completion was not successful")
            structured = json.loads(ok["structuredContent"]["result"])
            content = json.loads(ok["content"][0]["text"])
            if structured != content:
                raise AssertionError("selected MCP result representations differ")
            found.append(
                {
                    "source": str(source.relative_to(home)).replace("\\", "/"),
                    "line": line_number,
                    "line_sha256": sha(raw_line),
                    "source_file_sha256": source_hash,
                    "call_id": payload.get("call_id"),
                    "tool": invocation["tool"],
                    "arguments": invocation.get("arguments", {}),
                    "result": structured,
                }
            )
    if len(found) != 1:
        raise AssertionError(f"expected one selected completion, found {len(found)}")
    item = found[0]
    if sha(json.dumps(item["result"], sort_keys=True).encode("utf-8")) != report[
        "selected_completion"
    ]["result_sha256"]:
        raise AssertionError("selected completion result hash differs")
    target = report["target"]
    facts = item["result"].get("facts", [])
    target_fact = next((fact for fact in facts if fact.get("id") == target["fact_id"]), None)
    if not target_fact or target_fact.get("version") != target["version"]:
        raise AssertionError("selected completion did not return target")
    if target_fact.get("scope_id") != target["scope_id"]:
        raise AssertionError("selected completion scope differs")
    if target["foreign_id"] in {fact.get("id") for fact in facts}:
        raise AssertionError("foreign fact leaked into selected completion")
    return item


def selected_messages() -> list[dict[str, Any]]:
    home = FIXTURE / "codex-home/.codex"
    rows: list[dict[str, Any]] = []
    for source in sorted((home / "sessions").rglob("*.jsonl")):
        raw = source.read_bytes()
        source_hash = sha(raw)
        for line_number, raw_line in enumerate(raw.splitlines(keepends=True), 1):
            try:
                record = json.loads(raw_line)
            except json.JSONDecodeError:
                continue
            payload = record.get("payload", {})
            role = payload.get("type")
            message = payload.get("message")
            if role == "user_message" and isinstance(message, str) and message.startswith(
                "Call mnemos_retrieve exactly once for the project shutdown probe"
            ):
                rows.append(
                    {
                        "role": "user",
                        "source": str(source.relative_to(home)).replace("\\", "/"),
                        "line": line_number,
                        "line_sha256": sha(raw_line),
                        "source_file_sha256": source_hash,
                        "text_sha256": sha(message.encode("utf-8")),
                        "text": message,
                    }
                )
            elif role == "agent_message" and isinstance(message, str) and "river-cedar" in message:
                rows.append(
                    {
                        "role": "assistant",
                        "source": str(source.relative_to(home)).replace("\\", "/"),
                        "line": line_number,
                        "line_sha256": sha(raw_line),
                        "source_file_sha256": source_hash,
                        "text_sha256": sha(message.encode("utf-8")),
                        "text": message,
                    }
                )
    if [row["role"] for row in rows] != ["user", "assistant"]:
        raise AssertionError("native prompt/final answer records were not selected")
    return rows


def sqlite_capture_summary(report: dict[str, Any], selected: dict[str, Any]) -> dict[str, Any]:
    target = report["target"]
    expected_events = {item["id"]: item for item in report["capture_events"]}
    db_path = FIXTURE / "memory.sqlite"
    with sqlite3.connect(db_path.as_uri() + "?mode=ro", uri=True) as db:
        db.row_factory = sqlite3.Row
        rows = db.execute(
            "select id, content, role, session_id, scope_id, metadata from episodic_events "
            "where id in (?, ?)",
            tuple(expected_events),
        ).fetchall()
        if len(rows) != 2:
            raise AssertionError("user/assistant capture rows are incomplete")
        captured = []
        for row in rows:
            metadata = json.loads(row["metadata"])
            item = {
                "id": row["id"],
                "role": row["role"],
                "session_id": row["session_id"],
                "project_id": metadata.get("project_id"),
                "scope": metadata.get("scope"),
                "scope_id": row["scope_id"],
                "profile": metadata.get("user_profile_id"),
                "content_sha256": sha(row["content"].encode("utf-8")),
                "source_kind": metadata.get("source_kind"),
                "host": metadata.get("host"),
            }
            expected = expected_events[row["id"]]
            for key in ("role", "session_id", "content_sha256"):
                if item[key] != expected[key]:
                    raise AssertionError(f"capture {key} differs from report")
            if item["project_id"] != target["scope_id"] or item["scope_id"] != target["scope_id"]:
                raise AssertionError("capture scope differs from target")
            if item["profile"] != PROFILE:
                raise AssertionError("capture profile differs")
            captured.append(item)
        if {item["role"] for item in captured} != {"user", "assistant"}:
            raise AssertionError("capture roles are incomplete")
        table_counts = {
            table: db.execute(f"select count(*) from {table}").fetchone()[0]
            for table in ("memory_chunks", "episodic_events", "fact_evidence")
        }
    return {
        "database_reopened_read_only": True,
        "target": {
            "fact_id": target["fact_id"],
            "content_sha256": sha(target["content"].encode("utf-8")),
            "version": target["version"],
            "project_id": target["scope_id"],
        },
        "captured_events": captured,
        "query_sha256": sha(selected["arguments"]["query"].encode("utf-8")),
        "table_counts": table_counts,
    }


def marker_and_profile(report: dict[str, Any]) -> tuple[dict[str, Any], list[dict[str, Any]]]:
    shell_text = report["stop-shell.txt"]["text"]
    shell = [line for line in shell_text.splitlines() if line]
    rows = [
        json.loads(line)
        for line in report["stop-python.jsonl"]["text"].splitlines()
        if line
    ]
    expected = [
        "shell_entry",
        "python_entry",
        "before_adapter_import",
        "after_adapter_import",
        "adapter_main_start",
        "run_hook_start",
        "run_hook_result",
        "run_hook_end",
        "handler_profile",
        "adapter_main_end",
        "stdout_flushed",
        "python_exit",
        "shell_exit",
    ]
    actual = ["shell_entry"] + [row["stage"] for row in rows] + ["shell_exit"]
    if actual != expected:
        raise AssertionError(f"marker order differs: {actual}")
    if len(shell) != 2:
        raise AssertionError("shell marker count differs")
    shell_start = int(shell[0].split("|")[1])
    shell_end = int(shell[1].split("|")[1])
    before = next(row for row in rows if row["stage"] == "before_adapter_import")
    after = next(row for row in rows if row["stage"] == "after_adapter_import")
    entry = next(row for row in rows if row["stage"] == "python_entry")
    hook_start = next(row for row in rows if row["stage"] == "run_hook_start")
    hook_result = next(row for row in rows if row["stage"] == "run_hook_result")
    profile = next(row for row in rows if row["stage"] == "handler_profile")
    functions = profile.get("functions")
    if not isinstance(functions, list) or len(functions) != 60:
        raise AssertionError("profile did not provide exactly 60 records")
    markers = [row for row in rows if row["stage"] != "handler_profile"]
    summary = {
        "sequence": actual,
        "shell_elapsed_ms": shell_end - shell_start,
        "shell_to_python_ms": round(entry["epoch_ns"] / 1_000_000 - shell_start, 4),
        "adapter_import_ms": round((after["epoch_ns"] - before["epoch_ns"]) / 1_000_000, 4),
        "handler_ms": round((hook_result["epoch_ns"] - hook_start["epoch_ns"]) / 1_000_000, 4),
        "service_capture_dispatch_ms": [
            op["elapsed_ms"]
            for op in report["dispatch_trace"]
            if op.get("operation") == "capture"
            and op.get("host") == "codex"
            and op.get("result_status") == "committed"
        ][-1],
        "profile_records_archived": len(functions),
        "python_trace_sha256": sha(report["stop-python.jsonl"]["text"].encode("utf-8")),
        "shell_trace_sha256": sha(shell_text.encode("utf-8")),
    }
    profile_summary = {
        "stage": profile["stage"],
        "pid": profile["pid"],
        "epoch_ns": profile["epoch_ns"],
        "monotonic_ns": profile["monotonic_ns"],
        "source_record_count": len(functions),
        "archived_record_count": len(functions),
        "records": functions,
    }
    return {"summary": summary, "marker_rows": markers}, profile_summary


def acl_observation() -> dict[str, Any]:
    lock = FIXTURE / "outbox.json.lock"
    if not lock.is_file():
        raise AssertionError("fixture lock file is missing")
    script = (
        "$a=Get-Acl -LiteralPath "
        + "'" + str(lock).replace("'", "''") + "'"
        + "; [pscustomobject]@{ path='outbox.json.lock'; "
        + "are_access_rules_protected=$a.AreAccessRulesProtected; "
        + "rule_count=@($a.Access).Count; "
        + "inherited_rule_count=@($a.Access | Where-Object IsInherited).Count; "
        + "explicit_rule_count=@($a.Access | Where-Object {-not $_.IsInherited}).Count }"
        + " | ConvertTo-Json -Compress"
    )
    result = subprocess.run(
        ["pwsh", "-NoProfile", "-Command", script],
        capture_output=True,
        text=True,
        check=True,
    )
    observation = json.loads(result.stdout)
    if observation["are_access_rules_protected"] is not False:
        raise AssertionError("old-wheel fixture no longer shows inherited lock ACL")
    if observation["inherited_rule_count"] < 1:
        raise AssertionError("fixture lock has no inherited access rule")
    return observation


def main() -> None:
    if ARCHIVE.exists():
        raise FileExistsError(ARCHIVE)
    report = read_json(REPORT)
    if report.get("status") != "observed_transport" or report.get("source_commit") != SOURCE:
        raise AssertionError("unexpected stop-profile status or source")
    if report.get("runner_sha256") != sha(RUNNER.read_bytes()):
        raise AssertionError("immutable executed runner hash differs")
    if report.get("wheel_sha256") != WHEEL:
        raise AssertionError("unexpected wheel hash")
    if report["process"]["returncode"] != 0 or report["process"]["timed_out"]:
        raise AssertionError("native process did not complete successfully")
    if report["selected_completion"]["target_verified"] is not True:
        raise AssertionError("selected completion was not target verified")
    selected = selected_completion(report)
    messages = selected_messages()
    capture = sqlite_capture_summary(report, selected)
    by_role = {item["role"]: item for item in messages}
    expected_capture = {item["role"]: item for item in report["capture_events"]}
    for role in ("user", "assistant"):
        if by_role[role]["text_sha256"] != expected_capture[role]["content_sha256"]:
            raise AssertionError(f"native {role} hash differs from persisted capture")
    capture["prompt_source"] = by_role["user"]
    capture["final_answer_source"] = by_role["assistant"]
    capture["prompt_sha256"] = by_role["user"]["text_sha256"]
    capture["final_answer_sha256"] = by_role["assistant"]["text_sha256"]
    marker_data, profile_data = marker_and_profile(report)
    acl = acl_observation()
    if not report["cleanup"]["auth_absent"] or not report["cleanup"]["descriptor_absent"]:
        raise AssertionError("cleanup was not verified")
    if report["cleanup"]["errors"]:
        raise AssertionError("cleanup reported errors")

    ARCHIVE.mkdir(parents=True)
    shutil.copyfile(REPORT, ARCHIVE / "report.json")
    shutil.copyfile(__file__, ARCHIVE / "archive_codex_stop_profile_v2.py.txt")
    shutil.copyfile(RUNNER, ARCHIVE / "native_codex_stop_profile_v2.py.txt")
    for name, expected in report["helper_pins"].items():
        path = ROOT / ".venv" / name
        if not path.exists() or sha(path.read_bytes()) != expected:
            raise AssertionError(f"helper pin mismatch: {name}")
        shutil.copyfile(path, ARCHIVE / f"{Path(name).name}.txt")
    bootstrap = FIXTURE / "stop-boundary-bootstrap.py"
    shutil.copyfile(bootstrap, ARCHIVE / "stop-boundary-bootstrap.py.txt")
    (ARCHIVE / "stop-python-markers.jsonl").write_text(
        "".join(json.dumps(row, separators=(",", ":")) + "\n" for row in marker_data["marker_rows"]),
        encoding="utf-8",
    )
    (ARCHIVE / "stop-shell.txt").write_text(report["stop-shell.txt"]["text"], encoding="utf-8")
    (ARCHIVE / "selected-mcp-completion.json").write_text(
        json.dumps(selected, indent=2) + "\n", encoding="utf-8"
    )
    (ARCHIVE / "selected-prompt-answer.json").write_text(
        json.dumps(messages, indent=2) + "\n", encoding="utf-8"
    )
    (ARCHIVE / "sqlite-capture-summary.json").write_text(
        json.dumps(capture, indent=2) + "\n", encoding="utf-8"
    )
    (ARCHIVE / "timing-summary.json").write_text(
        json.dumps(
            {
                **marker_data["summary"],
                "report_process_elapsed_ms": report["process"]["elapsed_ms"],
                "limits": report["stop_instrumentation"]["limits"],
            },
            indent=2,
        )
        + "\n",
        encoding="utf-8",
    )
    (ARCHIVE / "profile-top60.json").write_text(
        json.dumps(profile_data, indent=2) + "\n", encoding="utf-8"
    )
    (ARCHIVE / "acl-observation.json").write_text(
        json.dumps(
            {
                **acl,
                "source": "old installed wheel fixture",
                "interpretation": "The v2 fixture retained an inherited lock ACL; this run does not test the later source repair.",
            },
            indent=2,
        )
        + "\n",
        encoding="utf-8",
    )
    (ARCHIVE / ".gitattributes").write_text("* -text\n*.jsonl -whitespace\n", encoding="utf-8")
    readme = (
        "# Native Codex Stop profile v2\n\n"
        "This archive preserves one successful instrumented Codex Stop observation against the installed `5af3d15` package. The selected native MCP retrieve returned the seeded project fact and excluded the foreign project. A read-only SQLite reopen matched the report's user and assistant capture rows by content hash, role, session, project, scope, and profile.\n\n"
        "The marker summary records shell entry, Python bootstrap, adapter import, hook execution, profile emission, and exit. `profile-top60.json` contains the 60 records emitted by the fixture's bounded profiler. The timing summary records shell elapsed time, shell-to-Python, adapter import, handler, and the committed service capture dispatch observation. File tracing and the installed adapter bootstrap add overhead; host dispatch to shell entry is not measured, so this is not an uninstrumented latency acceptance result.\n\n"
        "The fixture's old-wheel `outbox.json.lock` observation has inherited access rules (`AreAccessRulesProtected=false`). A later source repair corrected the protection flag in separate tests; this v2 run does not verify that repair and must not be used as its evidence.\n\n"
        "This is bounded transport, capture, and profiling evidence for one fixture observation. It does not establish complete conversation capture, full native lifecycle acceptance, model quality, full-capture acceptance, or a latency gate. Earlier native timeout evidence remains open. Credentials, full native sessions, and the SQLite database are not archived.\n"
    )
    (ARCHIVE / "README.md").write_text(readme, encoding="utf-8")
    files = {}
    for path in sorted(ARCHIVE.iterdir()):
        if path.name == "evidence-index.json" or not path.is_file():
            continue
        files[path.name] = {"bytes": path.stat().st_size, "sha256": sha(path.read_bytes())}
    (ARCHIVE / "evidence-index.json").write_text(
        json.dumps({"archive": str(ARCHIVE), "files": files}, indent=2) + "\n",
        encoding="utf-8",
    )
    print(json.dumps({"status": "passed", "archive": str(ARCHIVE), "files": len(files)}))


if __name__ == "__main__":
    main()
