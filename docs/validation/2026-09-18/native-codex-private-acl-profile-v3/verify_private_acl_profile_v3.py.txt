from __future__ import annotations

from pathlib import Path

import subprocess
from typing import Any


ROOT = Path(__file__).resolve().parents[1]
ARCHIVE = ROOT / "docs/validation/2026-09-18/native-codex-private-acl-profile-v3"
REPORT = Path(
    "C:/Users/Ant/AppData/Local/Temp/mnemos-native-codex-acl-stop-profile-v3/report.json"
)
FIXTURE = Path(

RUNNER = ROOT / ".venv/native_codex_private_acl_profile_v3.py"
V2_DEPENDENCY = ROOT / ".venv/archive_codex_stop_profile_v2.py"
V2_DEPENDENCY_SHA = "17eed277722e75941d11ffe87179aa07b812e83c746ff40bae3e22b92cb0b065"
FOREIGN_CANARY = "foreign-canary-849"
PROFILE = "native-codex-stop-profile-v3"


def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def read_json(path: Path) -> Any:
    return json.loads(path.read_text(encoding="utf-8"))


def acl_check(path: Path) -> dict[str, Any]:
    escaped = str(path).replace("'", "''")
    command = (
        "$a=Get-Acl -LiteralPath '"
        + escaped
        + "'; $rules=@($a.Access); "
        + "$ownerSid=(New-Object System.Security.Principal.NTAccount($a.Owner)).Translate([System.Security.Principal.SecurityIdentifier]).Value; "
        + "$ruleSids=@($rules | ForEach-Object { $_.IdentityReference.Translate([System.Security.Principal.SecurityIdentifier]).Value }); "
        + "[pscustomobject]@{ owner_sid=$ownerSid; protected=$a.AreAccessRulesProtected; "
        + "rule_count=$rules.Count; inherited_count=@($rules | Where-Object IsInherited).Count; "
        + "rule_sids=$ruleSids; rule_types=@($rules | ForEach-Object {$_.AccessControlType.ToString()}); "
        + "rule_rights=@($rules | ForEach-Object {$_.FileSystemRights.ToString()}) } | ConvertTo-Json -Compress"
    )
    completed = subprocess.run(
        ["pwsh", "-NoProfile", "-Command", command],
        capture_output=True,
        text=True,
        check=True,
    )
    raw = json.loads(completed.stdout)
    sids = raw["rule_sids"] if isinstance(raw["rule_sids"], list) else [raw["rule_sids"]]
    rights = raw["rule_rights"] if isinstance(raw["rule_rights"], list) else [raw["rule_rights"]]
    types = raw["rule_types"] if isinstance(raw["rule_types"], list) else [raw["rule_types"]]
    if raw["protected"] is not True or raw["rule_count"] != 1 or raw["inherited_count"] != 0:
        raise AssertionError(f"ACL is not protected owner-only state: {path}")
    if sids != [raw["owner_sid"]] or types != ["Allow"] or "FullControl" not in rights[0]:
        raise AssertionError(f"ACL does not contain exactly one owner FullControl rule: {path}")
    return {
        "path": str(path),
        "owner_sid_sha256": sha(raw["owner_sid"].encode("utf-8")),
        "rule_sid_sha256": sha(sids[0].encode("utf-8")),
        "owner_rule_same_sid": True,
        "access_rules_protected": True,
        "rule_count": 1,
        "inherited_rule_count": 0,
        "rule_type": "Allow",
        "rule_rights": rights,
    }


def main() -> None:
    report = read_json(REPORT)
    selected = read_json(ARCHIVE / "selected-mcp-completion.json")
    target = report["target"]
    result = selected["result"]
    args = selected["arguments"]
    if args.get("current_scope") != "project" or args.get("allowed_scopes") != ["project"]:
        raise AssertionError("native query did not use exact project scope arguments")
    fact = next((item for item in result.get("facts", []) if item.get("id") == target["fact_id"]), None)
    if not fact or fact.get("version") != target["version"] or fact.get("content") != target["content"]:
        raise AssertionError("target content or version differs")
    if fact.get("scope") != "project" or fact.get("scope_id") != target["scope_id"]:
        raise AssertionError("target scope differs")
    serialized = json.dumps(result, sort_keys=True)
    if target["foreign_id"] in serialized or FOREIGN_CANARY in serialized:
        raise AssertionError("foreign ID or canary leaked into native result")
    if target["content"] not in result.get("context", ""):
        raise AssertionError("target content missing from native context")
    if sha(json.dumps(result, sort_keys=True).encode("utf-8")) != report["selected_completion"]["result_sha256"]:
        raise AssertionError("selected result hash differs from report")

    raw_stop = FIXTURE / "stop-python.jsonl"
    archive_raw_stop = ARCHIVE / "stop-python.jsonl"
    if archive_raw_stop.read_bytes() != raw_stop.read_bytes():
        raise AssertionError("archived stop-python bytes differ from fixture")
    normalized_raw_stop = raw_stop.read_bytes().replace(b"\r\n", b"\n")
    if normalized_raw_stop != report["stop-python.jsonl"]["text"].encode("utf-8"):
        raise AssertionError("report and fixture stop-python records differ after newline normalization")

    shell = [line for line in report["stop-shell.txt"]["text"].splitlines() if line]
    if [line.split("|")[0] for line in shell] != ["shell_entry", "shell_exit"]:
        raise AssertionError("literal shell marker labels differ")
    if shell[0].split("|")[2] != shell[1].split("|")[2]:
        raise AssertionError("shell marker PIDs differ")
    rows = [json.loads(line) for line in report["stop-python.jsonl"]["text"].splitlines() if line]
    if len({row.get("pid") for row in rows}) != 1:
        raise AssertionError("Python marker PID is not stable")
    start = next(row for row in rows if row.get("stage") == "run_hook_start")
    hook = next(row for row in rows if row.get("stage") == "run_hook_result")
    end = next(row for row in rows if row.get("stage") == "adapter_main_end")
    if start.get("event") != "Stop" or hook.get("status") != "ok" or hook.get("captured") != 1:
        raise AssertionError("Stop status or capture count differs")
    if len(hook.get("event_ids", [])) != 1 or end.get("status") != 0:
        raise AssertionError("Stop event IDs or main status differs")
    origin = next(row for row in rows if row.get("stage") == "after_adapter_import").get("origin", "")
    installed_root = str(Path(report["provenance"]["installed_package"]).parent)
    if not str(origin).lower().startswith(installed_root.lower()) or not str(origin).lower().endswith(
        "mnemos\\adapters\\codex.py"
    ):
        raise AssertionError("adapter module origin is not installed package")
    if report["process"].get("returncode") != 0 or report["process"].get("timed_out"):
        raise AssertionError("native process was not successful")

    auth = FIXTURE / "codex-home/.codex/auth.json"
    runtime = FIXTURE / "runtime.json"
    if auth.exists() or runtime.exists():
        raise AssertionError("auth or runtime descriptor remains")

    expected = {item["id"]: item for item in report["capture_events"]}
    rows_out = []
    with sqlite3.connect((FIXTURE / "memory.sqlite").as_uri() + "?mode=ro", uri=True) as db:
        db.row_factory = sqlite3.Row
        rows_db = db.execute(
            "select id, content, role, source, scope, scope_id, session_id, metadata "
            "from episodic_events where id in (?, ?)",
            tuple(expected),
        ).fetchall()
        if len(rows_db) != 2:
            raise AssertionError("SQLite reopen did not find both captures")
        if db.execute("select count(*) from episodic_events where content like ?", (f"%{FOREIGN_CANARY}%",)).fetchone()[0] < 1:
            raise AssertionError("foreign canary was not present in fixture control")
        for row in rows_db:
            metadata = json.loads(row["metadata"])
            item = {
                "id": row["id"],
                "role": row["role"],
                "source": row["source"],
                "source_kind": metadata.get("source_kind"),
                "host": metadata.get("host"),
                "scope": row["scope"],
                "scope_id": row["scope_id"],
                "project_id": metadata.get("project_id"),
                "session_id": row["session_id"],
                "profile": metadata.get("user_profile_id"),
                "content_sha256": sha(row["content"].encode("utf-8")),
            }
            exp = expected[row["id"]]
            if item["content_sha256"] != exp["content_sha256"] or item["role"] != exp["role"]:
                raise AssertionError("SQLite capture content differs")
            if item["source"] != "codex" or item["source_kind"] != "codex_hook" or item["host"] != "codex":
                raise AssertionError("SQLite source metadata differs")
            if item["scope"] != "project" or item["scope_id"] != target["scope_id"] or item["project_id"] != target["scope_id"]:
                raise AssertionError("SQLite project scope differs")
            if item["session_id"] != exp["session_id"] or item["profile"] != PROFILE:
                raise AssertionError("SQLite session/profile differs")
            rows_out.append(item)

    if sha(RUNNER.read_bytes()) != report["runner_sha256"]:
        raise AssertionError("runner hash differs")
    if sha(V2_DEPENDENCY.read_bytes()) != V2_DEPENDENCY_SHA:
        raise AssertionError("v2 archiver dependency hash differs")
    dependency_dest = ARCHIVE / "archive_codex_stop_profile_v2_dependency.py.txt"
    if not dependency_dest.exists():
        dependency_dest.write_bytes(V2_DEPENDENCY.read_bytes())
    acl = {
        "status": "owner_only_protected",
        "observations": {
            "fixture_root": acl_check(FIXTURE),
            "outbox_lock": acl_check(FIXTURE / "outbox.json.lock"),
        },
        "privacy": "Account names and raw SIDs are omitted; hashes bind owner and rule identities.",
    }
    supplement = {
        "status": "supplementary_assertions_passed",
        "report_input": {
            "archive_file": "report.json",
            "sha256": sha((ARCHIVE / "report.json").read_bytes()),
            "source_commit": report["source_commit"],
            "runner_sha256": report["runner_sha256"],
            "wheel_sha256": report["wheel_sha256"],
        },
        "native_query": {
            "current_scope": args["current_scope"],
            "allowed_scopes": args["allowed_scopes"],
            "target_content_verified": True,
            "foreign_id_and_canary_absent": True,
        },
        "raw_stop_python": {
            "archive_file": "stop-python.jsonl",
            "sha256": sha(archive_raw_stop.read_bytes()),
            "report_normalized_sha256": sha(normalized_raw_stop),
        },
        "sqlite": {"reopened_read_only": True, "captured_rows": rows_out},
        "cleanup": {"runtime_descriptor_absent": True, "auth_copy_absent": True},
        "acl": acl,
        "dependency": {
            "archive_file": dependency_dest.name,
            "sha256": sha(dependency_dest.read_bytes()),
            "expected_sha256": V2_DEPENDENCY_SHA,
        },
        "limits": [
            "This supplements one instrumented fixture observation.",
            "It does not establish host-dispatch-to-shell timing, uninstrumented speed, complete capture, or model quality.",
        ],
    }
    verifier_copy = ARCHIVE / "verify_private_acl_profile_v3.py.txt"
    verifier_copy.write_bytes(Path(__file__).read_bytes())
    supplement["verifier"] = {"archive_file": verifier_copy.name, "sha256": sha(Path(__file__).read_bytes())}
    result_path = ARCHIVE / "supplemental-verification-v1.json"
    result_path.write_text(json.dumps(supplement, indent=2) + "\n", encoding="utf-8")
    index_path = ARCHIVE / "evidence-index.json"
    index = read_json(index_path)
    for path in (dependency_dest, verifier_copy, result_path):
        index["files"][path.name] = {"bytes": path.stat().st_size, "sha256": sha(path.read_bytes())}
    index_path.write_text(json.dumps(index, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"status": "passed", "archive": str(ARCHIVE), "files": len(index["files"])}))


if __name__ == "__main__":
    main()
