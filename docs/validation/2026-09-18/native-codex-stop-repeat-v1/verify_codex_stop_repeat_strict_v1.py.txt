def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def load(p): return json.loads(p.read_text(encoding='utf-8'))
def dump(p, value): p.write_text(json.dumps(value, indent=2)+'\n', encoding='utf-8')
def native_ns(t):
    value = datetime.fromisoformat(t.replace('Z','+00:00'))
    return int(value.timestamp()) * 1_000_000_000 + value.microsecond * 1000

def physical_files(db_path):
    result=[]
    for p in (db_path, Path(str(db_path)+'-wal')):
        result.append({'path':str(p), 'exists':p.exists(), 'bytes':p.stat().st_size if p.exists() else None, 'sha256':sha(p) if p.exists() else None})
    return result
