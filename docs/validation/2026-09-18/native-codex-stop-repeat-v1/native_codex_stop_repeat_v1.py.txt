not profiling wrappers, determine whether Stop completes within its 3-second

def sha(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()


def load(path: Path, name: str):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    assert spec.loader is not None
    spec.loader.exec_module(module)
    return module

assert sys.stdin.isatty() and sys.stdout.isatty()
for name, expected in PINS.items():
    assert sha((ROOT / ".venv" / name).read_bytes()) == expected
assert sha(WHEEL.read_bytes()) == WHEEL_SHA256
assert INSTALLED_PYTHON.exists()
assert Path(sys.executable).resolve() == INSTALLED_PYTHON.resolve()

service = store = process = auth = binding = None

report = {

    "started_at": datetime.now(timezone.utc).isoformat(),

try:

    context=registry.resolve(project,user_profile_id=wrapper.PROFILE,session_id='fixture-seed',host='manual')
    foreign_context=registry.resolve(project_b,user_profile_id=wrapper.PROFILE,session_id='foreign-seed',host='manual')
    service.submit('remember',{'event_id':'control-target','context':context.model_dump(mode='json'),'content':content,'scope':'project'})
    service.submit('remember',{'event_id':'control-foreign','context':foreign_context.model_dump(mode='json'),'content':foreign,'scope':'project'})
    service.submit('maintenance',{})
    target_id,version=wrapper._fact_row(store,content)
    foreign_id,_=wrapper._fact_row(store,foreign)
    assert version==1
    report['target']={'fact_id':target_id,'version':version,'content':content,'scope_id':context.project_id,'foreign_id':foreign_id}
    settings = fixture / "settings.toml"
    settings.write_text("# disposable MCP-only control\n", encoding="utf-8")

    home = fixture / "codex-home"
    lifecycle._private_mkdir(home)

        "MNEMOS_SCOPE_REGISTRY": str(registry_path),
        "MNEMOS_USER_PROFILE_ID": PROFILE,
        "MNEMOS_RUNTIME_DESCRIPTOR": str(fixture / "runtime.json"),
        "MNEMOS_PROJECT_ROOT": str(project),
        "MNEMOS_HOST": "codex",
        "MNEMOS_SESSION_ID": PROFILE,
        "MNEMOS_AUTO_START": "false",
    }

        mnemos_config_path=settings,

        integration_env={k: selectors[k] for k in ("MNEMOS_CONFIG_PATH", "MNEMOS_SCOPE_REGISTRY", "MNEMOS_USER_PROFILE_ID", "MNEMOS_RUNTIME_DESCRIPTOR")},

    combined = original.rstrip() + "\n\n[windows]\nsandbox = \"unelevated\"\n"

    assert "mnemos" in parsed["mcp_servers"]

    # Preserve the generated hook commands and their native deadlines verbatim.

    assert parsed['hooks'] == generated_hooks
    assert generated_hooks['Stop'][0]['hooks'][0]['timeout'] == 3
    report['generated_hooks'] = generated_hooks
    report['trial'] = args.trial
    report['predeclared_trials'] = 3
    report['hook_instrumentation'] = False

    report["config"] = {

        "hook_marker_offset": marker,
        "hook_names": sorted(parsed['hooks']),
        "mcp_server_names": sorted(parsed["mcp_servers"]),

    decoded = json.loads(result["structuredContent"]["result"])

    (OUTPUT / "selected-mcp-completion.json").write_text(json.dumps(selected, indent=2) + "\n", encoding="utf-8")
    report["selected_completion"] = {"tool": "mnemos_retrieve", "result_sha256": sha(json.dumps(decoded, sort_keys=True).encode()), "target_verified": True}
    after=tomllib.loads(config_path.read_text(encoding='utf-8'))

    assert process.returncode == 0 and not timed_out
    report['status'] = 'observed_transport' 
except Exception as exc:

finally:

        report['dispatch_trace'] = trace
        native_events = []
        for path in sorted((config_path.parent / 'sessions').rglob('*.jsonl')):

                payload = value.get('payload')
                if not isinstance(payload, dict):
                    continue
                kind = payload.get('type', '')
                if (kind.startswith('hook_') or kind in ('user_message','agent_message','mcp_tool_call_end','turn_complete','task_complete')):
                    native_events.append({'source': path.relative_to(config_path.parent).as_posix(),

    finally:

        "auth_absent": auth is None or not auth.exists(),

    (OUTPUT / "report.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"status": report["status"], "report": str(OUTPUT / "report.json")}), flush=True)
