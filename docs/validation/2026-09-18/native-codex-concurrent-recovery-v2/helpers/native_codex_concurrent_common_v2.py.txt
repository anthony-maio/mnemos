def transport_origin(client_port, server_port):
    """Read Windows TCP ownership and process ancestry; never read command lines."""
    import ctypes as ct
    from ctypes import wintypes as wt
    import socket
    iphlp = ct.WinDLL('iphlpapi', use_last_error=True)
    get_table = iphlp.GetExtendedTcpTable
    get_table.argtypes = [ct.c_void_p, ct.POINTER(wt.DWORD), wt.BOOL, wt.ULONG, ct.c_int, wt.ULONG]
    get_table.restype = wt.DWORD
    class TcpRow(ct.Structure):
        _fields_ = [(name, wt.DWORD) for name in ('state', 'local_addr', 'local_port', 'remote_addr', 'remote_port', 'pid')]
    size = wt.DWORD(0)
    assert get_table(None, ct.byref(size), False, socket.AF_INET, 5, 0) == 122
    for _ in range(3):
        buffer = ct.create_string_buffer(size.value)
        code = get_table(buffer, ct.byref(size), False, socket.AF_INET, 5, 0)
        if code != 122:
            break
    assert code == 0, code
    count = wt.DWORD.from_buffer(buffer).value
    rows = (TcpRow * count).from_buffer(buffer, ct.sizeof(wt.DWORD))
    matches = [int(row.pid) for row in rows if socket.ntohs(row.local_port & 65535) == client_port
               and socket.ntohs(row.remote_port & 65535) == server_port and row.state == 5]
    assert len(set(matches)) == 1, 'Could not identify exactly one TCP client process'
    client_pid = matches[0]
    class ProcessEntry(ct.Structure):
        _fields_ = [('size', wt.DWORD), ('usage', wt.DWORD), ('pid', wt.DWORD),
                    ('heap', ct.c_size_t), ('module', wt.DWORD), ('threads', wt.DWORD),
                    ('parent', wt.DWORD), ('priority', wt.LONG), ('flags', wt.DWORD),
                    ('exe', wt.WCHAR * 260)]
    kernel = ct.WinDLL('kernel32', use_last_error=True)
    snapshot = kernel.CreateToolhelp32Snapshot
    snapshot.argtypes = [wt.DWORD, wt.DWORD]
    snapshot.restype = wt.HANDLE
    first, following = kernel.Process32FirstW, kernel.Process32NextW
    for method in (first, following):
        method.argtypes = [wt.HANDLE, ct.POINTER(ProcessEntry)]
        method.restype = wt.BOOL
    kernel.CloseHandle.argtypes = [wt.HANDLE]
    kernel.CloseHandle.restype = wt.BOOL
    handle = snapshot(2, 0)
    assert handle not in (None, ct.c_void_p(-1).value)
    parents = {}
    try:
        entry = ProcessEntry()
        entry.size = ct.sizeof(entry)
        more = first(handle, ct.byref(entry))
        assert more
        while more:
            parents[int(entry.pid)] = int(entry.parent)
            more = following(handle, ct.byref(entry))
    finally:
        kernel.CloseHandle(handle)
    ancestry = [client_pid]
    while ancestry[-1] in parents and len(ancestry) < 32:
        parent = parents[ancestry[-1]]
        if parent == 0 or parent in ancestry:
            break
        ancestry.append(parent)
    return {'tcp_client_pid': client_pid, 'process_ancestry': ancestry,
            'client_port': client_port, 'server_port': server_port}

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def load(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8'))

def write(path, value):
    from mnemos.service import _write_private_json
    _write_private_json(Path(path), value)

def wait_files(paths, timeout=360):
    deadline = time.monotonic() + timeout
    while not all(path.exists() for path in paths):
        if time.monotonic() >= deadline:
            raise TimeoutError('Native coordination deadline exceeded: ' + ','.join(p.name for p in paths if not p.exists()))
        time.sleep(0.1)

def native_rows(home, excluded=()):
    selected = []
    for path in sorted((home / 'sessions').rglob('*.jsonl')):
        relative = path.relative_to(home).as_posix()
        if relative in excluded:
            continue
        raw = path.read_bytes()
        for number, line in enumerate(raw.splitlines(keepends=True), 1):
            record = json.loads(line)
            payload = record.get('payload')
            if isinstance(payload, dict) and (record.get('type') == 'session_meta' or payload.get('type') in (
                'user_message', 'agent_message', 'task_complete', 'mcp_tool_call_end', 'custom_tool_call', 'function_call')):
                selected.append({'source': relative, 'source_file_sha256': sha(raw), 'line': number,
                                 'line_sha256': sha(line), 'record': record})
    return selected

def session_summary(rows, key):
    payloads = [r['record']['payload'] for r in rows]
    metas = [r['record']['payload'] for r in rows if r['record']['type'] == 'session_meta']
    users = [p for p in payloads if p.get('type') == 'user_message']
    finals = [p for p in payloads if p.get('type') == 'agent_message' and p.get('phase') == 'final_answer']
    turns = [p for p in payloads if p.get('type') == 'task_complete']
    assert len(metas) == len(users) == len(finals) == len(turns) == 1
    assert not [p for p in payloads if p.get('type') in ('mcp_tool_call_end', 'custom_tool_call', 'function_call')]
    prompt = 'Do not use tools. Reply with exactly these words, without punctuation: ' + WORDS[key]
    assert users[0]['message'] == prompt
    assert finals[0]['message'] == WORDS[key]
    return {'session': metas[0]['id'], 'turn': turns[0]['turn_id'], 'user': prompt, 'assistant': WORDS[key]}

def database_state(fixture):
    result = {}
    if not (fixture / 'memory.sqlite').exists():
        return {'exists': False}
    with sqlite3.connect((fixture / 'memory.sqlite').as_uri() + '?mode=ro', uri=True) as db:
        db.row_factory = sqlite3.Row
        for table in ('memory_chunks', 'episodic_events', 'fact_evidence', 'memory_jobs'):
            rows = [{k: {'bytes_sha256': sha(v)} if isinstance(v, bytes) else v for k, v in dict(row).items()}
                    for row in db.execute('select * from ' + table)]
            result[table] = {'rows': len(rows), 'sha256': sha(json.dumps(sorted(json.dumps(row, sort_keys=True) for row in rows)).encode())}
        result['capture_count'] = db.execute("select count(*) from episodic_events where source='codex'").fetchone()[0]
    return result

def verify_captures(fixture, sessions, project_id, foreign_id, foreign_content):
    from mnemos.adapters.codex import CodexAdapter
    from mnemos.scopes import ScopeContext
    captures = []
    with sqlite3.connect((fixture / 'memory.sqlite').as_uri() + '?mode=ro', uri=True) as db:
        db.row_factory = sqlite3.Row
        rows = db.execute("select * from episodic_events where source='codex'").fetchall()
        assert len(rows) == 2 * len(sessions)
        assert {(r['session_id'], r['role']) for r in rows} == {(s, role) for s in sessions for role in ('user', 'assistant')}
        jobs = [dict(r) for r in db.execute('select * from memory_jobs')]
        for row in rows:
            meta = json.loads(row['metadata'])
            session = sessions[row['session_id']]
            role = row['role']
            assert row['content'] == session[role]
            assert row['scope_id'] == meta['scope_id'] == meta['project_id'] == project_id
            assert meta['user_profile_id'] == PROFILE and meta['host'] == 'codex'
            assert meta['session_id'] == row['session_id'] and meta['source_kind'] == 'codex_hook'
            event = 'UserPromptSubmit' if role == 'user' else 'Stop'
            assert meta['source_ref'] == event + ':' + session['turn']
            context = ScopeContext(user_profile_id=PROFILE, project_id=project_id,
                workspace_id=meta.get('workspace_id'), session_id=row['session_id'], host='codex')
            source_id = CodexAdapter._event_id(context, event=event, turn_id=session['turn'], role=role, content=row['content'])
            assert meta['source_event_id'] == source_id
            evidence = [dict(r) for r in db.execute('select * from fact_evidence where source_event_id=?', (row['id'],))]
            matching = [j for j in jobs if json.loads(j['payload']).get('event_id') == row['id']]
            assert len(evidence) == len(matching) == 1
            assert matching[0]['status'] == 'completed'
            fact = db.execute('select content,metadata from memory_chunks where id=?', (evidence[0]['fact_id'],)).fetchone()
            fm = json.loads(fact['metadata'])
            assert fm['user_profile_id'] == PROFILE and fm['scope_id'] == project_id
            captures.append({'event': dict(row), 'evidence': evidence[0], 'job_id': matching[0]['id'], 'source_event_id': source_id})
        foreign = db.execute('select content,metadata from memory_chunks where id=?', (foreign_id,)).fetchone()
        assert foreign['content'] == foreign_content
        assert json.loads(foreign['metadata'])['scope_id'] != project_id
    return captures
