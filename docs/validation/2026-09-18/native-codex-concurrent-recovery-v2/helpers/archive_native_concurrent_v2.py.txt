def dump(path, value):
    path.write_text(json.dumps(value, indent=2) + '\n', encoding='utf-8')

def safe_report(value):
    """Retain synthetic messages and selected session identity, not base instructions."""
    value = copy.deepcopy(value)
    for worker in value.get('workers', {}).values():
        selected = []
        for row in worker['native_rows']:
            record = row['record']
            payload = record['payload']
            if record['type'] == 'session_meta':
                record['payload'] = {k: payload[k] for k in ('id', 'timestamp', 'cwd', 'originator', 'cli_version', 'model_provider') if k in payload}
                row['source_record_reduced'] = True
            elif payload.get('type') == 'agent_message' and payload.get('phase') != 'final_answer':
                continue
            selected.append(row)
        worker['native_rows'] = selected
    value['archive_selection'] = 'Session metadata reduced to identity/version fields; original line and file hashes retained. Synthetic user/final records unchanged. Original local report is preserved separately and hashed by verification.json.'
    return value
