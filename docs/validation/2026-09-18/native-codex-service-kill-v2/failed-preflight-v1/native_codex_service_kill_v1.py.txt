def database_state(fixture):
    result={}
    path=fixture/'memory.sqlite'
    if not path.exists(): return {'exists':False}
    with sqlite3.connect(path.as_uri()+'?mode=ro',uri=True) as db:
        db.row_factory=sqlite3.Row
        for table in ('memory_chunks','episodic_events','fact_evidence','memory_jobs'):
            values=[]
            for row in db.execute('select * from '+table):
                item={key:({'bytes_sha256':sha(value)} if isinstance(value,bytes) else value) for key,value in dict(row).items()}
                values.append(json.dumps(item,sort_keys=True))
            result[table]={'rows':len(values),'sha256':sha(json.dumps(sorted(values)).encode())}
        result['capture_count']=db.execute("select count(*) from episodic_events where source='codex'").fetchone()[0]
    return result

def read_native(home):
    selected=[]
    for path in sorted((home/'sessions').rglob('*.jsonl')):
        raw=path.read_bytes()
        for number,line in enumerate(raw.splitlines(keepends=True),1):
            record=json.loads(line); payload=record.get('payload')
            if isinstance(payload,dict) and (record.get('type')=='session_meta' or payload.get('type') in ('user_message','agent_message','mcp_tool_call_end','task_complete','custom_tool_call','function_call')):
                selected.append({'source':path.relative_to(home).as_posix(),'source_file_sha256':sha(raw),'line':number,'line_sha256':sha(line),'record':record})
    return selected

    kill_stop.set()
    if kill_thread is not None:
        kill_thread.join(timeout=12)
    if crash_child is not None and crash_child.poll() is None:
        crash_child.kill()
        crash_child.wait(timeout=10)
    for handle in crash_logs:
        handle.close()
    report['service_kill']=kill_observation
    if crash_child is not None:
        report['crash_service_final_returncode']=crash_child.poll()
    # Clean only this dead child's stale descriptor if recovery did not start.
    descriptor=fixture/'runtime.json'
    if service is None and crash_child is not None and crash_child.poll() is not None and descriptor.exists():
        value=json.loads(descriptor.read_text(encoding='utf-8'))
        if value.get('pid')==crash_child.pid:
            report['cleanup_stale_descriptor_sha256']=sha(descriptor.read_bytes())
            descriptor.unlink()
    errors = lifecycle._cleanup_native_resources(process, auth)
    errors += lifecycle._close_service(service, store)
    report["cleanup"] = {
        "errors": errors,
        "auth_absent": auth is None or not auth.exists(),
        "descriptor_absent": not (fixture / "runtime.json").exists(),
    }
    if errors or not report["cleanup"]["auth_absent"] or not report["cleanup"]["descriptor_absent"]:
        report["status"] = "failed"
    report['final_database']=database_state(fixture)
    report["finished_at"] = datetime.now(timezone.utc).isoformat()
    (OUTPUT / "report.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"status": report["status"], "report": str(OUTPUT / "report.json")}), flush=True)

raise SystemExit(0 if report["status"] == "observed_native_service_kill_recovery" else 1)
