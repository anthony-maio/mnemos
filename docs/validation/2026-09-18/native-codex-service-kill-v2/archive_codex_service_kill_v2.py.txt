def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def sha(path):
    return digest(path.read_bytes())

def load(path):
    return json.loads(path.read_text(encoding='utf-8'))

def dump(path, value):
    path.write_text(json.dumps(value, indent=2) + '\n', encoding='utf-8')

sessions = {}

    raw = path.read_bytes()
    records = [json.loads(line) for line in raw.splitlines()]

    assert len(meta) == 1
    events = [x['payload'] for x in records if isinstance(x.get('payload'), dict)]
    assert not [x for x in events if x.get('type') in ('function_call', 'custom_tool_call', 'mcp_tool_call_end')]
    users = [x for x in events if x.get('type') == 'user_message']
    finals = [x for x in events if x.get('type') == 'agent_message' and x.get('phase') == 'final_answer']
    turns = [x['turn_id'] for x in events if x.get('type') == 'task_complete']
    assert len(users) == len(finals) == len(turns) == 1
    sessions[meta[0]['id']] = {'turn': turns[0], 'user': users[0]['message'], 'assistant': finals[0]['message']}

before_sha = sha(db_path)
captures = []
snapshot = {}
with sqlite3.connect(db_path.as_uri() + '?mode=ro', uri=True) as db:
    db.row_factory = sqlite3.Row
    rows = db.execute("select * from episodic_events where source='codex'").fetchall()
    assert len(rows) == 4
    assert {(row['session_id'], row['role']) for row in rows} == {(s, role) for s in sessions for role in ('user', 'assistant')}
    for row in rows:
        meta = json.loads(row['metadata'])
        session = sessions[row['session_id']]
        assert row['content'] == session[row['role']]
        assert meta['session_id'] == row['session_id']
        assert row['scope_id'] == meta['scope_id'] == meta['project_id'] == r['target']['scope_id']
        assert meta['user_profile_id'] == PROFILE and meta['scope'] == 'project'
        assert meta['host'] == 'codex' and meta['source_kind'] == 'codex_hook'
        event = 'UserPromptSubmit' if row['role'] == 'user' else 'Stop'
        assert meta['source_ref'] == event + ':' + session['turn']
        identity = [PROFILE, meta['project_id'], meta.get('workspace_id'), row['session_id'], event, session['turn'], row['role'], row['content']]
        expected = 'codex:' + digest(json.dumps(identity, ensure_ascii=False, separators=(',', ':')).encode())
        assert meta['source_event_id'] == expected
        if row['session_id'] == r['outage_session_id']:
            entry = next(x for x in pending if x['payload']['role'] == row['role'])
            payload = entry['payload']
            assert row['id'] == entry['event_id'] and entry['operation'] == 'capture'
            assert payload['event_id'] == expected and payload['content'] == row['content']
            assert payload['context']['session_id'] == row['session_id']
            assert payload['context']['project_id'] == meta['project_id']
            assert payload['context']['user_profile_id'] == PROFILE
            assert payload['context']['host'] == 'codex' and payload['source_ref'] == meta['source_ref']
        evidence = [dict(e) for e in db.execute('select * from fact_evidence where source_event_id=?', (row['id'],))]
        assert len(evidence) == 1 and evidence[0]['source_ref'] == meta['source_ref']
        fact = db.execute('select metadata from memory_chunks where id=?', (evidence[0]['fact_id'],)).fetchone()
        fm = json.loads(fact['metadata'])
        assert fm['user_profile_id'] == PROFILE and fm['scope_id'] == r['target']['scope_id']
        captures.append({'id': row['id'], 'role': row['role'], 'content_sha256': digest(row['content'].encode()), 'metadata': meta, 'evidence': evidence})
    foreign = db.execute('select content,metadata from memory_chunks where id=?', (r['target']['foreign_id'],)).fetchone()
    fm = json.loads(foreign['metadata'])
    assert fm['project_id'] == fm['scope_id'] == foreign_id
    assert foreign['content'] == 'The project shutdown probe uses foreign-canary-849 release labels.'
    all_jobs=[dict(j) for j in db.execute('select * from memory_jobs')]
    for capture in captures:
        linked=[j for j in all_jobs if json.loads(j['payload']).get('event_id')==capture['id']]
        assert len(linked)==1 and linked[0]['status']=='completed'
        capture['job_id']=linked[0]['id']
    retained=db.execute('select * from episodic_events where id=?',(committed['id'],)).fetchone()
    assert retained['content']==committed['content'] and retained['session_id']==committed['session_id']
    assert retained['metadata']==committed['metadata']
    current_job=next(j for j in all_jobs if j['id']==old_jobs[0]['id'])
    assert current_job['payload']==old_jobs[0]['payload'] and current_job['created_at']==old_jobs[0]['created_at']
    assert next(c for c in captures if c['id']==committed['id'])['job_id']==old_jobs[0]['id']
    for table in ('memory_chunks', 'episodic_events', 'fact_evidence', 'memory_jobs'):
        values = []
        for row in db.execute('select * from ' + table):
            item = {k: ({'bytes_sha256': digest(v)} if isinstance(v, bytes) else v) for k, v in dict(row).items()}
            values.append(json.dumps(item, sort_keys=True))
        snapshot[table] = {'rows': len(values), 'sha256': digest(json.dumps(sorted(values)).encode())}
    snapshot['capture_count'] = len(rows)
assert snapshot == r['final_database'] and sha(db_path) == before_sha
