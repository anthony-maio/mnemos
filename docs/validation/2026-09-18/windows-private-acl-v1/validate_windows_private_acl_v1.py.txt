"""Record immutable inputs and full source validation of Windows private ACL protection."""
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
OUTPUT = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-windows-private-acl-v1')
FIXTURES = Path('C:/Users/Ant/AppData/Local/Temp/mnemos-windows-private-acl-fixtures-v1')
WEBSITE = Path('D:/Development/mnemos/web/index.html')
OUTPUT.mkdir(exist_ok=False)
FIXTURES.mkdir(exist_ok=False)
guard = subprocess.run(['git', '-C', str(FIXTURES), 'rev-parse', '--show-toplevel'], capture_output=True, text=True)
assert guard.returncode != 0, 'Fixtures must be outside Git'
sys.path.insert(0, str(ROOT))
import mnemos
assert Path(mnemos.__file__).resolve() == ROOT / 'mnemos/__init__.py'
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def inputs():
    paths = sorted((ROOT / 'mnemos').rglob('*.py')) + sorted((ROOT / 'tests').rglob('*.py'))
    paths += [ROOT / 'mnemos/ui/app.js', ROOT / 'pyproject.toml']
    return {p.relative_to(ROOT).as_posix(): sha(p) for p in paths}
before = inputs()
(OUTPUT / 'source.patch').write_bytes(subprocess.check_output(['git', 'diff', '--binary', 'HEAD'], cwd=ROOT))
report = {'status': 'running', 'started_at': datetime.now(timezone.utc).isoformat(),
          'base_commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip(),
          'source_inputs_before': before, 'runner_sha256': sha(Path(__file__)),
          'package_origin': str(Path(mnemos.__file__).resolve()),
          'fixtures': str(FIXTURES), 'fixture_git_discovery_exit_code': guard.returncode,
          'website_sha256_before': sha(WEBSITE), 'checks': []}
env = {k: v for k, v in os.environ.items() if not k.startswith('MNEMOS_') and k != 'PYTHONPATH'}
env.update(TEMP=str(FIXTURES), TMP=str(FIXTURES), TMPDIR=str(FIXTURES), MNEMOS_WEBSITE_INDEX=str(WEBSITE))
python = str(ROOT / '.venv/Scripts/python.exe')
checks = [
    ('black', [python, '-m', 'black', '--check', 'mnemos', 'tests/test_windows_private_paths.py']),
    ('mypy', [str(ROOT / '.venv/compat-py310/Scripts/python.exe'), '-m', 'mypy', 'mnemos']),
    ('javascript', ['node', '--check', 'mnemos/ui/app.js']),
    ('pytest', [python, '-m', 'pytest', '-q', 'tests', '--basetemp', str(FIXTURES / 'pytest'), '--junitxml', str(OUTPUT / 'pytest.xml')]),
]
def save():
    (OUTPUT / 'report.json').write_text(json.dumps(report, indent=2) + '\n', encoding='utf-8')
save()
for name, command in checks:
    with (OUTPUT / (name + '.log')).open('wb') as log:
        result = subprocess.run(command, cwd=ROOT, env=env, stdout=log, stderr=subprocess.STDOUT)
    report['checks'].append({'name': name, 'command': command, 'exit_code': result.returncode})
    save()
    print(name, result.returncode, flush=True)
    if result.returncode:
        break
report['source_inputs_after'] = inputs()
report['website_sha256_after'] = sha(WEBSITE)
report['status'] = 'passed' if (len(report['checks']) == len(checks)
    and all(c['exit_code'] == 0 for c in report['checks'])
    and report['source_inputs_before'] == report['source_inputs_after']
    and report['website_sha256_before'] == report['website_sha256_after']) else 'failed'
report['finished_at'] = datetime.now(timezone.utc).isoformat()
save()
print(report['status'], str(OUTPUT), flush=True)
sys.exit(0 if report['status'] == 'passed' else 1)
